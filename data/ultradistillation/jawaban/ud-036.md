(function () {
  'use strict';

  var $ = function (id) { return document.getElementById(id); };

  /* ================= Penyimpanan aman (try/catch) ================= */
  var HKEY = 'invoice_jasa_history_v1';
  var SKEY = 'invoice_jasa_seq_v1';
  var SELLER_KEY = 'invoice_jasa_seller_v1';

  var store = {
    get: function (key, fallback) {
      try {
        var raw = localStorage.getItem(key);
        if (raw === null) return fallback;
        var val = JSON.parse(raw);
        return (val === null || val === undefined) ? fallback : val;
      } catch (e) {
        return fallback;
      }
    },
    set: function (key, value) {
      try {
        localStorage.setItem(key, JSON.stringify(value));
        return true;
      } catch (e) {
        return false;
      }
    }
  };

  /* ================= Util ================= */
  var nf = new Intl.NumberFormat('id-ID');

  function fmtNum(n) { return nf.format(Math.round(Number(n) || 0)); }
  function fmtRp(n) { return 'Rp ' + fmtNum(n); }

  function parseMoney(s) {
    var d = String(s == null ? '' : s).replace(/[^0-9]/g, '');
    return d ? parseInt(d, 10) : 0;
  }

  function esc(s) {
    return String(s == null ? '' : s).replace(/[&<>"']/g, function (c) {
      return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
  }

  function pad(n, len) { return String(n).padStart(len, '0'); }

  function isoLocal(d) {
    return d.getFullYear() + '-' + pad(d.getMonth() + 1, 2) + '-' + pad(d.getDate(), 2);
  }

  /* ================= Terbilang ================= */
  function terbilang(n) {
    n = Math.floor(Math.abs(Number(n) || 0));
    if (n === 0) return 'nol';

    var s = ['', 'satu', 'dua', 'tiga', 'empat', 'lima', 'enam', 'tujuh', 'delapan',
             'sembilan', 'sepuluh', 'sebelas'];

    function h(x) {
      if (x < 12) return s[x];
      if (x < 20) return s[x - 10] + ' belas';
      if (x < 100) return s[Math.floor(x / 10)] + ' puluh' + (x % 10 ? ' ' + s[x % 10] : '');
      if (x < 200) return 'seratus' + (x % 100 ? ' ' + h(x % 100) : '');
      if (x < 1000) return s[Math.floor(x / 100)] + ' ratus' + (x % 100 ? ' ' + h(x % 100) : '');
      if (x < 2000) return 'seribu' + (x % 1000 ? ' ' + h(x % 1000) : '');
      if (x < 1e6) return h(Math.floor(x / 1000)) + ' ribu' + (x % 1000 ? ' ' + h(x % 1000) : '');
      if (x < 1e9) return h(Math.floor(x / 1e6)) + ' juta' + (x % 1e6 ? ' ' + h(x % 1e6) : '');
      if (x < 1e12) return h(Math.floor(x / 1e9)) + ' miliar' + (x % 1e9 ? ' ' + h(x % 1e9) : '');
      return h(Math.floor(x / 1e12)) + ' triliun' + (x % 1e12 ? ' ' + h(x % 1e12) : '');
    }

    return h(n).replace(/\s+/g, ' ').trim();
  }

  function terbilangRupiah(n) {
    var w = terbilang(n) + ' rupiah';
    return w.charAt(0).toUpperCase() + w.slice(1);
  }

  /* ================= State ================= */
  var state = { currentId: null, total: 0 };
  var itemsBody = $('itemsBody');

  /* ================= Nomor invoice ================= */
  function nextNumber() {
    var d = new Date();
    var y = d.getFullYear();
    var m = pad(d.getMonth() + 1, 2);
    var seq = store.get(SKEY, {});
    var key = y + '-' + m;
    var n = (Number(seq[key]) || 0) + 1;
    return 'INV/' + y + '/' + m + '/' + pad(n, 3);
  }

  function bumpSeq(number) {
    var m = /^INV\/(\d{4})\/(\d{2})\/(\d+)$/.exec(String(number || ''));
    if (!m) return;
    var key = m[1] + '-' + m[2];
    var n = parseInt(m[3], 10);
    var seq = store.get(SKEY, {});
    var cur = Number(seq[key]) || 0;
    if (n > cur) {
      seq[key] = n;
      store.set(SKEY, seq);
    }
  }

  function syncSeq() {
    var list = store.get(HKEY, []);
    var seq = store.get(SKEY, {});
    if (!Array.isArray(list)) return;
    list.forEach(function (inv) {
      var m = /^INV\/(\d{4})\/(\d{2})\/(\d+)$/.exec(String((inv && inv.number) || ''));
      if (m) {
        var key = m[1] + '-' + m[2];
        var n = parseInt(m[3], 10);
        if (n > (Number(seq[key]) || 0)) seq[key] = n;
      }
    });
    store.set(SKEY, seq);
  }

  /* ================= Baris item ================= */
  function bindMoneyInput(input) {
    input.addEventListener('input', function () {
      var raw = input.value.replace(/[^0-9]/g, '');
      input.value = raw ? nf.format(parseInt(raw, 10)) : '';
      recalc();
    });
  }

  function addRow(data) {
    data = data || {};
    var tr = document.createElement('tr');
    tr.innerHTML =
      '<td><input class="in cell desc" type="text" placeholder="Deskripsi jasa"></td>' +
      '<td class="num"><input class="in cell qty" type="number" min="0" step="0.01" value="1"></td>' +
      '<td class="num"><input class="in cell price" type="text" inputmode="numeric" placeholder="0"></td>' +
      '<td class="num amount">Rp 0</td>' +

    var descEl = tr.querySelector('.desc');
    var qtyEl = tr.querySelector('.qty');
    var priceEl = tr.querySelector('.price');

    if (data.desc) descEl.value = data.desc;
    if (data.qty !== undefined && data.qty !== null && data.qty !== '') qtyEl.value = data.qty;
    if (data.price) priceEl.value = fmtNum(data.price);

    bindMoneyInput(priceEl);
    qtyEl.addEventListener('input', recalc);

    tr.querySelector('.del').addEventListener('click', function () {
      tr.remove();

      recalc();
    });

    itemsBody.appendChild(tr);
    return tr;
  }

  /* ================= Perhitungan ================= */
  function recalc() {
    var subtotal = 0;
    var rows = itemsBody.querySelectorAll('tr');

    for (var i = 0; i < rows.length; i++) {
      var tr = rows[i];
      var qty = parseFloat(tr.querySelector('.qty').value) || 0;
      var price = parseMoney(tr.querySelector('.price').value);
      var amount = Math.round(qty * price);
      tr.querySelector('.amount').textContent = fmtRp(amount);
      subtotal += amount;
    }

    var discount = parseMoney($('discount').value);
    if (discount > subtotal) discount = subtotal;

    var rate = parseFloat($('taxRate').value);
    if (isNaN(rate) || rate < 0) rate = 0;
    if (rate > 100) rate = 100;

    var base = subtotal - discount;
    var tax = Math.round(base * rate / 100);
    var total = base + tax;

    $('outSubtotal').textContent = fmtRp(subtotal);
    $('outTax').textContent = fmtRp(tax);
    $('outTotal').textContent = fmtRp(total);
    $('outWords').textContent = terbilangRupiah(total);

    state.total = total;
  }

  /* ================= Kumpulkan / terapkan data ================= */
  function collect() {
    var items = [];
    var rows = itemsBody.querySelectorAll('tr');
    for (var i = 0; i < rows.length; i++) {
      var tr = rows[i];
      items.push({
        desc: tr.querySelector('.desc').value,
        qty: tr.querySelector('.qty').value,
        price: parseMoney(tr.querySelector('.price').value)
      });
    }
    return {
      number: $('invNumber').value,
      date: $('invDate').value,
      due: $('invDue').value,
      seller: {
        name: $('sellerName').value,
        addr: $('sellerAddr').value,
        contact: $('sellerContact').value
      },
      buyer: {
        name: $('buyerName').value,
        addr: $('buyerAddr').value,
        contact: $('buyerContact').value
      },
      note: $('note').value,
      items: items,
      discount: $('discount').value,
      taxRate: $('taxRate').value,
      total: state.total
    };
  }

  function apply(data) {
    $('invNumber').value = data.number || '';
    $('invDate').value = data.date || '';
    $('invDue').value = data.due || '';

    var seller = data.seller || {};
    $('sellerName').value = seller.name || '';
    $('sellerAddr').value = seller.addr || '';
    $('sellerContact').value = seller.contact || '';

    var buyer = data.buyer || {};
    $('buyerName').value = buyer.name || '';
    $('buyerAddr').value = buyer.addr || '';
    $('buyerContact').value = buyer.contact || '';

    $('note').value = data.note || '';

    itemsBody.innerHTML = '';
    var items = (data.items && data.items.length) ? data.items : [{}];
    items.forEach(function (it) { addRow(it); });

    $('discount').value = data.discount || '';
    $('taxRate').value = (data.taxRate !== undefined && data.taxRate !== '' && data.taxRate !== null)
      ? data.taxRate : 11;

    recalc();
  }

  function blankInvoice() {
    state.currentId = null;

    var today = new Date();
    var due = new Date(today.getTime() + 14 * 86400000);

    $('invNumber').value = nextNumber();
    $('invDate').value = isoLocal(today);
    $('invDue').value = isoLocal(due);

    $('buyerName').value = '';
    $('buyerAddr').value = '';
    $('buyerContact').value = '';
    $('note').value = '';
    $('discount').value = '';
    $('taxRate').value = 11;

    itemsBody.innerHTML = '';
    addRow();
    recalc();
    renderHistory();
  }

  /* ================= Riwayat ================= */
  function readHistory() {
    var list = store.get(HKEY, []);
    return Array.isArray(list) ? list : [];
  }

  function renderHistory() {
    var box = $('historyList');
    var list = readHistory();
    box.innerHTML = '';

    if (!list.length) {
      var p = document.createElement('p');
      p.className = 'empty';
      p.textContent = 'Belum ada invoice tersimpan.';
      box.appendChild(p);
      return;
    }

    list.forEach(function (inv) {
      var el = document.createElement('div');
      el.className = 'hitem' + (inv.id === state.currentId ? ' active' : '');
      el.innerHTML =
        '<div class="hnum">' + esc(inv.number || '-') + '</div>' +
        '<div class="hbuyer">' + esc((inv.buyer && inv.buyer.name) ? inv.buyer.name : 'Tanpa nama') + '</div>' +
        '<div class="hfoot"><span>' + fmtRp(Number(inv.total) || 0) + '</span><span>' + esc(inv.date || '') + '</span></div>' +
        '<div class="hacts">' +
          '<button type="button" class="btn small" data-act="open" data-id="' + esc(inv.id) + '">Buka</button>' +
          '<button type="button" class="btn small danger" data-act="del" data-id="' + esc(inv.id) + '">Hapus</button>' +
        '</div>';
      box.appendChild(el);
    });
  }

  function saveInvoice() {
    recalc();
    var data = collect();

    if (!data.number) {
      toast('Nomor invoice belum tersedia.');
      return;
    }

    var list = readHistory();
    var now = new Date().toISOString();

    if (state.currentId) {
      var found = false;
      for (var i = 0; i < list.length; i++) {
        if (list[i].id === state.currentId) {
          list[i] = Object.assign({}, list[i], data, { id: state.currentId, savedAt: now });
          found = true;
          break;
        }
      }
      if (!found) {
        list.unshift(Object.assign({}, data, { id: state.currentId, savedAt: now }));
      }
    } else {
      var id = 'inv' + Date.now() + Math.random().toString(16).slice(2, 6);
      state.currentId = id;
      list.unshift(Object.assign({}, data, { id: id, savedAt: now }));
      bumpSeq(data.number);
    }

    var ok = store.set(HKEY, list);
    store.set(SELLER_KEY, data.seller);

    renderHistory();
    toast(ok ? 'Invoice tersimpan.' : 'Gagal menyimpan: penyimpanan browser penuh.');
  }

  function openInvoice(id) {
    var list = readHistory();
    var inv = null;
    for (var i = 0; i < list.length; i++) {
      if (list[i].id === id) { inv = list[i]; break; }
    }
    if (!inv) return;

    state.currentId = inv.id;
    apply(inv);
    renderHistory();
    window.scrollTo({ top: 0, behavior: 'smooth' });
    toast('Invoice dibuka.');
  }

  function deleteInvoice(id) {
    if (!window.confirm('Hapus invoice ini dari riwayat?')) return;

    var list = readHistory().filter(function (x) { return x.id !== id; });
    store.set(HKEY, list);

    if (state.currentId === id) state.currentId = null;

    renderHistory();
    toast('Invoice dihapus dari riwayat.');
  }

  function onHistoryClick(e) {
    var btn = e.target.closest ? e.target.closest('button[data-act]') : null;
    if (!btn) return;
    var id = btn.getAttribute('data-id');
    if (btn.getAttribute('data-act') === 'open') openInvoice(id);
    else deleteInvoice(id);
  }

  /* ================= Toast ================= */
  var toastTimer = null;
  function toast(msg) {
    var el = $('toast');
    el.textContent = msg;
    el.classList.add('show');
    if (toastTimer) clearTimeout(toastTimer);
    toastTimer = setTimeout(function () { el.classList.remove('show'); }, 2200);
  }

  /* ================= Inisialisasi ================= */
  function init() {
    syncSeq();

    bindMoneyInput($('discount'));
    $('taxRate').addEventListener('input', recalc);

    $('btnAdd').addEventListener('click', function () {
      addRow();
      recalc();
    });

    $('btnSave').addEventListener('click', saveInvoice);

    $('btnPrint').addEventListener('click', function () {
      recalc();
      window.print();
    });

    $('btnNew').addEventListener('click', function () {
      if (!window.confirm('Mulai invoice baru? Perubahan yang belum disimpan akan hilang.')) return;
      blankInvoice();
      window.scrollTo({ top: 0, behavior: 'smooth' });
    });

    $('historyList').addEventListener('click', onHistoryClick);

    var seller = store.get(SELLER_KEY, null);
    if (seller && typeof seller === 'object') {
      $('sellerName').value = seller.name || '';
      $('sellerAddr').value = seller.addr || '';
      $('sellerContact').value = seller.contact || '';
    }

    blankInvoice();
  }

  init();
})();
</script>
</body>
</html>

**Cek kebutuhan**

- **Data penjual dan pembeli** → input `#sellerName`, `#sellerAddr`, `#sellerContact` pada `.brand`, serta `#buyerName`, `#buyerAddr`, `#buyerContact` pada `.parties`
- **Nomor invoice otomatis format INV/tahun/bulan/urut** → fungsi `nextNumber()` menghasilkan `INV/YYYY/MM/NNN`, ditampilkan di `#invNumber` (readonly) dan disinkronkan lewat `syncSeq()` + `bumpSeq()`
- **Tanggal dan jatuh tempo** → input `#invDate` dan `#invDue`, diisi otomatis di `blankInvoice()` (jatuh tempo +14 hari)
- **Daftar item (deskripsi, jumlah, harga satuan)** → `addRow()` membangun baris tabel dengan `.desc`, `.qty`, `.price`, dan kolom `.amount`
- **Diskon** → input `#discount` dengan pemformat ribuan `bindMoneyInput()`, dibatasi maksimal sebesar subtotal di `recalc()`
- **Persentase pajak yang bisa diubah** → input `#taxRate` (default 11, `min=0 max=100 step=0.1`), dipakai di `recalc()`
- **Subtotal, pajak, total format rupiah** → `recalc()` mengisi `#outSubtotal`, `#outTax`, `#outTotal` melalui `fmtRp()` berbasis `Intl.NumberFormat('id-ID')`
- **Total dalam huruf (terbilang)** → `terbilang()` + `terbilangRupiah()`, hasilnya ditulis ke `#outWords`
- **Tampilan cetak rapi A4 lewat `window.print()`** → tombol `#btnPrint` memanggil `window.print()`; aturan `@media print` dengan `@page { size: A4; margin: 14mm }` dan penyembunyian `.no-print`
- **Riwayat invoice tersimpan di localStorage dan bisa dibuka kembali** → `saveInvoice()`, `renderHistory()`, `openInvoice()`, `deleteInvoice()` dengan kunci `invoice_jasa_history_v1`
- **Akses localStorage dibungkus try/catch** → objek `store.get()` dan `store.set()`
- **Responsif sampai 375 px** → media query pada `.wrap` (900px), `.sheet-head` (620px), `.parties` (620px), dan `table.items` (520px) dengan lebar kolom mengecil
- **Tanpa library, font, gambar, atau file luar** → seluruh CSS dan JavaScript inline, hanya memakai API bawaan browser (`Intl`, `localStorage`)
- **Tanpa error JavaScript** → kode dibungkus IIFE `'use strict'`, seluruh `getElementById` dijamin ada di markup, dan `e.target.closest` dijaga dengan pemeriksaan

**Catatan teknis**

Nomor urut disimpan per periode bulan pada kunci `invoice_jasa_seq_v1` dan disinkronkan ulang dari riwayat saat halaman dimuat, sehingga urutan tidak bentrok meski riwayat dihapus sebagian. Perhitungan dibulatkan per baris item lalu per pajak agar subtotal, pajak, dan total selalu konsisten dengan angka pada terbilang. Data penjual ikut tersimpan terpisah agar otomatis terisi pada invoice berikutnya. Saat mencetak, seluruh input menjadi borderless, kolom hapus dan tombol aksi disembunyikan, dan warna blok terbilang dipaksa ikut tercetak lewat `print-color-adjust`.
