```html
<script>
(function () {
  'use strict';

  var KEY = 'buku-utang-warung-v1';
  var data = { pelanggan: [] };
  var idAktif = null;

  var el = {
    daftar: document.getElementById('daftar'),
    cari: document.getElementById('cari'),
    total: document.getElementById('totalPiutang'),
    formPelanggan: document.getElementById('formPelanggan'),
    inNama: document.getElementById('inNama'),
    inTelepon: document.getElementById('inTelepon'),
    formTransaksi: document.getElementById('formTransaksi'),
    inJenis: document.getElementById('inJenis'),
    inTanggal: document.getElementById('inTanggal'),
    inJumlah: document.getElementById('inJumlah'),
    inKeterangan: document.getElementById('inKeterangan'),
    judulTransaksi: document.getElementById('judulTransaksi'),
    judulRiwayat: document.getElementById('judulRiwayat'),
    isiRiwayat: document.getElementById('isiRiwayat')
  };

  /* ---------- utilitas ---------- */

  function uid() {
    return Date.now().toString(36) + Math.random().toString(36).slice(2, 7);
  }

  function esc(s) {
    return String(s == null ? '' : s).replace(/[&<>"']/g, function (c) {
      return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
  }

  function rp(n) {
    return 'Rp ' + Math.abs(Math.round(n)).toLocaleString('id-ID');
  }

  function fmtTanggal(iso) {
    if (!/^\d{4}-\d{2}-\d{2}$/.test(iso || '')) return iso || '-';
    var p = iso.split('-');
    return p[2] + '/' + p[1] + '/' + p[0];
  }

  function hariIni() {
    var d = new Date();
    var pad = function (n) { return String(n).padStart(2, '0'); };
    return d.getFullYear() + '-' + pad(d.getMonth() + 1) + '-' + pad(d.getDate());
  }

  function slug(s) {
    return String(s).toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '') || 'pelanggan';
  }

  function saldo(p) {
    return p.transaksi.reduce(function (a, t) {
      return a + (t.jenis === 'utang' ? t.jumlah : -t.jumlah);
    }, 0);
  }

  function totalPiutang() {
    return data.pelanggan.reduce(function (a, p) {
      var s = saldo(p);
      return a + (s > 0 ? s : 0);
    }, 0);
  }

  function cariPelanggan(id) {
    for (var i = 0; i < data.pelanggan.length; i++) {
      if (data.pelanggan[i].id === id) return data.pelanggan[i];
    }
    return null;
  }

  /* ---------- penyimpanan ---------- */

  function muat() {
    try {
      var raw = localStorage.getItem(KEY);
      if (!raw) return;
      var parsed = JSON.parse(raw);
      if (parsed && Array.isArray(parsed.pelanggan)) {
        data.pelanggan = parsed.pelanggan.map(function (p) {
          return {
            id: p && p.id ? String(p.id) : uid(),
            nama: p && p.nama ? String(p.nama) : 'Tanpa Nama',
            telepon: p && p.telepon ? String(p.telepon) : '',
            transaksi: Array.isArray(p && p.transaksi) ? p.transaksi.map(function (t) {
              return {
                id: t && t.id ? String(t.id) : uid(),
                tanggal: t && t.tanggal ? String(t.tanggal) : hariIni(),
                jenis: t && t.jenis === 'bayar' ? 'bayar' : 'utang',
                jumlah: Math.max(0, Math.round(Number(t && t.jumlah) || 0)),
                keterangan: t && t.keterangan ? String(t.keterangan) : ''
              };
            }) : []
          };
        });
      }
    } catch (e) {
      console.warn('Gagal memuat data dari localStorage:', e);
    }
  }

  function simpan() {
    try {
      localStorage.setItem(KEY, JSON.stringify(data));
    } catch (e) {
      console.warn('Gagal menyimpan data ke localStorage:', e);
      alert('Data gagal disimpan di penyimpanan lokal browser ini.');
    }
  }

  /* ---------- tampilan ---------- */

  function kartuHTML(p) {
    var s = saldo(p);
    var kelas = s > 0 ? 'utang' : (s < 0 ? 'lebih' : 'lunas');
    var label = s > 0 ? rp(s) : (s < 0 ? rp(s) + ' (lebih)' : 'Lunas');
    return '' +
      '<article class="kartu">' +
        '<div class="kartu-atas">' +
          '<div class="identitas">' +
            '<h3>' + esc(p.nama) + '</h3>' +
            (p.telepon ? '<p class="muted">' + esc(p.telepon) + '</p>' : '') +
          '</div>' +
          '<div class="saldo ' + kelas + '"><span>Saldo</span><strong>' + label + '</strong></div>' +
        '</div>' +
        '<div class="aksi">' +
          '<button class="btn kecil primary" type="button" data-aksi="utang" data-id="' + esc(p.id) + '">+ Utang</button>' +
          '<button class="btn kecil" type="button" data-aksi="bayar" data-id="' + esc(p.id) + '">Bayar</button>' +
          '<button class="btn kecil" type="button" data-aksi="riwayat" data-id="' + esc(p.id) + '">Riwayat</button>' +
          '<button class="btn kecil danger" type="button" data-aksi="hapus" data-id="' + esc(p.id) + '">Hapus</button>' +
        '</div>' +
      '</article>';
  }

  function render() {
    var q = el.cari.value.trim().toLowerCase();
    var urut = data.pelanggan.slice().sort(function (a, b) {
      return a.nama.localeCompare(b.nama, 'id');
    });
    var tampil = urut.filter(function (p) {
      if (!q) return true;
      return p.nama.toLowerCase().indexOf(q) !== -1 ||
             (p.telepon || '').toLowerCase().indexOf(q) !== -1;
    });

    el.total.textContent = rp(totalPiutang());

    if (!tampil.length) {
      el.daftar.innerHTML = '<p class="kosong">' +
        (data.pelanggan.length
          ? 'Tidak ada pelanggan yang cocok dengan pencarian.'
          : 'Belum ada pelanggan. Tambahkan pelanggan pertama di atas.') +
        '</p>';
      return;
    }

    el.daftar.innerHTML = tampil.map(kartuHTML).join('');
  }

  /* ---------- modal ---------- */

  function buka(id) {
    document.getElementById(id).hidden = false;
    document.body.classList.add('no-scroll');
  }

  function tutup(id) {
    document.getElementById(id).hidden = true;
    if (!document.querySelector('.overlay:not([hidden])')) {
      document.body.classList.remove('no-scroll');
    }
  }

  function bukaTransaksi(id, jenis) {
    var p = cariPelanggan(id);
    if (!p) return;
    idAktif = id;
    el.judulTransaksi.textContent = (jenis === 'utang' ? 'Catat Utang' : 'Catat Pembayaran') + ' — ' + p.nama;
    el.inJenis.value = jenis;
    el.inTanggal.value = hariIni();
    el.inJumlah.value = '';
    el.inKeterangan.value = '';
    buka('overlayTransaksi');
    el.inJumlah.focus();
  }

  function bukaRiwayat(id) {
    var p = cariPelanggan(id);
    if (!p) return;
    idAktif = id;
    el.judulRiwayat.textContent = 'Riwayat — ' + p.nama;

    var trx = p.transaksi.slice().sort(function (a, b) {
      var c = b.tanggal.localeCompare(a.tanggal);
      return c !== 0 ? c : b.id.localeCompare(a.id);
    });

    if (!trx.length) {
      el.isiRiwayat.innerHTML = '<p class="kosong">Belum ada transaksi untuk pelanggan ini.</p>';
    } else {
      var s = saldo(p);
      var saldoTeks = s === 0 ? 'Lunas' : rp(s) + (s < 0 ? ' (lebih)' : '');
      el.isiRiwayat.innerHTML =
        '<ul class="riwayat">' +
        trx.map(function (t) {
          var utang = t.jenis === 'utang';
          return '' +
            '<li class="riwayat-item">' +
              '<div class="riwayat-info">' +
                '<span class="tag ' + (utang ? 'utang' : 'bayar') + '">' + (utang ? 'Utang' : 'Bayar') + '</span>' +
                '<span class="muted">' + fmtTanggal(t.tanggal) + '</span>' +
                (t.keterangan ? '<p>' + esc(t.keterangan) + '</p>' : '') +
              '</div>' +
              '<div class="riwayat-kanan">' +
                '<span class="nominal ' + (utang ? 'utang' : 'bayar') + '">' +
                  (utang ? '+' : '−') + rp(t.jumlah) +
                '</span>' +
                '<button class="btn mini danger" type="button" data-hapus="' + esc(t.id) + '" data-pelanggan="' + esc(p.id) + '">Hapus</button>' +
              '</div>' +
            '</li>';
        }).join('') +
        '</ul>' +
        '<p class="saldo-akhir">Saldo saat ini: <strong>' + saldoTeks + '</strong></p>';
    }

    buka('overlayRiwayat');
  }

  /* ---------- ekspor CSV ---------- */

  function kutip(v) {
    var s = String(v == null ? '' : v);
    return /[",\r\n]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s;
  }

  function unduhCSV(namaFile, baris) {
    var csv = baris.map(function (r) {
      return r.map(kutip).join(',');
    }).join('\r\n');

    var blob = new Blob(['﻿' + csv], { type: 'text/csv;charset=utf-8;' });
    var url = URL.createObjectURL(blob);
    var a = document.createElement('a');
    a.href = url;
    a.download = namaFile;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    setTimeout(function () { URL.revokeObjectURL(url); }, 1500);
  }

  function eksporPelanggan() {
    var p = cariPelanggan(idAktif);
    if (!p) return;
    if (!p.transaksi.length) {
      alert('Belum ada transaksi untuk diekspor.');
      return;
    }
    var baris = [['Tanggal', 'Jenis', 'Jumlah', 'Keterangan']];
    p.transaksi.slice().sort(function (a, b) {
      return a.tanggal.localeCompare(b.tanggal);
    }).forEach(function (t) {
      baris.push([t.tanggal, t.jenis === 'utang' ? 'Utang' : 'Pembayaran', t.jumlah, t.keterangan || '']);
    });
    baris.push([]);
    baris.push(['Saldo', '', saldo(p), '']);
    unduhCSV('riwayat-' + slug(p.nama) + '.csv', baris);
  }

  function eksporSemua() {
    var baris = [['Pelanggan', 'Tanggal', 'Jenis', 'Jumlah', 'Keterangan']];
    var ada = false;
    data.pelanggan.forEach(function (p) {
      p.transaksi.slice().sort(function (a, b) {
        return a.tanggal.localeCompare(b.tanggal);
      }).forEach(function (t) {
        ada = true;
        baris.push([p.nama, t.tanggal, t.jenis === 'utang' ? 'Utang' : 'Pembayaran', t.jumlah, t.keterangan || '']);
      });
    });
    if (!ada) {
      alert('Belum ada transaksi untuk diekspor.');
      return;
    }
    baris.push([]);
    baris.push(['Total Piutang Warung', '', '', totalPiutang(), '']);
    unduhCSV('buku-utang-warung-' + hariIni() + '.csv', baris);
  }

  /* ---------- interaksi ---------- */

  el.formPelanggan.addEventListener('submit', function (e) {
    e.preventDefault();
    var nama = el.inNama.value.trim();
    var telepon = el.inTelepon.value.trim();
    if (!nama) {
      alert('Nama pelanggan wajib diisi.');
      return;
    }
    var dobel = data.pelanggan.some(function (p) {
      return p.nama.toLowerCase() === nama.toLowerCase();
    });
    if (dobel) {
      alert('Pelanggan dengan nama "' + nama + '" sudah ada.');
      return;
    }
    data.pelanggan.push({ id: uid(), nama: nama, telepon: telepon, transaksi: [] });
    simpan();
    el.formPelanggan.reset();
    el.inNama.focus();
    render();
  });

  el.cari.addEventListener('input', render);

  el.daftar.addEventListener('click', function (e) {
    var btn = e.target.closest('button[data-aksi]');
    if (!btn) return;
    var id = btn.getAttribute('data-id');
    var aksi = btn.getAttribute('data-aksi');
    var p = cariPelanggan(id);
    if (!p) return;

    if (aksi === 'utang') {
      bukaTransaksi(id, 'utang');
    } else if (aksi === 'bayar') {
      bukaTransaksi(id, 'bayar');
    } else if (aksi === 'riwayat') {
      bukaRiwayat(id);
    } else if (aksi === 'hapus') {
      if (confirm('Hapus pelanggan "' + p.nama + '" beserta seluruh riwayat transaksinya?')) {
        data.pelanggan = data.pelanggan.filter(function (x) { return x.id !== id; });
        simpan();
        render();
      }
    }
  });

  el.formTransaksi.addEventListener('submit', function (e) {
    e.preventDefault();
    var p = cariPelanggan(idAktif);
    if (!p) {
      alert('Pelanggan tidak ditemukan.');
      tutup('overlayTransaksi');
      return;
    }
    var jenis = el.inJenis.value === 'bayar' ? 'bayar' : 'utang';
    var tanggal = el.inTanggal.value;
    var jumlah = Math.round(Number(el.inJumlah.value));

    if (!tanggal) {
      alert('Tanggal wajib diisi.');
      return;
    }
    if (!isFinite(jumlah) || jumlah <= 0) {
      alert('Jumlah harus berupa angka lebih besar dari 0.');
      return;
    }

    p.transaksi.push({
      id: uid(),
      tanggal: tanggal,
      jenis: jenis,
      jumlah: jumlah,
      keterangan: el.inKeterangan.value.trim()
    });

    simpan();
    render();
    tutup('overlayTransaksi');
  });

  el.isiRiwayat.addEventListener('click', function (e) {
    var btn = e.target.closest('button[data-hapus]');
    if (!btn) return;
    var p = cariPelanggan(btn.getAttribute('data-pelanggan'));
    if (!p) return;
    if (!confirm('Hapus transaksi ini?')) return;
    var idTrx = btn.getAttribute('data-hapus');
    p.transaksi = p.transaksi.filter(function (t) { return t.id !== idTrx; });
    simpan();
    render();
    bukaRiwayat(p.id);
  });

  document.getElementById('btnEksporPelanggan').addEventListener('click', eksporPelanggan);
  document.getElementById('btnEksporSemua').addEventListener('click', eksporSemua);

  document.addEventListener('click', function (e) {
    var t = e.target.closest('[data-tutup]');
    if (t) tutup(t.getAttribute('data-tutup'));
  });

  Array.prototype.forEach.call(document.querySelectorAll('.overlay'), function (o) {
    o.addEventListener('mousedown', function (e) {
      if (e.target === o) tutup(o.id);
    });
  });

  document.addEventListener('keydown', function (e) {
    if (e.key !== 'Escape') return;
    Array.prototype.forEach.call(document.querySelectorAll('.overlay'), function (o) {
      if (!o.hidden) tutup(o.id);
    });
  });

  /* ---------- mulai ---------- */

  muat();
  render();
})();
</script>
</body>
</html>
```

**Cek kebutuhan**

- **Menambah pelanggan** → `formPelanggan` + handler `submit` yang mendorong objek `{id, nama, telepon, transaksi}` ke `data.pelanggan`.
- **Mencatat utang baru dan pembayaran per pelanggan dengan tanggal dan keterangan** → `bukaTransaksi()` dengan `jenis` `utang`/`bayar`, form `formTransaksi` berisi `inTanggal`, `inJumlah`, `inKeterangan`.
- **Melihat saldo utang tiap pelanggan** → fungsi `saldo()` dan blok `.saldo` di `kartuHTML()`.
- **Melihat total piutang warung** → `totalPiutang()` yang dirender ke `#totalPiutang` pada `render()`.
- **Mencari pelanggan** → input `#cari` dengan listener `input` yang memfilter nama/nomor HP di `render()`.
- **Mengekspor riwayat ke CSV** → `eksporPelanggan()` (per pelanggan) dan `eksporSemua()` (seluruh warung) via `unduhCSV()` dengan Blob + BOM.
- **Data tersimpan di localStorage dan bertahan setelah reload** → `simpan()`, `muat()`, dan pemanggilan `muat(); render();` di akhir skrip.
- **Akses localStorage dibungkus try/catch** → `muat()` dan `simpan()` memakai `try { ... } catch (e) { ... }`.
- **Responsif sampai lebar 375 px** → CSS `width:100%`, `max-width`, `flex-wrap` pada `.topbar-in`, `.aksi`, `.toolbar`, serta breakpoint `@media (min-width:560px)`.
- **Tanpa library/font/gambar eksternal** → seluruh CSS dan JavaScript inline; tidak ada `<link>`, `<script src>`, atau `@import`.
- **Tanpa error JavaScript** → semua elemen diambil via `getElementById` setelah DOM ada, skrip dibungkus IIFE `'use strict'`, dan tidak ada referensi variabel di luar cakupan.

**Catatan teknis**

Tanggal disimpan dalam format `YYYY-MM-DD` agar pengurutan kronologis konsisten, lalu ditampilkan sebagai `DD/MM/YYYY`. Saldo pelanggan dihitung sebagai total utang dikurangi total pembayaran; saldo negatif ditandai "(lebih)" dan tidak ikut dihitung dalam total piutang warung. Berkas CSV memakai pemisah koma dengan BOM UTF-8 agar karakter non-ASCII tampil benar di Excel. Kunci penyimpanan adalah `buku-utang-warung-v1` dan data divalidasi ulang saat dimuat agar struktur rusak tidak membuat halaman gagal jalan.
