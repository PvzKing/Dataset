<script>
(function () {
  'use strict';

  var KEY = 'iuran_rt_v1';
  var MONTHS_SHORT = ['Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun', 'Jul', 'Agu', 'Sep', 'Okt', 'Nov', 'Des'];

  var state;
  var editingId = null;
  var statusTimer = null;
  var gagalSimpan = false;

  /* ---------- utilitas ---------- */

  function uid() {
    return 'w' + Date.now().toString(36) + Math.random().toString(36).slice(2, 7);
  }

  function emptyRow() {
    var a = [];
    for (var i = 0; i < 12; i++) { a.push(false); }
    return a;
  }

  function makePayments(n) {
    var a = emptyRow();
    for (var i = 0; i < n && i < 12; i++) { a[i] = true; }
    return a;
  }

  function fmtMoney(n) {
    return 'Rp ' + Number(n || 0).toLocaleString('id-ID');
  }

  function fmtCompact(n) {
    n = Number(n || 0);
    if (n >= 1000000) { return (n / 1000000).toLocaleString('id-ID', { maximumFractionDigits: 1 }) + ' jt'; }
    if (n >= 1000) { return (n / 1000).toLocaleString('id-ID', { maximumFractionDigits: 0 }) + ' rb'; }
    return String(n);
  }

  function monthsElapsed(year) {
    var now = new Date();
    var cy = now.getFullYear();
    if (year < cy) { return 12; }
    if (year > cy) { return 0; }
    return now.getMonth() + 1;
  }

  function setStatus(msg) {
    var el = document.getElementById('status');
    if (!el) { return; }
    el.textContent = gagalSimpan ? msg + ' Peringatan: perubahan tidak tersimpan karena localStorage diblokir atau penuh.' : msg;
    if (statusTimer) { clearTimeout(statusTimer); }
    statusTimer = setTimeout(function () { el.textContent = ''; }, 4000);
  }

  /* ---------- data contoh & penyimpanan ---------- */

  function defaultState() {
    var y = new Date().getFullYear();
    var cur = new Date().getMonth() + 1;
    var seed = [
      ['Budi Santoso', '01', 12],
      ['Siti Aminah', '02', 2],
      ['Ahmad Fauzi', '03', cur],
      ['Dewi Lestari', '04', Math.max(cur - 2, 0)],
      ['Joko Purnomo', '05', 0],
      ['Rina Marlina', '06', cur]
    ];
    var residents = seed.map(function (s) {
      var p = {};
      p[String(y)] = makePayments(s[2]);
      return { id: uid(), name: s[0], house: s[1], payments: p };
    });
    return { dues: 50000, year: y, residents: residents };
  }

  function load() {
    try {
      var raw = window.localStorage.getItem(KEY);
      if (!raw) { return defaultState(); }
      var parsed = JSON.parse(raw);
      if (!parsed || typeof parsed !== 'object' || !Array.isArray(parsed.residents)) {
        return defaultState();
      }
      var dues = Number(parsed.dues);
      var year = Number(parsed.year);
      return {
        dues: (isFinite(dues) && dues >= 0) ? dues : 50000,
        year: (isFinite(year) && year > 1900 && year < 2200) ? year : new Date().getFullYear(),
        residents: parsed.residents.filter(function (r) {
          return r && typeof r === 'object' && typeof r.id === 'string';
        }).map(function (r) {
          return {
            id: r.id,
            name: String(r.name == null ? '' : r.name),
            house: String(r.house == null ? '' : r.house),
            payments: (r.payments && typeof r.payments === 'object') ? r.payments : {}
          };
        })
      };
    } catch (e) {
      return defaultState();
    }
  }

  function save() {
    try {
      window.localStorage.setItem(KEY, JSON.stringify(state));
      gagalSimpan = false;
    } catch (e) {
      gagalSimpan = true;
    }
  }

  function paymentsOf(r, year) {
    var k = String(year);
    if (!r.payments || typeof r.payments !== 'object') { r.payments = {}; }
    var arr = r.payments[k];
    if (!Array.isArray(arr) || arr.length !== 12) {
      var fresh = emptyRow();
      if (Array.isArray(arr)) {
        for (var i = 0; i < 12 && i < arr.length; i++) { fresh[i] = arr[i] === true; }
      }
      r.payments[k] = fresh;
      arr = fresh;
    }
    return arr;
  }

  function findResident(id) {
    for (var i = 0; i < state.residents.length; i++) {
      if (state.residents[i].id === id) { return state.residents[i]; }
    }
    return null;
  }

  /* ---------- render tabel ---------- */

  function updateRowTotal(tr, r) {
    var pay = paymentsOf(r, state.year);
    var n = 0;
    for (var i = 0; i < 12; i++) { if (pay[i] === true) { n++; } }
    var cell = tr.querySelector('[data-role="rowtotal"]');
    if (cell) {
      cell.textContent = n + '/12';
      cell.title = fmtMoney(n * state.dues);
    }
  }

  function buildRow(r) {
    var pay = paymentsOf(r, state.year);
    var tr = document.createElement('tr');
    tr.setAttribute('data-id', r.id);

    var tdName = document.createElement('td');
    tdName.className = 'sticky nm';
    tdName.textContent = r.name;
    tdName.title = r.name;
    tr.appendChild(tdName);

    var tdHouse = document.createElement('td');
    tdHouse.className = 'c';
    tdHouse.textContent = r.house;
    tr.appendChild(tdHouse);

    for (var i = 0; i < 12; i++) {
      (function (idx) {
        var td = document.createElement('td');
        td.className = 'c';
        var cb = document.createElement('input');
        cb.type = 'checkbox';
        cb.checked = (pay[idx] === true);
        cb.setAttribute('aria-label', 'Iuran ' + MONTHS_SHORT[idx] + ' untuk ' + r.name);
        cb.addEventListener('change', function () {
          pay[idx] = cb.checked;
          save();
          updateRowTotal(tr, r);
          renderSummary();
          setStatus('Status iuran ' + MONTHS_SHORT[idx] + ' ' + r.name + ' diperbarui.');
        });
        td.appendChild(cb);
        tr.appendChild(td);
      })(i);
    }

    var tdTotal = document.createElement('td');
    tdTotal.className = 'c total';
    tdTotal.setAttribute('data-role', 'rowtotal');
    tr.appendChild(tdTotal);
    updateRowTotal(tr, r);

    var tdAct = document.createElement('td');
    tdAct.className = 'c';
    var btnEdit = document.createElement('button');
    btnEdit.type = 'button';
    btnEdit.className = 'mini ghost';
    btnEdit.textContent = 'Ubah';
    btnEdit.addEventListener('click', function () { startEdit(r.id); });
    var btnDel = document.createElement('button');
    btnDel.type = 'button';
    btnDel.className = 'mini danger';
    btnDel.textContent = 'Hapus';
    btnDel.style.marginLeft = '6px';
    btnDel.addEventListener('click', function () { removeResident(r.id); });
    tdAct.appendChild(btnEdit);
    tdAct.appendChild(btnDel);
    tr.appendChild(tdAct);

    return tr;
  }

  function renderTable() {
    var tbody = document.getElementById('tbody');
    tbody.innerHTML = '';
    if (state.residents.length === 0) {
      var tr = document.createElement('tr');
      var td = document.createElement('td');
      td.colSpan = 16;
      td.className = 'empty';
      td.textContent = 'Belum ada data warga. Tambahkan melalui formulir di atas.';
      tr.appendChild(td);
      tbody.appendChild(tr);
      return;
    }
    for (var i = 0; i < state.residents.length; i++) {
      tbody.appendChild(buildRow(state.residents[i]));
    }
  }

  /* ---------- render ringkasan & tunggakan ---------- */

  function renderSummary() {
    var counts = [];
    var i;
    for (i = 0; i < 12; i++) { counts.push(0); }

    for (i = 0; i < state.residents.length; i++) {
      var pay = paymentsOf(state.residents[i], state.year);
      for (var k = 0; k < 12; k++) {
        if (pay[k] === true) { counts[k] += 1; }
      }
    }

    var grand = 0;
    for (i = 0; i < 12; i++) { grand += counts[i] * state.dues; }

    /* kaki tabel: total kas masuk tiap bulan */
    var tfoot = document.getElementById('tfoot');
    tfoot.innerHTML = '';
    var tr = document.createElement('tr');

    var head = document.createElement('td');
    head.className = 'sticky';
    head.colSpan = 2;
    head.textContent = 'Total kas masuk';
    tr.appendChild(head);

    for (i = 0; i < 12; i++) {
      var amount = counts[i] * state.dues;
      var td = document.createElement('td');
      td.className = 'c';
      td.textContent = fmtCompact(amount);
      td.title = fmtMoney(amount) + ' (' + counts[i] + ' warga lunas)';
      tr.appendChild(td);
    }

    var tdGrand = document.createElement('td');
    tdGrand.className = 'c';
    tdGrand.textContent = fmtCompact(grand);
    tdGrand.title = fmtMoney(grand);
    tr.appendChild(tdGrand);

    tr.appendChild(document.createElement('td'));
    tfoot.appendChild(tr);

    /* tunggakan sampai bulan berjalan */
    var limit = monthsElapsed(state.year);
    var list = [];
    if (limit > 0) {
      for (i = 0; i < state.residents.length; i++) {
        var r = state.residents[i];
        var p = paymentsOf(r, state.year);
        var n = 0;
        for (var j = 0; j < limit; j++) { if (p[j] !== true) { n++; } }
        if (n > 0) { list.push({ r: r, months: n, amount: n * state.dues }); }
      }
      list.sort(function (a, b) {
        if (b.amount !== a.amount) { return b.amount - a.amount; }
        return a.r.name.localeCompare(b.r.name, 'id');
      });
    }

    var totalDue = 0;
    for (i = 0; i < list.length; i++) { totalDue += list[i].amount; }

    var box = document.getElementById('tunggakan');
    box.innerHTML = '';

    if (limit === 0) {
      var p0 = document.createElement('p');
      p0.className = 'empty';
      p0.textContent = 'Tahun ' + state.year + ' belum berjalan, belum ada tunggakan yang dihitung.';
      box.appendChild(p0);
    } else if (list.length === 0) {
      var p1 = document.createElement('p');
      p1.className = 'empty';
      p1.textContent = 'Tidak ada tunggakan sampai ' + MONTHS_SHORT[limit - 1] + ' ' + state.year + '. Semua warga sudah lunas.';
      box.appendChild(p1);
    } else {
      var ul = document.createElement('ul');
      ul.className = 'tunggak';
      for (i = 0; i < list.length; i++) {
        var item = list[i];
        var li = document.createElement('li');

        var left = document.createElement('div');
        var who = document.createElement('div');
        who.className = 'who';
        who.textContent = item.r.name + ' · Rumah ' + item.r.house;
        var meta = document.createElement('div');
        meta.className = 'meta';
        meta.textContent = item.months + ' bulan belum dibayar (s.d. ' + MONTHS_SHORT[limit - 1] + ' ' + state.year + ')';
        left.appendChild(who);
        left.appendChild(meta);

        var amt = document.createElement('div');
        amt.className = 'amt';
        amt.textContent = fmtMoney(item.amount);

        li.appendChild(left);
        li.appendChild(amt);
        ul.appendChild(li);
      }
      box.appendChild(ul);

      var total = document.createElement('p');
      total.className = 'total-line';
      total.textContent = 'Total tunggakan: ' + fmtMoney(totalDue);
      box.appendChild(total);
    }

    document.getElementById('statKas').textContent = fmtMoney(grand);
    document.getElementById('statTunggakan').textContent = (limit === 0) ? '—' : fmtMoney(totalDue);
    document.getElementById('statWarga').textContent = String(state.residents.length);
  }

  /* ---------- render utama ---------- */

  function render() {
    var selTahun = document.getElementById('selTahun');
    var inIuran = document.getElementById('inIuran');
    if (selTahun.value !== String(state.year)) { selTahun.value = String(state.year); }
    inIuran.value = String(state.dues);
    document.getElementById('lblTahun').textContent = String(state.year);
    renderTable();
    renderSummary();
  }

  /* ---------- pengaturan tahun ---------- */

  function buildYearOptions() {
    var cy = new Date().getFullYear();
    var years = [];
    for (var y = cy - 3; y <= cy + 3; y++) { years.push(y); }
    if (years.indexOf(state.year) === -1) {
      years.push(state.year);
      years.sort(function (a, b) { return a - b; });
    }
    var sel = document.getElementById('selTahun');
    sel.innerHTML = '';
    for (var i = 0; i < years.length; i++) {
      var o = document.createElement('option');
      o.value = String(years[i]);
      o.textContent = String(years[i]);
      sel.appendChild(o);
    }
    sel.value = String(state.year);
  }

  /* ---------- form warga ---------- */

  function startEdit(id) {
    var r = findResident(id);
    if (!r) { return; }
    editingId = id;
    document.getElementById('inNama').value = r.name;
    document.getElementById('inRumah').value = r.house;
    document.getElementById('btnSubmit').textContent = 'Simpan perubahan';
    document.getElementById('btnBatal').hidden = false;
    document.getElementById('inNama').focus();
    var card = document.getElementById('formWarga').closest('.card');
    if (card && card.scrollIntoView) {
      card.scrollIntoView({ behavior: 'smooth', block: 'start' });
    }
  }

  function exitEdit() {
    editingId = null;
    document.getElementById('formWarga').reset();
    document.getElementById('btnSubmit').textContent = 'Tambah warga';
    document.getElementById('btnBatal').hidden = true;
  }

  function removeResident(id) {
    var r = findResident(id);
    if (!r) { return; }
    var ok = window.confirm('Hapus data warga "' + r.name + '" (rumah ' + r.house + ')? Tindakan ini tidak bisa dibatalkan.');
    if (!ok) { return; }
    state.residents = state.residents.filter(function (x) { return x.id !== id; });
    if (editingId === id) { exitEdit(); }
    save();
    render();
    setStatus('Data warga dihapus.');
  }

  /* ---------- ekspor CSV ---------- */

  function csvCell(v) {
    var s = String(v == null ? '' : v);
    if (/[;"\r\n]/.test(s)) {
      return '"' + s.replace(/"/g, '""') + '"';
    }
    return s;
  }

  function exportCSV() {
    var limit = monthsElapsed(state.year);
    var i, j;

    var head = ['Nama Kepala Keluarga', 'Nomor Rumah'];
    for (i = 0; i < 12; i++) { head.push(MONTHS_SHORT[i]); }
    head.push('Total Lunas (Rp)');
    head.push('Tunggakan s.d. ' + (limit > 0 ? MONTHS_SHORT[limit - 1] : '-') + ' (Rp)');

    var rows = [head];
    var counts = [];
    for (i = 0; i < 12; i++) { counts.push(0); }
    var totalPaid = 0;
    var totalDue = 0;

    for (i = 0; i < state.residents.length; i++) {
      var r = state.residents[i];
      var pay = paymentsOf(r, state.year);
      var row = [r.name, r.house];
      var paid = 0;
      var due = 0;
      for (j = 0; j < 12; j++) {
        if (pay[j] === true) { row.push('Lunas'); paid++; counts[j]++; }
        else { row.push('Belum'); }
      }
      for (j = 0; j < limit; j++) { if (pay[j] !== true) { due++; } }
      var paidAmt = paid * state.dues;
      var dueAmt = due * state.dues;
      totalPaid += paidAmt;
      totalDue += dueAmt;
      row.push(paidAmt);
      row.push(dueAmt);
      rows.push(row);
    }

    var foot = ['TOTAL', ''];
    for (i = 0; i < 12; i++) { foot.push(counts[i] * state.dues); }
    foot.push(totalPaid);
    foot.push(totalDue);
    rows.push(foot);

    var csv = rows.map(function (row) {
      return row.map(csvCell).join(';');
    }).join('\r\n');

    try {
      var blob = new Blob(['﻿' + csv], { type: 'text/csv;charset=utf-8;' });
      var url = URL.createObjectURL(blob);
      var a = document.createElement('a');
      a.href = url;
      a.download = 'iuran-rt-' + state.year + '.csv';
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      setTimeout(function () { URL.revokeObjectURL(url); }, 1500);
      setStatus('Berkas CSV tahun ' + state.year + ' diunduh.');
    } catch (e) {
      setStatus('Ekspor CSV gagal dijalankan pada peramban ini.');
    }
  }

  /* ---------- inisialisasi & event ---------- */

  state = load();
  buildYearOptions();
  render();

  document.getElementById('formWarga').addEventListener('submit', function (e) {
    e.preventDefault();
    var inNama = document.getElementById('inNama');
    var inRumah = document.getElementById('inRumah');
    var nama = inNama.value.trim();
    var rumah = inRumah.value.trim();
    if (!nama || !rumah) {
      setStatus('Nama kepala keluarga dan nomor rumah wajib diisi.');
      return;
    }
    if (editingId) {
      var r = findResident(editingId);
      if (r) { r.name = nama; r.house = rumah; }
      exitEdit();
      setStatus('Data warga diperbarui.');
    } else {
      state.residents.push({ id: uid(), name: nama, house: rumah, payments: {} });
      document.getElementById('formWarga').reset();
      setStatus('Warga baru ditambahkan.');
    }
    save();
    render();
    inNama.focus();
  });

  document.getElementById('btnBatal').addEventListener('click', function () {
    exitEdit();
    setStatus('Penyuntingan dibatalkan.');
  });

  document.getElementById('inIuran').addEventListener('change', function () {
    var v = Math.round(Number(this.value));
    if (!isFinite(v) || v < 0) {
      this.value = String(state.dues);
      setStatus('Besar iuran tidak valid.');
      return;
    }
    state.dues = v;
    save();
    render();
    setStatus('Besar iuran bulanan diperbarui menjadi ' + fmtMoney(v) + '.');
  });

  document.getElementById('selTahun').addEventListener('change', function () {
    var y = Number(this.value);
    if (!isFinite(y) || y < 1900 || y > 2200) { return; }
    state.year = y;
    save();
    render();
    setStatus('Tahun buku diubah ke ' + y + '.');
  });

  document.getElementById('btnExport').addEventListener('click', exportCSV);

})();
</script>

**Cek kebutuhan**
- **Satu file HTML lengkap** → seluruh dokumen `<!DOCTYPE html>` dengan CSS di `<style>` dan JavaScript di `<script>`, tanpa aset eksternal.
- **Menambah warga (nama kepala keluarga + nomor rumah)** → `#formWarga` dan handler `submit` yang mendorong objek baru ke `state.residents`.
- **Mengubah data warga** → `startEdit()` mengisi formulir, `submit` dalam mode `editingId` memperbarui data, `exitEdit()` mengembalikan mode tambah.
- **Menghapus data warga** → `removeResident()` dengan konfirmasi `window.confirm`.
- **Menetapkan besar iuran bulanan** → input `#inIuran` dan listener `change` yang memperbarui `state.dues`.
- **Centang status lunas per bulan Januari–Desember** → 12 checkbox per baris di `buildRow()`, tersimpan ke `paymentsOf(r, state.year)`.
- **Pemilihan tahun** → `#selTahun` diisi `buildYearOptions()`; data tersimpan per tahun pada `r.payments[String(year)]`.
- **Total kas masuk per bulan** → baris `<tfoot>` yang dibangun di `renderSummary()` dengan `counts[i] * state.dues`.
- **Daftar warga menunggak + jumlah tunggakan sampai bulan berjalan** → blok `limit = monthsElapsed(state.year)` dan render `#tunggakan`.
- **Tombol ekspor CSV** → `#btnExport` dengan fungsi `exportCSV()` (Blob + `URL.createObjectURL`).
- **Penyimpanan localStorage dengan try/catch** → `load()` dan `save()` membungkus `localStorage.getItem/setItem` dalam `try/catch`.
- **Data contoh fiktif** → `defaultState()` berisi enam warga fiktif (Budi Santoso, Siti Aminah, dan seterusnya).
- **Responsif sampai 375 px** → `.tablewrap{overflow-x:auto}`, `.field{flex:1 1 170px}`, dan media query `max-width:520px`.
- **Tanpa error JavaScript** → IIFE dengan `'use strict'`, validasi `isFinite`, pengecekan `null`, dan tanpa dependensi luar.

**Catatan teknis**
CSV memakai pemisah titik koma (`;`) plus BOM UTF-8 agar langsung rapi di Excel berlokal Indonesia. Tabel memakai kolom nama yang lengket (`position: sticky`) saat digeser horizontal. Perhitungan tunggakan mengikuti aturan: tahun lampau dihitung 12 bulan, tahun berjalan dihitung sampai bulan sekarang, tahun mendatang dihitung nol. Jika localStorage diblokir, aplikasi tetap berjalan di memori dan menampilkan peringatan pada `#status`.
