<script>
(function () {
  'use strict';

  /* ---------- Konstanta ---------- */
  var COLS = 25, ROWS = 15;
  var EMPTY = 0, WALL = 1, MUD = 2;
  var DIRS = [[-1, 0], [1, 0], [0, -1], [0, 1]];
  var MAZE_DIRS = [[0, 2], [0, -2], [2, 0], [-2, 0]];

  var gridEl = document.getElementById('grid');
  var statVisited = document.getElementById('statVisited');
  var statPath = document.getElementById('statPath');

  /* ---------- State ---------- */
  var grid = [];
  var cells = [];
  var start = { r: 7, c: 4 };
  var goal = { r: 7, c: 20 };
  var brush = WALL;
  var currentAlgo = 'bfs';
  var runToken = 0;
  var dragging = null;
  var lastPainted = null;

  gridEl.style.setProperty('--cols', COLS);

  /* ---------- Utilitas ---------- */
  function mk2(v) {
    var out = [];
    for (var r = 0; r < ROWS; r++) {
      var row = [];
      for (var c = 0; c < COLS; c++) row.push(v);
      out.push(row);
    }
    return out;
  }

  function shuffle(arr) {
    var a = arr.slice();
    for (var i = a.length - 1; i > 0; i--) {
      var j = Math.floor(Math.random() * (i + 1));
      var t = a[i]; a[i] = a[j]; a[j] = t;
    }
    return a;
  }

  function sleep(ms) {
    return new Promise(function (res) { setTimeout(res, ms); });
  }

  function isOpen(r, c) {
    return r >= 0 && r < ROWS && c >= 0 && c < COLS && grid[r][c] !== WALL;
  }

  function terrainCost(r, c) {
    return grid[r][c] === MUD ? 5 : 1;
  }

  /* ---------- Priority queue (min-heap) ---------- */
  function MinHeap() { this.a = []; }
  MinHeap.prototype.size = function () { return this.a.length; };
  MinHeap.prototype.push = function (node) {
    var a = this.a;
    a.push(node);
    var i = a.length - 1;
    while (i > 0) {
      var p = (i - 1) >> 1;
      if (a[p].f <= a[i].f) break;
      var t = a[p]; a[p] = a[i]; a[i] = t;
      i = p;
    }
  };
  MinHeap.prototype.pop = function () {
    var a = this.a;
    var top = a[0];
    var last = a.pop();
    if (a.length) {
      a[0] = last;
      var i = 0;
      for (;;) {
        var l = 2 * i + 1, r = 2 * i + 2, m = i;
        if (l < a.length && a[l].f < a[m].f) m = l;
        if (r < a.length && a[r].f < a[m].f) m = r;
        if (m === i) break;
        var t = a[m]; a[m] = a[i]; a[i] = t;
        i = m;
      }
    }
    return top;
  };

  /* ---------- DOM grid ---------- */
  function buildDOM() {
    var frag = document.createDocumentFragment();
    cells = [];
    for (var r = 0; r < ROWS; r++) {
      var row = [];
      for (var c = 0; c < COLS; c++) {
        var d = document.createElement('div');
        d.className = 'cell';
        d.dataset.r = r;
        d.dataset.c = c;
        frag.appendChild(d);
        row.push(d);
      }
      cells.push(row);
    }
    gridEl.appendChild(frag);
  }

  function initGrid() {
    grid = [];
    for (var r = 0; r < ROWS; r++) {
      var row = [];
      for (var c = 0; c < COLS; c++) row.push(EMPTY);
      grid.push(row);
    }
  }

  function updateCell(r, c) {
    var el = cells[r][c];
    var cls = 'cell';
    var v = grid[r][c];
    if (v === WALL) cls += ' wall';
    else if (v === MUD) cls += ' mud';
    if (r === start.r && c === start.c) cls += ' start';
    else if (r === goal.r && c === goal.c) cls += ' goal';
    el.className = cls;
  }

  function render() {
    for (var r = 0; r < ROWS; r++) {
      for (var c = 0; c < COLS; c++) updateCell(r, c);
    }
  }

  function resetStats() {
    statVisited.textContent = '—';
    statPath.textContent = '—';
  }

  function cancelRun() {
    runToken++;
  }

  /* ---------- Preferensi (localStorage, dibungkus try/catch) ---------- */
  function loadPrefs() {
    try {
      var raw = localStorage.getItem('pf-viz-prefs');
      if (!raw) return;
      var p = JSON.parse(raw);
      if (p && (p.algo === 'bfs' || p.algo === 'dijkstra' || p.algo === 'astar')) {
        currentAlgo = p.algo;
      }
      if (p && (p.brush === 'wall' || p.brush === 'mud')) {
        brush = p.brush === 'wall' ? WALL : MUD;
      }
    } catch (err) {
      /* localStorage tidak tersedia atau data rusak: abaikan */
    }
  }

  function savePrefs() {
    try {
      localStorage.setItem('pf-viz-prefs', JSON.stringify({
        algo: currentAlgo,
        brush: brush === WALL ? 'wall' : 'mud'
      }));
    } catch (err) {
      /* penyimpanan gagal: abaikan */
    }
  }

  /* ---------- Interaksi pointer ---------- */
  function cellFromPoint(x, y) {
    var el = document.elementFromPoint(x, y);
    if (!el || !el.closest) return null;
    var cell = el.closest('.cell');
    if (!cell || !gridEl.contains(cell)) return null;
    return cell;
  }

  function applyPaint(r, c, value) {
    if ((r === start.r && c === start.c) || (r === goal.r && c === goal.c)) return;
    if (grid[r][c] === value) return;
    grid[r][c] = value;
    updateCell(r, c);
  }

  function clearOverlay() {
    render();
    resetStats();
  }

  function onDown(e) {
    if (e.pointerType === 'mouse' && e.button !== 0) return;
    var cell = cellFromPoint(e.clientX, e.clientY);
    if (!cell) return;
    e.preventDefault();
    cancelRun();

    var r = +cell.dataset.r, c = +cell.dataset.c;

    if (r === start.r && c === start.c) {
      dragging = { type: 'start' };
      lastPainted = null;
      clearOverlay();
      return;
    }
    if (r === goal.r && c === goal.c) {
      dragging = { type: 'goal' };
      lastPainted = null;
      clearOverlay();
      return;
    }

    clearOverlay();
    var value = grid[r][c] === brush ? EMPTY : brush;
    dragging = { type: 'paint', value: value };
    lastPainted = r + ':' + c;
    applyPaint(r, c, value);
  }

  function onMove(e) {
    if (!dragging) return;
    var cell = cellFromPoint(e.clientX, e.clientY);
    if (!cell) return;
    if (e.cancelable) e.preventDefault();

    var r = +cell.dataset.r, c = +cell.dataset.c;

    if (dragging.type === 'start') {
      if (r === goal.r && c === goal.c) return;
      if (grid[r][c] === WALL) return;
      var oldS = start;
      start = { r: r, c: c };
      updateCell(oldS.r, oldS.c);
      updateCell(r, c);
      return;
    }
    if (dragging.type === 'goal') {
      if (r === start.r && c === start.c) return;
      if (grid[r][c] === WALL) return;
      var oldG = goal;
      goal = { r: r, c: c };
      updateCell(oldG.r, oldG.c);
      updateCell(r, c);
      return;
    }

    var key = r + ':' + c;
    if (key === lastPainted) return;
    lastPainted = key;
    applyPaint(r, c, dragging.value);
  }

  function onUp() {
    dragging = null;
    lastPainted = null;
  }

  gridEl.addEventListener('pointerdown', onDown, { passive: false });
  window.addEventListener('pointermove', onMove, { passive: false });
  window.addEventListener('pointerup', onUp);
  window.addEventListener('pointercancel', onUp);

  /* ---------- Algoritma ---------- */
  function rebuildPath(parent) {
    var path = [];
    var cur = { r: goal.r, c: goal.c };
    var guard = 0;
    while (cur && guard++ < ROWS * COLS + 5) {
      path.push(cur);
      if (cur.r === start.r && cur.c === start.c) break;
      cur = parent[cur.r][cur.c];
    }
    if (!path.length) return [];
    var last = path[path.length - 1];
    if (last.r !== start.r || last.c !== start.c) return [];
    return path.reverse();
  }

  function runBFS() {
    var seen = mk2(false);
    var parent = mk2(null);
    var order = [];
    var queue = [{ r: start.r, c: start.c }];
    seen[start.r][start.c] = true;
    var found = false;

    for (var i = 0; i < queue.length; i++) {
      var cur = queue[i];
      order.push(cur);
      if (cur.r === goal.r && cur.c === goal.c) { found = true; break; }
      for (var d = 0; d < DIRS.length; d++) {
        var nr = cur.r + DIRS[d][0], nc = cur.c + DIRS[d][1];
        if (isOpen(nr, nc) && !seen[nr][nc]) {
          seen[nr][nc] = true;
          parent[nr][nc] = cur;
          queue.push({ r: nr, c: nc });
        }
      }
    }
    return { order: order, path: found ? rebuildPath(parent) : [] };
  }

  function runWeighted(useHeuristic) {
    var dist = mk2(Infinity);
    var parent = mk2(null);
    var done = mk2(false);
    var order = [];
    var heap = new MinHeap();
    var found = false;

    dist[start.r][start.c] = 0;
    heap.push({ r: start.r, c: start.c, f: 0 });

    while (heap.size() > 0) {
      var cur = heap.pop();
      if (done[cur.r][cur.c]) continue;
      done[cur.r][cur.c] = true;
      order.push({ r: cur.r, c: cur.c });

      if (cur.r === goal.r && cur.c === goal.c) { found = true; break; }

      var g0 = dist[cur.r][cur.c];
      for (var d = 0; d < DIRS.length; d++) {
        var nr = cur.r + DIRS[d][0], nc = cur.c + DIRS[d][1];
        if (!isOpen(nr, nc) || done[nr][nc]) continue;
        var ng = g0 + terrainCost(nr, nc);
        if (ng < dist[nr][nc]) {
          dist[nr][nc] = ng;
          parent[nr][nc] = { r: cur.r, c: cur.c };
          var h = useHeuristic ? (Math.abs(nr - goal.r) + Math.abs(nc - goal.c)) : 0;
          heap.push({ r: nr, c: nc, f: ng + h });
        }
      }
    }
    return { order: order, path: found ? rebuildPath(parent) : [] };
  }

  function pathCost(path) {
    var cost = 0;
    for (var i = 1; i < path.length; i++) cost += terrainCost(path[i].r, path[i].c);
    return cost;
  }

  /* ---------- Animasi ---------- */
  function animate(order, path) {
    var token = ++runToken;
    render();

    var stepSize = Math.max(1, Math.ceil(order.length / 55));
    var i = 0;

    function phaseExplore() {
      if (token !== runToken) return;
      if (i >= order.length) { phasePath(); return; }
      var end = Math.min(i + stepSize, order.length);
      for (var j = i; j < end; j++) {
        var p = order[j];
        if ((p.r === start.r && p.c === start.c) || (p.r === goal.r && p.c === goal.c)) continue;
        cells[p.r][p.c].classList.add('visited');
      }
      i = end;
      sleep(16).then(phaseExplore);
    }

    function phasePath() {
      if (token !== runToken) return;
      if (path.length <= 1) return;
      var pStep = Math.max(1, Math.ceil(path.length / 28));
      var k = 0;
      (function step() {
        if (token !== runToken) return;
        if (k >= path.length) return;
        var end = Math.min(k + pStep, path.length);
        for (var m = k; m < end; m++) {
          var q = path[m];
          var el = cells[q.r][q.c];
          el.classList.remove('visited');
          if ((q.r === start.r && q.c === start.c) || (q.r === goal.r && q.c === goal.c)) continue;
          el.classList.add('path');
        }
        k = end;
        sleep(18).then(step);
      })();
    }

    phaseExplore();
  }

  /* ---------- Eksekusi ---------- */
  function run() {
    cancelRun();
    var res = currentAlgo === 'bfs' ? runBFS() : runWeighted(currentAlgo === 'astar');

    statVisited.textContent = String(res.order.length);
    if (!res.path.length) {
      statPath.textContent = 'tidak ditemukan';
    } else {
      statPath.textContent = (res.path.length - 1) + ' langkah · biaya ' + pathCost(res.path);
    }

    animate(res.order, res.path);
  }

  /* ---------- Aksi papan ---------- */
  function clearBoard() {
    cancelRun();
    for (var r = 0; r < ROWS; r++) {
      for (var c = 0; c < COLS; c++) grid[r][c] = EMPTY;
    }
    start = { r: 7, c: 4 };
    goal = { r: 7, c: 20 };
    render();
    resetStats();
  }

  function randomMaze() {
    cancelRun();

    for (var r = 0; r < ROWS; r++) {
      for (var c = 0; c < COLS; c++) grid[r][c] = WALL;
    }

    grid[1][1] = EMPTY;
    var stack = [[1, 1]];
    while (stack.length) {
      var cur = stack[stack.length - 1];
      var dirs = shuffle(MAZE_DIRS);
      var advanced = false;
      for (var k = 0; k < dirs.length; k++) {
        var nr = cur[0] + dirs[k][0];
        var nc = cur[1] + dirs[k][1];
        if (nr > 0 && nr < ROWS - 1 && nc > 0 && nc < COLS - 1 && grid[nr][nc] === WALL) {
          grid[cur[0] + dirs[k][0] / 2][cur[1] + dirs[k][1] / 2] = EMPTY;
          grid[nr][nc] = EMPTY;
          stack.push([nr, nc]);
          advanced = true;
          break;
        }
      }
      if (!advanced) stack.pop();
    }

    var rr, cc;
    for (rr = 1; rr < ROWS - 1; rr++) {
      for (cc = 1; cc < COLS - 1; cc++) {
        if (grid[rr][cc] === WALL && Math.random() < 0.09) grid[rr][cc] = EMPTY;
      }
    }
    for (rr = 1; rr < ROWS - 1; rr++) {
      for (cc = 1; cc < COLS - 1; cc++) {
        if (grid[rr][cc] === EMPTY && Math.random() < 0.1) grid[rr][cc] = MUD;
      }
    }

    start = { r: 1, c: 1 };
    goal = { r: ROWS - 2, c: COLS - 2 };
    grid[start.r][start.c] = EMPTY;
    grid[goal.r][goal.c] = EMPTY;

    render();
    resetStats();
  }

  /* ---------- Pengikatan tombol ---------- */
  var algoBtns = document.querySelectorAll('[data-algo]');
  Array.prototype.forEach.call(algoBtns, function (btn) {
    btn.addEventListener('click', function () {
      currentAlgo = btn.dataset.algo;
      Array.prototype.forEach.call(algoBtns, function (b) {
        b.classList.toggle('active', b === btn);
      });
      savePrefs();
      run();
    });
  });

  var brushBtns = document.querySelectorAll('[data-brush]');
  Array.prototype.forEach.call(brushBtns, function (btn) {
    btn.addEventListener('click', function () {
      brush = btn.dataset.brush === 'wall' ? WALL : MUD;
      Array.prototype.forEach.call(brushBtns, function (b) {
        b.classList.toggle('active', b === btn);
      });
      savePrefs();
    });
  });

  document.getElementById('btnMaze').addEventListener('click', randomMaze);
  document.getElementById('btnClear').addEventListener('click', clearBoard);

  /* ---------- Inisialisasi ---------- */
  buildDOM();
  initGrid();
  loadPrefs();

  Array.prototype.forEach.call(algoBtns, function (b) {
    b.classList.toggle('active', b.dataset.algo === currentAlgo);
  });
  Array.prototype.forEach.call(brushBtns, function (b) {
    var isWallBtn = b.dataset.brush === 'wall';
    b.classList.toggle('active', isWallBtn === (brush === WALL));
  });

  render();
  resetStats();
})();
</script>

**Cek kebutuhan**
- **Grid berukuran 25×15** → konstanta `COLS = 25, ROWS = 15` serta pembentukan sel di `buildDOM()` dengan `grid-template-columns: repeat(var(--cols,25),1fr)`.
- **Klik atau seret untuk menggambar dinding** → `onDown()`, `onMove()`, `applyPaint()` dengan mode kuas `dragging.value`.
- **Titik awal dan tujuan bisa dipindah** → cabang `dragging.type === 'start'` dan `'goal'` pada `onDown()`/`onMove()`, memakai `updateCell()` untuk menggambar ulang dua petak.
- **Pilihan algoritma BFS, Dijkstra (lumpur biaya 5), dan A\*** → `runBFS()`, `runWeighted(false)`, `runWeighted(true)`, dengan `terrainCost()` mengembalikan 5 untuk `MUD`.
- **Animasi petak yang dijelajahi lalu jalur akhir** → fungsi `animate()` dengan dua fase, `phaseExplore()` dan `phasePath()`.
- **Jumlah petak dijelajahi serta panjang/biaya jalur** → `statVisited` diisi `res.order.length`; `statPath` diisi `res.path.length - 1` langkah dan `pathCost(res.path)`.
- **Tombol labirin acak** → `randomMaze()` dengan recursive backtracker, pembukaan dinding acak, dan sebaran lumpur.
- **Responsif sampai lebar 375 px** → `max-width:860px`, `aspect-ratio:1/1`, `flex-wrap` pada panel, serta media query `@media (max-width:420px)`.
- **Tanpa library, font, gambar, atau berkas luar** → seluruh CSS dan JS tertulis di dalam berkas, memakai `system-ui` dan warna CSS.
- **Tanpa error JavaScript dan `localStorage` dibungkus try/catch** → `loadPrefs()` dan `savePrefs()` memakai blok `try { ... } catch (err) { ... }`.
- **Animasi bisa dibatalkan saat papan diedit** → `cancelRun()` menaikkan `runToken`, dan setiap fase animasi memeriksa `token !== runToken`.

**Catatan teknis**
BFS mengabaikan biaya lumpur (semua petak dianggap satu langkah), sedangkan Dijkstra dan A* memakai biaya masuk petak (1 untuk kosong, 5 untuk lumpur). Heuristik A* adalah jarak Manhattan yang konsisten karena biaya minimum per langkah adalah 1, sehingga penanganan simpul ganda lewat `done` tetap optimal. Prioritas antrean memakai min-heap buatan sendiri, dan animasi memakai potongan petak per bingkai agar tetap ringan di layar kecil.
