```html
<script>
(function () {
  'use strict';

  var $ = function (id) { return document.getElementById(id); };
  var inp = {
    awal: $('awal'),
    tambahan: $('tambahan'),
    bunga: $('bunga'),
    bulan: $('bulan'),
    pajak: $('pajak')
  };
  var statSaldo = $('statSaldo');
  var statSetoran = $('statSetoran');
  var statBunga = $('statBunga');
  var statPajak = $('statPajak');
  var tbody = $('tbody');
  var canvas = $('chart');
  var KUNCI = 'simulatorDeposito.v1';
  var titik = [];

  function angka(v) {
    var x = parseFloat(v);
    return isFinite(x) ? x : 0;
  }

  function rupiah(n) {
    var bulat = Math.round(n);
    var negatif = bulat < 0;
    var s = Math.abs(bulat).toLocaleString('id-ID');
    return (negatif ? '−Rp ' : 'Rp ') + s;
  }

  function ringkas(v) {
    var f = function (x) {
      if (Math.abs(x) >= 100) return String(Math.round(x));
      return String(Math.round(x * 10) / 10).replace('.', ',');
    };
    if (v >= 1e12) return f(v / 1e12) + ' T';
    if (v >= 1e9) return f(v / 1e9) + ' M';
    if (v >= 1e6) return f(v / 1e6) + ' jt';
    if (v >= 1e3) return f(v / 1e3) + ' rb';
    return String(Math.round(v));
  }

  function skala(max) {
    var target = max > 0 ? max : 1000;
    var kasar = target / 4;
    var pangkat = Math.pow(10, Math.floor(Math.log10(kasar)));
    var f = kasar / pangkat;
    var nf = f <= 1 ? 1 : f <= 2 ? 2 : f <= 2.5 ? 2.5 : f <= 5 ? 5 : 10;
    var langkah = nf * pangkat;
    var atas = Math.ceil(target / langkah - 1e-9) * langkah;
    return { langkah: langkah, atas: atas };
  }

  function gambarGrafik() {
    var wrap = canvas.parentNode;
    var lebar = wrap.clientWidth || 300;
    if (lebar < 200) lebar = 200;
    var tinggi = lebar < 420 ? 220 : 280;
    var dpr = Math.min(window.devicePixelRatio || 1, 2);

    canvas.width = Math.round(lebar * dpr);
    canvas.height = Math.round(tinggi * dpr);
    canvas.style.width = lebar + 'px';
    canvas.style.height = tinggi + 'px';

    var ctx = canvas.getContext('2d');
    if (!ctx) return;
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    ctx.clearRect(0, 0, lebar, tinggi);

    var padKiri = lebar < 400 ? 48 : 58;
    var padKanan = 10;
    var padAtas = 16;
    var padBawah = 26;
    var w = lebar - padKiri - padKanan;
    var h = tinggi - padAtas - padBawah;
    if (w <= 10 || h <= 10) return;

    var data = titik.length ? titik : [{ x: 0, y: 0 }];
    var maks = 0;
    var i;
    for (i = 0; i < data.length; i++) {
      if (data[i].y > maks) maks = data[i].y;
    }
    var sk = skala(maks);
    var n = data[data.length - 1].x;

    function xPada(v) { return padKiri + (n <= 0 ? w / 2 : (v / n) * w); }
    function yPada(v) { return padAtas + h - (v / sk.atas) * h; }

    ctx.font = '11px system-ui, -apple-system, "Segoe UI", Roboto, Arial, sans-serif';
    ctx.textBaseline = 'middle';

    var jumlahGaris = Math.round(sk.atas / sk.langkah);
    if (jumlahGaris < 1) jumlahGaris = 1;
    for (var s = 0; s <= jumlahGaris; s++) {
      var nilai = s * sk.langkah;
      var y = yPada(nilai);
      ctx.beginPath();
      ctx.strokeStyle = s === 0 ? '#cbd5e1' : '#e8edf5';
      ctx.lineWidth = 1;
      ctx.moveTo(padKiri, Math.round(y) + 0.5);
      ctx.lineTo(padKiri + w, Math.round(y) + 0.5);
      ctx.stroke();
      ctx.fillStyle = '#7c8aa0';
      ctx.textAlign = 'right';
      ctx.fillText(ringkas(nilai), padKiri - 8, y);
    }

    var dasarY = padAtas + h;

    ctx.beginPath();
    for (i = 0; i < data.length; i++) {
      var ax = xPada(data[i].x);
      var ay = yPada(data[i].y);
      if (i === 0) ctx.moveTo(ax, ay); else ctx.lineTo(ax, ay);
    }
    ctx.lineTo(xPada(data[data.length - 1].x), dasarY);
    ctx.lineTo(xPada(data[0].x), dasarY);
    ctx.closePath();
    var grad = ctx.createLinearGradient(0, padAtas, 0, dasarY);
    grad.addColorStop(0, 'rgba(79,70,229,0.30)');
    grad.addColorStop(1, 'rgba(79,70,229,0.02)');
    ctx.fillStyle = grad;
    ctx.fill();

    ctx.beginPath();
    for (i = 0; i < data.length; i++) {
      var lx = xPada(data[i].x);
      var ly = yPada(data[i].y);
      if (i === 0) ctx.moveTo(lx, ly); else ctx.lineTo(lx, ly);
    }
    ctx.strokeStyle = '#4f46e5';
    ctx.lineWidth = 2;
    ctx.lineJoin = 'round';
    ctx.lineCap = 'round';
    ctx.stroke();

    if (data.length <= 25) {
      for (i = 0; i < data.length; i++) {
        ctx.beginPath();
        ctx.arc(xPada(data[i].x), yPada(data[i].y), 2.6, 0, Math.PI * 2);
        ctx.fillStyle = '#4f46e5';
        ctx.fill();
      }
    }

    var akhir = data[data.length - 1];
    ctx.beginPath();
    ctx.arc(xPada(akhir.x), yPada(akhir.y), 4, 0, Math.PI * 2);
    ctx.fillStyle = '#ffffff';
    ctx.fill();
    ctx.strokeStyle = '#4f46e5';
    ctx.lineWidth = 2;
    ctx.stroke();

    var ticks = [0];
    if (n > 0) {
      var ideal = Math.max(1, Math.round(n / 5));
      var mag = Math.pow(10, Math.floor(Math.log10(ideal)));
      var ff = ideal / mag;
      var step = (ff <= 1 ? 1 : ff <= 2 ? 2 : ff <= 5 ? 5 : 10) * mag;
      for (var t = step; t < n; t += step) ticks.push(t);
      ticks.push(n);
    }

    var tampil = [];
    for (var k = ticks.length - 1; k >= 0; k--) {
      var tx = xPada(ticks[k]);
      if (tampil.length === 0 || tampil[0].x - tx >= 30) {
        tampil.unshift({ t: ticks[k], x: tx });
      }
    }

    ctx.textAlign = 'center';
    ctx.textBaseline = 'top';
    ctx.fillStyle = '#7c8aa0';
    for (var m = 0; m < tampil.length; m++) {
      ctx.fillText(String(tampil[m].t), tampil[m].x, padAtas + h + 8);
    }
  }

  function hitung() {
    var awal = Math.max(0, angka(inp.awal.value));
    var tambahan = Math.max(0, angka(inp.tambahan.value));
    var bungaTahun = Math.max(0, angka(inp.bunga.value));
    var bulan = Math.floor(angka(inp.bulan.value));
    if (!(bulan >= 1)) bulan = 1;
    if (bulan > 600) bulan = 600;
    var pakaiPajak = inp.pajak.checked;

    var rBulan = bungaTahun / 100 / 12;
    var saldo = awal;
    var totalSetoran = awal;
    var totalBungaKotor = 0;
    var totalPajak = 0;

    var baris = ['<tr><td>0</td><td>' + rupiah(awal) + '</td><td class="pos">' + rupiah(0) +
      '</td><td class="neg">' + rupiah(0) + '</td><td class="strong">' + rupiah(awal) + '</td></tr>'];
    titik = [{ x: 0, y: awal }];

    for (var m = 1; m <= bulan; m++) {
      var bungaKotor = saldo * rBulan;
      var pajakBulan = pakaiPajak ? bungaKotor * 0.2 : 0;
      var bungaBersih = bungaKotor - pajakBulan;
      saldo = saldo + bungaBersih + tambahan;
      totalBungaKotor += bungaKotor;
      totalPajak += pajakBulan;
      totalSetoran += tambahan;

      baris.push(
        '<tr><td>' + m + '</td><td>' + rupiah(tambahan) +
        '</td><td class="pos">' + rupiah(bungaBersih) +
        '</td><td class="neg">' + (pajakBulan > 0 ? '−' + rupiah(pajakBulan) : rupiah(0)) +
        '</td><td class="strong">' + rupiah(saldo) + '</td></tr>'
      );
      titik.push({ x: m, y: saldo });
    }

    statSaldo.textContent = rupiah(saldo);
    statSetoran.textContent = rupiah(totalSetoran);
    statBunga.textContent = rupiah(totalBungaKotor - totalPajak);
    statPajak.textContent = rupiah(totalPajak);
    tbody.innerHTML = baris.join('');

    gambarGrafik();
  }

  function simpan() {
    try {
      localStorage.setItem(KUNCI, JSON.stringify({
        awal: inp.awal.value,
        tambahan: inp.tambahan.value,
        bunga: inp.bunga.value,
        bulan: inp.bulan.value,
        pajak: inp.pajak.checked
      }));
    } catch (e) {
      /* penyimpanan tidak tersedia atau diblokir */
    }
  }

  function muat() {
    try {
      var mentah = localStorage.getItem(KUNCI);
      if (!mentah) return;
      var d = JSON.parse(mentah);
      if (!d || typeof d !== 'object') return;
      if (typeof d.awal === 'string') inp.awal.value = d.awal;
      if (typeof d.tambahan === 'string') inp.tambahan.value = d.tambahan;
      if (typeof d.bunga === 'string') inp.bunga.value = d.bunga;
      if (typeof d.bulan === 'string') inp.bulan.value = d.bulan;
      if (typeof d.pajak === 'boolean') inp.pajak.checked = d.pajak;
    } catch (e) {
      /* data rusak diabaikan */
    }
  }

  ['awal', 'tambahan', 'bunga', 'bulan'].forEach(function (k) {
    inp[k].addEventListener('input', function () { hitung(); simpan(); });
  });
  inp.pajak.addEventListener('change', function () { hitung(); simpan(); });

  $('form').addEventListener('submit', function (e) { e.preventDefault(); });

  var raf = 0;
  window.addEventListener('resize', function () {
    if (raf) cancelAnimationFrame(raf);
    raf = requestAnimationFrame(function () { raf = 0; gambarGrafik(); });
  });

  muat();
  hitung();
})();
</script>
</body>
</html>
```

**Cek kebutuhan**
- **Input setoran awal** → `input#awal` dan pembacaan `angka(inp.awal.value)` di fungsi `hitung()`
- **Input setoran tambahan per bulan** → `input#tambahan` dan variabel `tambahan` pada loop `hitung()`
- **Input bunga per tahun** → `input#bunga` dan `rBulan = bungaTahun / 100 / 12`
- **Input jangka waktu dalam bulan** → `input#bulan` dengan pembatasan 1–600 bulan
- **Pajak bunga 20% yang bisa diaktifkan** → `input#pajak` (checkbox) dan `pajakBulan = pakaiPajak ? bungaKotor * 0.2 : 0`
- **Saldo akhir** → kartu `#statSaldo` diisi `rupiah(saldo)` pada akhir `hitung()`
- **Total bunga bersih** → kartu `#statBunga` diisi `rupiah(totalBungaKotor - totalPajak)`
- **Tabel per bulan** → `tbody#tbody` dibangun dari array `baris` (bulan, setoran, bunga bersih, pajak, saldo akhir)
- **Grafik garis dengan canvas tanpa library** → fungsi `gambarGrafik()` memakai `getContext('2d')`, `moveTo/lineTo`, gradien area, dan penanda titik akhir
- **Responsif sampai 375 px** → `.grid`/`.stats` dengan `auto-fit minmax()`, `.table-wrap` ber-`overflow:auto`, dan kanvas dihitung dari `wrap.clientWidth`
- **Tanpa error JavaScript** → semua elemen diambil setelah DOM siap, `parseFloat` dijaga `isFinite`, dan `getContext` diperiksa null
- **localStorage dibungkus try/catch** → fungsi `simpan()` dan `muat()`
- **Tanpa library/font/gambar eksternal** → seluruh CSS, JS, dan grafik berada di dalam file, memakai `system-ui`

**Catatan teknis**
Bunga dihitung dari saldo awal bulan dengan tarif `bunga tahunan / 12`; pajak 20% dipotong dari bunga bruto tiap bulan, lalu setoran tambahan ditambahkan di akhir bulan. Angka besar diringkas pada sumbu grafik (rb/jt/M/T) agar tetap terbaca di layar sempit, dan kanvas digambar ulang saat ukuran jendela berubah.
