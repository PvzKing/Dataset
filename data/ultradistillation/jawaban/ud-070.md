```html
  var MAKS = 5;
  var JEDA = 150;
  var KUNCI = 'kota-tujuan-pengiriman:v1';

  /* ---------- Normalisasi: tanpa huruf besar-kecil & diakritik, dengan peta indeks ---------- */
  function lipat(teks) {
    var hasil = '';
    var peta = [];
    for (var i = 0; i < teks.length; i++) {
      var c = teks.charAt(i);
      var n = typeof c.normalize === 'function' ? c.normalize('NFD') : c;
      n = n.replace(/[̀-ͯ]/g, '').toLowerCase();
      for (var j = 0; j < n.length; j++) {
        hasil += n.charAt(j);
        peta.push(i); // indeks karakter asli
      }
    }
    return { teks: hasil, peta: peta };
  }

  var DATA = KOTA.slice()
    .sort(function (a, b) { return a.localeCompare(b, 'id'); })
    .map(function (nama, i) {
      var f = lipat(nama);
      return { id: 'kota-opsi-' + i, nama: nama, norm: f.teks, peta: f.peta };
    });
  var PER_NAMA = new Map(DATA.map(function (d) { return [d.nama, d]; }));

  /* ---------- Elemen ---------- */
  var combo = document.getElementById('combo');
  var input = document.getElementById('kota-input');
  var toggle = document.getElementById('kota-toggle');
  var listbox = document.getElementById('kota-listbox');
  var chips = document.getElementById('kota-chips');
  var pesan = document.getElementById('kota-pesan');
  var kosong = document.getElementById('kota-kosong');
  var live = document.getElementById('kota-live');
  var form = document.getElementById('form-kirim');
  var hasilSec = document.getElementById('hasil');
  var hasilJson = document.getElementById('hasil-json');

  /* ---------- State ---------- */
  var hasil = [];      // [{ d, pos }]
  var aktif = -1;
  var terbuka = false;
  var kueri = '';
  var terpilih = [];
  var timer = 0;
  var timerLive = 0;

  /* ---------- localStorage (selalu dalam try/catch) ---------- */
  function muat() {
    try {
      var raw = window.localStorage.getItem(KUNCI);
      if (!raw) return [];
      var arr = JSON.parse(raw);
      if (!Array.isArray(arr)) return [];
      var out = [];
      arr.forEach(function (n) {
        if (typeof n === 'string' && PER_NAMA.has(n) && out.indexOf(n) === -1 && out.length < MAKS) out.push(n);
      });
      return out;
    } catch (e) {
      return [];
    }
  }
  function simpan() {
    try {
      window.localStorage.setItem(KUNCI, JSON.stringify(terpilih));
    } catch (e) { /* penyimpanan tidak tersedia: abaikan */ }
  }

  /* ---------- Live region ---------- */
  function umumkan(teks) {
    clearTimeout(timerLive);
    live.textContent = '';
    timerLive = setTimeout(function () { live.textContent = teks; }, 60);
  }

  /* ---------- Penyaringan ---------- */
  function saring(nilai) {
    kueri = lipat(nilai.trim().replace(/\s+/g, ' ')).teks;
    if (!kueri) return DATA.map(function (d) { return { d: d, pos: -1 }; });
    var out = [];
    DATA.forEach(function (d) {
      var p = d.norm.indexOf(kueri);
      if (p !== -1) out.push({ d: d, pos: p });
    });
    // Yang cocok di awal nama didahulukan; urutan alfabet tetap (sort stabil).
    out.sort(function (a, b) { return (a.pos === 0 ? 0 : 1) - (b.pos === 0 ? 0 : 1); });
    return out;
  }

  /* ---------- Penyorotan dengan <mark> lewat DOM (tanpa innerHTML) ---------- */
  function sorot(d, pos, panjang) {
    var frag = document.createDocumentFragment();
    if (pos < 0 || !panjang) {
      frag.appendChild(document.createTextNode(d.nama));
      return frag;
    }
    var awal = d.peta[pos];
    var akhir = d.peta[pos + panjang - 1] + 1;
    if (awal > 0) frag.appendChild(document.createTextNode(d.nama.slice(0, awal)));
    var m = document.createElement('mark');
    m.textContent = d.nama.slice(awal, akhir);
    frag.appendChild(m);
    if (akhir < d.nama.length) frag.appendChild(document.createTextNode(d.nama.slice(akhir)));
    return frag;
  }

  function renderDaftar() {
    var penuh = terpilih.length >= MAKS;
    var frag = document.createDocumentFragment();
    hasil.forEach(function (h, i) {
      var li = document.createElement('li');
      li.id = h.d.id;
      li.className = 'opt';
      li.setAttribute('role', 'option');
      li.dataset.indeks = String(i);
      var dipilih = terpilih.indexOf(h.d.nama) !== -1;
      li.setAttribute('aria-selected', dipilih ? 'true' : 'false');
      if (penuh && !dipilih) li.setAttribute('aria-disabled', 'true');

      var cek = document.createElement('span');
      cek.className = 'check';
      cek.setAttribute('aria-hidden', 'true');
      cek.textContent = '✓';

      var label = document.createElement('span');
      label.className = 'label';
      label.appendChild(sorot(h.d, h.pos, kueri.length));

      li.appendChild(cek);
      li.appendChild(label);
      frag.appendChild(li);
    });
    listbox.replaceChildren(frag);
    aktif = -1;
    input.removeAttribute('aria-activedescendant');
  }

  function setAktif(i, gulir) {
    var lama = listbox.querySelector('.opt.active');
    if (lama) lama.classList.remove('active');
    aktif = i;
    if (i < 0) {
      input.removeAttribute('aria-activedescendant');
      return;
    }
    var el = listbox.children[i];
    if (!el) return;
    el.classList.add('active');
    input.setAttribute('aria-activedescendant', el.id);
    if (gulir !== false) el.scrollIntoView({ block: 'nearest' });
  }

  function setExpanded(v) {
    var s = v ? 'true' : 'false';
    input.setAttribute('aria-expanded', s);
    toggle.setAttribute('aria-expanded', s);
  }

  function buka() {
    if (!hasil.length) { tutup(); return false; }
    listbox.hidden = false;
    terbuka = true;
    setExpanded(true);
    return true;
  }

  function tutup() {
    listbox.hidden = true;
    terbuka = false;
    setExpanded(false);
    setAktif(-1);
  }

  function teksJumlah() {
    return hasil.length
      ? hasil.length + ' kota ditemukan. Gunakan panah atas dan bawah untuk menelusuri.'
      : 'Tidak ada kota yang cocok.';
  }

  function perbarui(umumkanHasil) {
    clearTimeout(timer);
    timer = 0;
    hasil = saring(input.value);
    renderDaftar();
    if (hasil.length) buka(); else tutup();

    var ketikan = input.value.trim();
    if (!hasil.length && ketikan) {
      kosong.textContent = 'Tidak ada kota yang cocok dengan “' + ketikan + '”.'; // textContent: aman
      kosong.hidden = false;
    } else {
      kosong.hidden = true;
    }
    if (umumkanHasil) umumkan(teksJumlah());
  }

  /* ---------- Chip & pesan batas ---------- */
  function renderChip() {
    var frag = document.createDocumentFragment();
    terpilih.forEach(function (nama) {
      var li = document.createElement('li');
      li.className = 'chip';
      var span = document.createElement('span');
      span.className = 'chip-label';
      span.textContent = nama;
      var btn = document.createElement('button');
      btn.type = 'button';
      btn.className = 'chip-hapus';
      btn.dataset.nama = nama;
      btn.setAttribute('aria-label', 'Hapus ' + nama);
      var x = document.createElement('span');
      x.setAttribute('aria-hidden', 'true');
      x.textContent = '×';
      btn.appendChild(x);
      li.appendChild(span);
      li.appendChild(btn);
      frag.appendChild(li);
    });
    chips.replaceChildren(frag);
    chips.hidden = terpilih.length === 0;

    if (terpilih.length >= MAKS) {
      pesan.textContent = 'Batas maksimal ' + MAKS + ' kota sudah tercapai. Hapus salah satu kota untuk memilih kota lain.';
      pesan.classList.add('warn');
    } else {
      pesan.textContent = terpilih.length + ' dari ' + MAKS + ' kota dipilih.';
      pesan.classList.remove('warn');
    }
  }

  function peringatanBatas() {
    pesan.classList.remove('kedip');
    void pesan.offsetWidth; // mulai ulang animasi
    pesan.classList.add('kedip');
  }

  function pilih(d) {
    if (terpilih.indexOf(d.nama) !== -1) {
      hapus(d.nama, false);
      return;
    }
    if (terpilih.length >= MAKS) {
      peringatanBatas();
      umumkan('Batas maksimal ' + MAKS + ' kota tercapai. ' + d.nama + ' tidak ditambahkan. Hapus salah satu kota terlebih dahulu.');
      return;
    }
    terpilih.push(d.nama);
    simpan();
    renderChip();
    input.value = '';
    perbarui(false);
    var i = hasil.findIndex(function (h) { return h.d === d; });
    if (terbuka && i >= 0) setAktif(i);
    var teks = d.nama + ' ditambahkan. ' + terpilih.length + ' dari ' + MAKS + ' kota dipilih.';
    if (terpilih.length >= MAKS) teks += ' Batas maksimal tercapai.';
    umumkan(teks);
  }

  function hapus(nama, fokusInput) {
    var idx = terpilih.indexOf(nama);
    if (idx === -1) return;
    terpilih.splice(idx, 1);
    simpan();
    renderChip();
    if (terbuka) {
      var a = aktif;
      renderDaftar();
      setAktif(a, false);
    }
    umumkan(nama + ' dihapus. ' + terpilih.length + ' dari ' + MAKS + ' kota dipilih.');
    if (fokusInput) input.focus();
  }

  /* ---------- Event ---------- */
  input.addEventListener('input', function () {
    clearTimeout(timer);
    timer = setTimeout(function () { perbarui(true); }, JEDA);
  });

  input.addEventListener('keydown', function (e) {
    var n;
    switch (e.key) {
      case 'ArrowDown':
        e.preventDefault();
        if (!terbuka || timer) {
          perbarui(false);
          if (terbuka) { setAktif(0); umumkan(teksJumlah()); }
        } else {
          n = hasil.length;
          setAktif(aktif < 0 ? 0 : (aktif + 1) % n);
        }
        break;

      case 'ArrowUp':
        e.preventDefault();
        if (!terbuka || timer) {
          perbarui(false);
          if (terbuka) { setAktif(hasil.length - 1); umumkan(teksJumlah()); }
        } else {
          n = hasil.length;
          setAktif(aktif <= 0 ? n - 1 : aktif - 1);
        }
        break;

      case 'Home':
        if (terbuka && hasil.length) { e.preventDefault(); setAktif(0); }
        break;

      case 'End':
        if (terbuka && hasil.length) { e.preventDefault(); setAktif(hasil.length - 1); }
        break;

      case 'Enter':
        e.preventDefault(); // jangan mengirim formulir dari kolom ini
        if (timer) perbarui(true);
        if (terbuka && aktif >= 0 && hasil[aktif]) pilih(hasil[aktif].d);
        break;

      case 'Escape':
        if (terbuka) {
          e.preventDefault();
          tutup();
        } else if (input.value !== '') {
          e.preventDefault();
          clearTimeout(timer);
          timer = 0;
          input.value = '';
          kosong.hidden = true;
          umumkan('Kolom pencarian dikosongkan.');
        }
        break;

      case 'Backspace':
        if (input.value === '' && terpilih.length) {
          e.preventDefault();
          hapus(terpilih[terpilih.length - 1], false);
        }
        break;
    }
  });

  input.addEventListener('click', function () {
    if (!terbuka) {
      perbarui(false);
      if (terbuka) umumkan(teksJumlah());
    }
  });

    if (terbuka) {
      tutup();
    } else {
      perbarui(false);
      if (terbuka) umumkan(teksJumlah());
    }
    input.focus();
  });

  // Fokus tetap di input saat opsi diklik.
  listbox.addEventListener('mousedown', function (e) { e.preventDefault(); });
  listbox.addEventListener('click', function (e) {
    var li = e.target.closest('[role="option"]');
    if (!li) return;
    var i = Number(li.dataset.indeks);
    if (!hasil[i]) return;
    setAktif(i, false);
    pilih(hasil[i].d);
  });
  listbox.addEventListener('mousemove', function (e) {
    var li = e.target.closest('[role="option"]');
    if (!li) return;
    var i = Number(li.dataset.indeks);
    if (i !== aktif) setAktif(i, false);
  });

  chips.addEventListener('click', function (e) {
    var btn = e.target.closest('button[data-nama]');
    if (btn) hapus(btn.dataset.nama, true);
  });

  // Klik di luar komponen menutup daftar.
  document.addEventListener('pointerdown', function (e) {
    if (terbuka && !combo.contains(e.target)) tutup();
  });
  // Fokus keluar dari komponen (mis. Tab) juga menutup daftar.
  combo.addEventListener('focusout', function (e) {
    if (!e.relatedTarget || !combo.contains(e.relatedTarget)) tutup();
  });

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    tutup();
    var data = { kota_tujuan: terpilih.slice(), jumlah: terpilih.length };
    hasilJson.textContent = JSON.stringify(data, null, 2);
    hasilSec.hidden = false;
    umumkan(terpilih.length
      ? 'Terkirim. ' + terpilih.length + ' kota tujuan ditampilkan sebagai JSON.'
      : 'Terkirim tanpa kota tujuan. Daftar kosong ditampilkan sebagai JSON.');
  });

  /* ---------- Inisialisasi ---------- */
  terpilih = muat();
  renderChip();
})();
</script>
</body>
</html>

```

Cek kebutuhan

* Satu file HTML tanpa sumber eksternal → seluruh CSS ada di `<style>` dan JS di `<script>`. Font memakai `system-ui`, ikon berupa karakter `▾ ✓ ×`.
* Minimal 100 kota/kabupaten disimpan di file → array `KOTA` berisi 125 entri dari Aceh sampai Papua.
* Input `role="combobox"` dengan `aria-expanded`, `aria-controls`, `aria-autocomplete="list"`, `aria-activedescendant` → atribut ada pada `#kota-input`. `aria-expanded` diatur oleh `setExpanded()`, dan `aria-activedescendant` diatur oleh `setAktif()`.
* `role="listbox"` berisi `role="option"` dengan `aria-selected` → `#kota-listbox` (dengan `aria-multiselectable="true"`), sedangkan opsinya dibuat di `renderDaftar()`.
* Penyaringan tanpa peka huruf besar-kecil dan diakritik → `lipat()` memakai `normalize('NFD')`, membuang tanda diakritik `̀–ͯ`, lalu `toLowerCase()`.
* Debounce sekitar 150 ms → konstanta `JEDA = 150` dipakai di listener `input`.
* Bagian yang cocok ditandai `<mark>` → `sorot()` memakai peta indeks dari `lipat()` sehingga posisi sorotan tetap tepat walaupun teks dinormalisasi.
* Panah atas/bawah berputar di ujung daftar → kasus `ArrowDown`/`ArrowUp` memakai `(aktif + 1) % n` dan `aktif <= 0 ? n - 1`.
* Home/End → kasus `Home`/`End` saat daftar terbuka.
* Enter memilih → kasus `Enter` memanggil `pilih()`. Enter pada opsi yang sudah dipilih akan membatalkan pilihan itu.
* Escape menutup, Escape kedua mengosongkan input → kasus `Escape` dengan dua cabang: `terbuka` dan `input.value !== ''`.
* Backspace pada input kosong menghapus pilihan terakhir → kasus `Backspace` memanggil `hapus(terpilih[terpilih.length - 1])`.
* Chip dengan tombol hapus ber-`aria-label` "Hapus <nama kota>" → `renderChip()` memasang `btn.setAttribute('aria-label', 'Hapus ' + nama)`.
* Maksimal 5 pilihan dengan pesan yang jelas → `MAKS = 5`, pengecekan di `pilih()`, pesan peringatan di `#kota-pesan` (kelas `.warn`), opsi lain diberi `aria-disabled`, dan pengumuman lewat live region.
* Jumlah hasil diumumkan lewat live region → `#kota-live` (`role="status"`, `aria-live="polite"`) melalui `umumkan(teksJumlah())`.
* Klik di luar menutup daftar → listener `pointerdown` pada `document`, ditambah `focusout` pada `#combo`.
* Ketikan tidak pernah disisipkan sebagai HTML → tidak ada `innerHTML` sama sekali. Semua teks masuk lewat `textContent` atau `createTextNode`, termasuk pesan "Tidak ada kota yang cocok dengan …". Karena itu `<img src=x onerror=alert(1)>` hanya tampil sebagai teks biasa.
* Tombol Kirim menampilkan JSON → handler `submit` mengisi `#hasil-json` dengan `JSON.stringify({ kota_tujuan, jumlah })`.
* Responsif sampai 375 px → `max-width` cair, chip yang bisa membungkus baris, `@media (max-width: 400px)`, ukuran font input 16px agar iOS tidak memperbesar halaman, dan target sentuh minimal 32–44px.
* Akses localStorage dibungkus try/catch → `muat()` dan `simpan()`. Data yang dimuat juga divalidasi terhadap `PER_NAMA` dan dibatasi maksimal 5.

Catatan teknis

* `aria-activedescendant` hanya dipasang saat ada opsi yang disorot dan dihapus jika tidak ada. Ini sesuai praktik APG, karena atribut yang berisi ID kosong atau tidak valid bisa membingungkan pembaca layar.
* Home/End hanya mengambil alih sorotan saat daftar terbuka. Saat daftar tertutup, kedua tombol tetap menggerakkan kursor teks seperti biasa.
* Setelah sebuah kota dipilih, kolom pencarian dikosongkan dan daftar tetap terbuka dengan sorotan di kota tersebut, sehingga pengguna bisa langsung memilih kota berikutnya.
* Enter di dalam combobox tidak mengirim formulir. Pengiriman hanya terjadi lewat tombol Kirim.
* Jika Enter atau panah ditekan sebelum debounce selesai, hasil penyaringan langsung dihitung ulang agar sorotan tidak mengacu ke daftar yang sudah usang.
