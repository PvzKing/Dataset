```html
(function () {
  'use strict';

  var KEY = 'perkara-advokat-v1';
  var JENIS = ['Perdata', 'Pidana', 'TUN', 'Niaga'];
  var STATUS = ['Baru', 'Berjalan', 'Ditunda', 'Selesai'];
  var HARI = ['Min', 'Sen', 'Sel', 'Rab', 'Kam', 'Jum', 'Sab'];
  var BULAN = ['Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun', 'Jul', 'Agu', 'Sep', 'Okt', 'Nov', 'Des'];

  var editId = null;
  var penyimpananSiap = true;
  var timerPesan = null;

  /* ---------- Utilitas ---------- */

  function el(id) { return document.getElementById(id); }

  function isoDari(d) {
    return d.getFullYear() + '-' +
      String(d.getMonth() + 1).padStart(2, '0') + '-' +
      String(d.getDate()).padStart(2, '0');
  }

  function hariIni() {
    var d = new Date();
    d.setHours(0, 0, 0, 0);
    return d;
  }

  function geserHari(n) {
    var d = hariIni();
    d.setDate(d.getDate() + n);
    return isoDari(d);
  }

  function keTanggal(iso) {
    if (!iso || typeof iso !== 'string') return null;
    var p = iso.split('-');
    if (p.length !== 3) return null;
    var d = new Date(Number(p[0]), Number(p[1]) - 1, Number(p[2]));
    return isNaN(d.getTime()) ? null : d;
  }

  function selisihHari(iso) {
    var d = keTanggal(iso);
    if (!d) return null;
    d.setHours(0, 0, 0, 0);
    return Math.round((d - hariIni()) / 86400000);
  }

  function formatTanggal(iso) {
    var d = keTanggal(iso);
    if (!d) return '—';
    return HARI[d.getDay()] + ', ' + d.getDate() + ' ' + BULAN[d.getMonth()] + ' ' + d.getFullYear();
  }

  function labelSelisih(n) {
    if (n === null) return '';
    if (n === 0) return 'Hari ini';
    if (n === 1) return 'Besok';
    if (n > 1) return n + ' hari lagi';
    return Math.abs(n) + ' hari lalu';
  }

  function esc(s) {
    return String(s === null || s === undefined ? '' : s).replace(/[&<>"']/g, function (c) {
      return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
  }

  function slug(s) {
    return String(s).toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '');
  }

  /* ---------- Data contoh (fiktif) ---------- */

  function dataContoh() {

  /* ---------- Penyimpanan ---------- */

  function normalkan(p) {
    if (!p || typeof p !== 'object') return null;
    return {
      id: p.id ? String(p.id) : 'c' + Math.random().toString(36).slice(2, 10),
      nomor: String(p.nomor || ''),
      jenis: JENIS.indexOf(p.jenis) >= 0 ? p.jenis : 'Perdata',
      pengadilan: String(p.pengadilan || ''),
      pihak: String(p.pihak || ''),
      klien: String(p.klien || ''),
      status: STATUS.indexOf(p.status) >= 0 ? p.status : 'Baru',
      sidang: keTanggal(p.sidang) ? p.sidang : ''
    };
  }

  function muat() {
    try {
      var mentah = window.localStorage.getItem(KEY);
      if (mentah) {
        var data = JSON.parse(mentah);
        if (Array.isArray(data)) {
          perkara = data.map(normalkan).filter(function (x) { return x !== null; });
          return;
        }
      }
    } catch (e) {
      penyimpananSiap = false;
      if (window.console && console.warn) console.warn('localStorage tidak dapat dibaca:', e);
    }
    perkara = dataContoh();
    simpan();
  }

  function simpan() {
    try {
      window.localStorage.setItem(KEY, JSON.stringify(perkara));
      penyimpananSiap = true;
    } catch (e) {
      penyimpananSiap = false;
      if (window.console && console.warn) console.warn('localStorage tidak dapat ditulis:', e);
    }
    perbaruiNotif();
  }

  function perbaruiNotif() {
    var n = el('notifPenyimpanan');
    if (!n) return;
    if (penyimpananSiap) {
      n.className = '';
      n.textContent = 'Data tersimpan otomatis di peramban ini (localStorage).';
    } else {
      n.className = 'peringatan';
      n.textContent = 'Peringatan: localStorage tidak tersedia. Perubahan hanya berlaku selama halaman ini terbuka.';
    }
  }

  /* ---------- Render: sidang 7 hari ke depan ---------- */

  function renderSidang() {
    var wadah = el('daftarSidang');

    var items = perkara.filter(function (p) {
      if (!p.sidang || p.status === 'Selesai') return false;
      var s = selisihHari(p.sidang);
      return s !== null && s >= 0 && s <= 7;
    }).sort(function (a, b) {
      return a.sidang < b.sidang ? -1 : a.sidang > b.sidang ? 1 : 0;
    });

    el('hitungSidang').textContent = items.length;

    if (!items.length) {
      wadah.innerHTML = '<p class="kosong">Tidak ada sidang dalam 7 hari ke depan.</p>';
      return;
    }

    wadah.innerHTML = '<ul class="sidang-list">' + items.map(function (p) {
      var s = selisihHari(p.sidang);
      var besok = (s === 1);
      return '<li class="sidang-item' + (besok ? ' besok' : '') + '">' +
        '<div class="sidang-tgl">' +
          '<span class="tgl">' + esc(formatTanggal(p.sidang)) + '</span>' +
          '<span class="selisih' + (besok ? ' merah' : '') + '">' +

          '</span>' +
        '</div>' +
        '<div class="sidang-isi">' +
          '<p class="sidang-nomor">' + esc(p.nomor) + '</p>' +

        '</div>' +
      '</li>';
    }).join('') + '</ul>';
  }

    if (p.sidang) {
      jadwal = esc(formatTanggal(p.sidang)) +

          esc(besok ? 'besok' : labelSelisih(s).toLowerCase()) + ')</span>');
    } else {
      jadwal = '<span class="muted">Belum dijadwalkan</span>';
    }

    return '<article class="card' + (besok ? ' besok' : '') + '">' +
      '<div class="card-top">' +
        '<h3>' + esc(p.nomor) + '</h3>' +
        '<div class="badges">' +
          '<span class="badge b-' + slug(p.jenis) + '">' + esc(p.jenis) + '</span>' +
          '<span class="badge s-' + slug(p.status) + '">' + esc(p.status) + '</span>' +
        '</div>' +
      '</div>' +
      '<dl>' +
        '<div><dt>Pengadilan</dt><dd>' + esc(p.pengadilan || '—') + '</dd></div>' +
        '<div><dt>Para pihak</dt><dd>' + esc(p.pihak || '—') + '</dd></div>' +
        '<div><dt>Klien</dt><dd>' + esc(p.klien || '—') + '</dd></div>' +
        '<div><dt>Sidang</dt><dd>' + jadwal + '</dd></div>' +

        '<button type="button" class="btn btn-sm btn-ghost" data-aksi="edit" data-id="' + esc(p.id) + '">Ubah</button>' +

      '</div>' +
    '</article>';
  }

  function renderPerkara() {
    var q = el('cari').value.trim().toLowerCase();
    var fj = el('filterJenis').value;
    var fs = el('filterStatus').value;

    var items = perkara.filter(function (p) {
      if (fj && p.jenis !== fj) return false;
      if (fs && p.status !== fs) return false;
      if (q) {
        var teks = [p.nomor, p.pihak, p.klien, p.pengadilan].join(' ').toLowerCase();
        if (teks.indexOf(q) === -1) return false;
      }
      return true;
    }).sort(function (a, b) {
      var sa = a.sidang || '9999-99-99';
      var sb = b.sidang || '9999-99-99';
      if (sa < sb) return -1;
      if (sa > sb) return 1;
      return a.nomor < b.nomor ? -1 : a.nomor > b.nomor ? 1 : 0;
    });

    var total = perkara.length;
    el('hitungPerkara').textContent = (items.length === total)
      ? String(total)
      : items.length + ' / ' + total;

    el('daftarPerkara').innerHTML = items.map(kartuPerkara).join('');
    el('kosongPerkara').hidden = items.length > 0;
  }

  function render() {
    renderSidang();
    renderPerkara();
  }

  /* ---------- Pesan ---------- */

  function tampilPesan(teks, tipe) {
    var p = el('pesan');
    p.textContent = teks;
    p.className = 'pesan ' + (tipe || 'ok');
    p.hidden = false;
    if (timerPesan) window.clearTimeout(timerPesan);
    timerPesan = window.setTimeout(function () { p.hidden = true; }, 4000);
  }

  /* ---------- Aksi ---------- */

  function cariPerkara(id) {
    for (var i = 0; i < perkara.length; i++) {
      if (perkara[i].id === id) return perkara[i];
    }
    return null;
  }

  function resetForm() {
    el('formPerkara').reset();
    editId = null;
    el('judulForm').textContent = 'Tambah Perkara';
    el('btnBatal').hidden = true;
    el('formPerkara').querySelector('button[type="submit"]').textContent = 'Simpan perkara';
  }

  function mulaiEdit(id) {
    var p = cariPerkara(id);
    if (!p) return;

    editId = p.id;
    el('fNomor').value = p.nomor;
    el('fJenis').value = p.jenis;
    el('fPengadilan').value = p.pengadilan;
    el('fPihak').value = p.pihak;
    el('fKlien').value = p.klien;
    el('fStatus').value = p.status;
    el('fSidang').value = p.sidang || '';

    el('judulForm').textContent = 'Ubah Perkara';
    el('btnBatal').hidden = false;
    el('formPerkara').querySelector('button[type="submit"]').textContent = 'Simpan perubahan';
    el('pesan').hidden = true;

    var form = el('formPerkara');
    if (form.scrollIntoView) form.scrollIntoView({ behavior: 'smooth', block: 'start' });
    try { el('fNomor').focus({ preventScroll: true }); } catch (e) { el('fNomor').focus(); }
  }

  function hapusPerkara(id) {
    var p = cariPerkara(id);
    if (!p) return;
    if (!window.confirm('Hapus perkara ' + p.nomor + '?')) return;

    perkara = perkara.filter(function (x) { return x.id !== id; });
    if (editId === id) resetForm();
    simpan();
    render();
    tampilPesan('Perkara dihapus.', 'ok');
  }

  /* ---------- Isi pilihan select ---------- */

  function isiPilihan() {
    var fj = el('filterJenis'), fs = el('filterStatus');
    var fJenis = el('fJenis'), fStatus = el('fStatus');

    JENIS.forEach(function (j) {
      var a = document.createElement('option');
      a.value = j; a.textContent = j; fj.appendChild(a);

      var b = document.createElement('option');
      b.value = j; b.textContent = j; fJenis.appendChild(b);
    });

    STATUS.forEach(function (s) {
      var a = document.createElement('option');
      a.value = s; a.textContent = s; fs.appendChild(a);

      var b = document.createElement('option');
      b.value = s; b.textContent = s; fStatus.appendChild(b);
    });
  }

  /* ---------- Event ---------- */

  function bind() {
    el('cari').addEventListener('input', renderPerkara);
    el('filterJenis').addEventListener('change', renderPerkara);
    el('filterStatus').addEventListener('change', renderPerkara);

    el('btnReset').addEventListener('click', function () {
      el('cari').value = '';
      el('filterJenis').value = '';
      el('filterStatus').value = '';
      renderPerkara();
    });

    el('btnBatal').addEventListener('click', function () {
      resetForm();
      el('pesan').hidden = true;
    });

    document.addEventListener('click', function (e) {
      var tombol = e.target && e.target.closest ? e.target.closest('[data-aksi]') : null;
      if (!tombol) return;
      var id = tombol.getAttribute('data-id');
      if (tombol.getAttribute('data-aksi') === 'edit') {
        mulaiEdit(id);
      } else if (tombol.getAttribute('data-aksi') === 'hapus') {
        hapusPerkara(id);
      }
    });

    el('formPerkara').addEventListener('submit', function (e) {
      e.preventDefault();

      var nomor = el('fNomor').value.trim();
      var jenis = el('fJenis').value;
      var pengadilan = el('fPengadilan').value.trim();
      var pihak = el('fPihak').value.trim();
      var klien = el('fKlien').value.trim();
      var status = el('fStatus').value;
      var sidang = el('fSidang').value;

      if (!nomor || !klien) {
        tampilPesan('Nomor perkara dan nama klien wajib diisi.', 'error');
        (!nomor ? el('fNomor') : el('fKlien')).focus();
        return;
      }

      var kembar = perkara.some(function (p) {
        return p.id !== editId && p.nomor.toLowerCase() === nomor.toLowerCase();
      });
      if (kembar) {
        tampilPesan('Nomor perkara tersebut sudah terdaftar.', 'error');
        el('fNomor').focus();
        return;
      }

      var sedangEdit = editId !== null;

      if (sedangEdit) {
        var target = cariPerkara(editId);
        if (target) {
          target.nomor = nomor;
          target.jenis = jenis;
          target.pengadilan = pengadilan;
          target.pihak = pihak;
          target.klien = klien;
          target.status = status;
          target.sidang = sidang;
        }
      } else {
        perkara.push({
          id: 'c' + Date.now().toString(36) + Math.random().toString(36).slice(2, 6),
          nomor: nomor,
          jenis: jenis,
          pengadilan: pengadilan,
          pihak: pihak,
          klien: klien,
          status: status,
          sidang: sidang
        });
      }

      simpan();
      render();
      resetForm();
      tampilPesan(sedangEdit ? 'Perkara berhasil diperbarui.' : 'Perkara baru berhasil ditambahkan.', 'ok');
    });
  }

  /* ---------- Inisialisasi ---------- */

  isiPilihan();
  muat();
  bind();
  render();
  perbaruiNotif();
})();
</script>
</body>
</html>
```

**Cek kebutuhan**

- **Setiap perkara punya nomor perkara, jenis (perdata, pidana, TUN, niaga), pengadilan, para pihak, klien, status, dan jadwal sidang berikutnya** → objek perkara pada `dataContoh()`, `normalkan()`, dan field `#fNomor`, `#fJenis`, `#fPengadilan`, `#fPihak`, `#fKlien`, `#fStatus`, `#fSidang` di `#formPerkara`
- **Pencarian** → input `#cari` dan penyaringan `q` di dalam `renderPerkara()`
- **Filter jenis dan status** → `#filterJenis` dan `#filterStatus` (diisi lewat `isiPilihan()`), dipakai di `renderPerkara()`
- **Daftar sidang 7 hari ke depan di bagian atas** → `renderSidang()` dengan `selisihHari()` bernilai 0–7, dirender ke `#daftarSidang` yang berada di panel pertama
- **Tanda merah untuk sidang besok** → kelas `besok` pada `renderSidang()` (`.sidang-item.besok`, `.selisih.merah`, penanda `●`) dan pada `kartuPerkara()`
- **Data tersimpan di localStorage** → `muat()`, `simpan()`, kunci `perkara-advokat-v1`, seluruh akses `localStorage` dibungkus `try/catch` dan dilaporkan lewat `perbaruiNotif()`
- **Data contoh fiktif** → `dataContoh()` berisi delapan perkara dengan tanggal sidang relatif terhadap hari ini
- **Responsif sampai 375 px** → `meta viewport`, `.toolbar` dengan `flex-wrap`, `.grid` `minmax(280px, 1fr)`, media query 520 px dan 600 px
- **Tanpa library, font, gambar, atau file luar** → seluruh CSS dan JavaScript tertulis inline di dalam file

**Catatan teknis**

- Tanggal contoh dibuat dinamis (`geserHari()`) agar daftar sidang 7 hari selalu terisi saat file dibuka kapan pun.
- Perhitungan selisih hari memakai tengah malam lokal, sehingga "besok" tidak terpengaruh jam akses.
- Perkara berstatus `Selesai` tidak dimunculkan di daftar sidang mendatang, tetapi tetap tampil di daftar perkara.
- Validasi menolak nomor perkara kosong, klien kosong, dan nomor perkara duplikat.
- Bila localStorage diblokir peramban, aplikasi tetap berjalan dan menampilkan peringatan di bagian bawah halaman.
