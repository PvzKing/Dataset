```html
<script>
(function () {
  'use strict';

  var COLS = ['A','B','C','D','E','F','G','H','I','J'];
  var ROWS = 20;
  var KEY = 'mini-spreadsheet-v1';

  var cells = Object.create(null);   // id -> teks mentah
  var cache = new Map();             // id -> {value, err}
  var cycleSet = new Set();          // id yang berada dalam lingkaran
  var stack = [];                    // tumpukan evaluasi
  var onStack = new Set();

  var ERR = { k: 'err' };
  var CYCLE = { k: 'cycle' };

  /* ---------- util ---------- */

  function isRef(s) {
    return typeof s === 'string' && /^[A-J]([1-9]|1[0-9]|20)$/.test(s);
  }

  function fmt(n) {
    if (!isFinite(n)) return '#ERROR';
    var r = Math.round(n * 1e10) / 1e10;
    return String(r);
  }

  /* ---------- evaluator ---------- */

  function evalCell(id) {
    var hit = cache.get(id);
    if (hit) return hit;

    if (onStack.has(id)) {
      var start = stack.indexOf(id);
      for (var k = start; k < stack.length; k++) cycleSet.add(stack[k]);
      throw CYCLE;
    }

    stack.push(id);
    onStack.add(id);

    var res;
    try {
      var raw = cells[id] === undefined ? '' : String(cells[id]);
      if (raw === '') {
        res = { value: '', err: null };
      } else if (raw.trim().charAt(0) === '=') {
        res = { value: evalFormula(raw.trim().slice(1)), err: null };
      } else {
        var t = raw.trim();
        var n = Number(t);
        res = { value: (t !== '' && !isNaN(n)) ? n : raw, err: null };
      }
    } catch (e) {
      if (e === CYCLE) {
        var inCycle = cycleSet.has(id);
        res = { value: inCycle ? '#SIKLUS' : '#ERROR', err: inCycle ? 'cycle' : 'error' };
      } else {
        res = { value: '#ERROR', err: 'error' };
      }
    }

    stack.pop();
    onStack.delete(id);
    cache.set(id, res);
    return res;
  }

  function refValue(id) {
    if (!isRef(id)) throw ERR;
    var r = evalCell(id);
    if (r.err) throw (r.err === 'cycle' ? CYCLE : ERR);
    return r.value;
  }

  function refNumber(id) {
    var v = refValue(id);
    if (v === '') return 0;
    if (typeof v === 'number') return v;
    throw ERR;
  }

  function rangeValues(a, b) {
    if (!isRef(a) || !isRef(b)) throw ERR;
    var c1 = Math.min(COLS.indexOf(a.charAt(0)), COLS.indexOf(b.charAt(0)));
    var c2 = Math.max(COLS.indexOf(a.charAt(0)), COLS.indexOf(b.charAt(0)));
    var r1 = Math.min(parseInt(a.slice(1), 10), parseInt(b.slice(1), 10));
    var r2 = Math.max(parseInt(a.slice(1), 10), parseInt(b.slice(1), 10));
    var out = [];
    for (var c = c1; c <= c2; c++) {
      for (var r = r1; r <= r2; r++) {
        var res = evalCell(COLS[c] + r);
        if (res.err) throw (res.err === 'cycle' ? CYCLE : ERR);
        if (typeof res.value === 'number') out.push(res.value);
      }
    }
    return out;
  }

  /* ---------- tokenizer ---------- */

  function tokenize(src) {
    var out = [];
    var i = 0;
    while (i < src.length) {
      var ch = src.charAt(i);
      if (ch === ' ' || ch === '\t' || ch === '\n' || ch === '\r') { i++; continue; }
      if ((ch >= '0' && ch <= '9') || ch === '.') {
        var j = i;
        while (j < src.length && ((src[j] >= '0' && src[j] <= '9') || src[j] === '.')) j++;
        var num = Number(src.slice(i, j));
        if (!isFinite(num)) throw ERR;
        out.push({ t: 'num', v: num });
        i = j;
        continue;
      }
      if (/[A-Za-z]/.test(ch)) {
        var k = i;
        while (k < src.length && /[A-Za-z0-9]/.test(src[k])) k++;
        out.push({ t: 'id', v: src.slice(i, k).toUpperCase() });
        i = k;
        continue;
      }
      if ('+-*/():,'.indexOf(ch) >= 0) {
        out.push({ t: ch });
        i++;
        continue;
      }
      throw ERR;
    }
    return out;
  }

  /* ---------- parser (recursive descent) ---------- */

  function makeParser(tokens) {
    var p = 0;

    function peek() { return tokens[p] || null; }
    function expect(t) {
      if (!tokens[p] || tokens[p].t !== t) throw ERR;
      p++;
    }

    function parseExpr() {
      var v = parseTerm();
      for (;;) {
        var tk = peek();
        if (tk && (tk.t === '+' || tk.t === '-')) {
          p++;
          var r = parseTerm();
          v = (tk.t === '+') ? v + r : v - r;
        } else break;
      }
      return v;
    }

    function parseTerm() {
      var v = parseFactor();
      for (;;) {
        var tk = peek();
        if (tk && (tk.t === '*' || tk.t === '/')) {
          p++;
          var r = parseFactor();
          if (tk.t === '/') {
            if (r === 0) throw ERR;
            v = v / r;
          } else {
            v = v * r;
          }
        } else break;
      }
      return v;
    }

    function parseFactor() {
      var tk = peek();
      if (tk && tk.t === '-') { p++; return -parseFactor(); }
      if (tk && tk.t === '+') { p++; return parseFactor(); }
      return parsePrimary();
    }

    function parsePrimary() {
      var tk = peek();
      if (!tk) throw ERR;
      if (tk.t === 'num') { p++; return tk.v; }
      if (tk.t === '(') { p++; var v = parseExpr(); expect(')'); return v; }
      if (tk.t === 'id') {
        if (tokens[p + 1] && tokens[p + 1].t === '(') {
          var name = tk.v;
          p += 2;
          var args = parseArgs();
          expect(')');
          return callFunc(name, args);
        }
        p++;
        return refNumber(tk.v);
      }
      throw ERR;
    }

    function parseArgs() {
      var args = [];
      if (peek() && peek().t === ')') return args;
      args.push(parseArg());
      while (peek() && peek().t === ',') {
        p++;
        args.push(parseArg());
      }
      return args;
    }

    function parseArg() {
      var tk = peek();
      if (tk && tk.t === 'id' && tokens[p + 1] && tokens[p + 1].t === ':') {
        var a = tk.v;
        p += 2;
        var bt = peek();
        if (!bt || bt.t !== 'id') throw ERR;
        p++;
        return { range: [a, bt.v] };
      }
      return { num: parseExpr() };
    }

    return { parseExpr: parseExpr, pos: function () { return p; } };
  }

  function callFunc(name, args) {
    var nums = [], i, j;
    for (i = 0; i < args.length; i++) {
      if (args[i].range) {
        var vals = rangeValues(args[i].range[0], args[i].range[1]);
        for (j = 0; j < vals.length; j++) nums.push(vals[j]);
      } else {
        nums.push(args[i].num);
      }
    }
    var s = 0, m;
    switch (name) {
      case 'SUM':
        for (i = 0; i < nums.length; i++) s += nums[i];
        return s;
      case 'AVERAGE':
        if (!nums.length) throw ERR;
        for (i = 0; i < nums.length; i++) s += nums[i];
        return s / nums.length;
      case 'MIN':
        if (!nums.length) throw ERR;
        m = nums[0];
        for (i = 1; i < nums.length; i++) if (nums[i] < m) m = nums[i];
        return m;
      case 'MAX':
        if (!nums.length) throw ERR;
        m = nums[0];
        for (i = 1; i < nums.length; i++) if (nums[i] > m) m = nums[i];
        return m;
      default:
        throw ERR;
    }
  }

  function evalFormula(src) {
    var tokens = tokenize(src);
    if (!tokens.length) throw ERR;
    var parser = makeParser(tokens);
    var v = parser.parseExpr();
    if (parser.pos() !== tokens.length) throw ERR;
    if (typeof v !== 'number' || !isFinite(v)) throw ERR;
    return v;
  }

  /* ---------- tampilan ---------- */

  function displayOf(id) {
    var raw = cells[id];
    if (raw === undefined || raw === '') return { text: '', err: false };
    var res = evalCell(id);
    if (res.err) return { text: res.value, err: true };
    if (typeof res.value === 'number') return { text: fmt(res.value), err: false };
    return { text: String(res.value), err: false };
  }

  var table = document.getElementById('sheet');
  var statusEl = document.getElementById('status');
  var inputs = [];

  function buildTable() {
    var thead = document.createElement('thead');
    var hr = document.createElement('tr');
    var corner = document.createElement('th');
    corner.setAttribute('scope', 'col');
    hr.appendChild(corner);
    for (var c = 0; c < COLS.length; c++) {
      var th = document.createElement('th');
      th.setAttribute('scope', 'col');
      th.textContent = COLS[c];
      hr.appendChild(th);
    }
    thead.appendChild(hr);
    table.appendChild(thead);

    var tbody = document.createElement('tbody');
    for (var r = 1; r <= ROWS; r++) {
      var tr = document.createElement('tr');
      var rh = document.createElement('th');
      rh.setAttribute('scope', 'row');
      rh.textContent = r;
      tr.appendChild(rh);
      for (var k = 0; k < COLS.length; k++) {
        var td = document.createElement('td');
        var inp = document.createElement('input');
        inp.type = 'text';
        inp.autocomplete = 'off';
        inp.spellcheck = false;
        inp.setAttribute('aria-label', COLS[k] + r);
        inp.dataset.id = COLS[k] + r;
        td.appendChild(inp);
        tr.appendChild(td);
        inputs.push(inp);
      }
      tbody.appendChild(tr);
    }
    table.appendChild(tbody);
  }

  function refresh(skipId) {
    cache.clear();
    cycleSet.clear();
    for (var i = 0; i < inputs.length; i++) {
      var inp = inputs[i];
      var id = inp.dataset.id;
      if (id === skipId) continue;
      var d = displayOf(id);
      if (inp.value !== d.text) inp.value = d.text;
      if (d.err) inp.classList.add('err');
      else inp.classList.remove('err');
    }
  }

  /* ---------- penyimpanan ---------- */

  function setStatus(msg) {
    if (statusEl) statusEl.textContent = msg;
  }

  function save() {
    try {
      localStorage.setItem(KEY, JSON.stringify(cells));
      setStatus('Tersimpan otomatis di peramban.');
    } catch (e) {
      setStatus('Gagal menyimpan: penyimpanan lokal tidak tersedia.');
    }
  }

  function load() {
    try {
      var s = localStorage.getItem(KEY);
      if (!s) return;
      var obj = JSON.parse(s);
      if (!obj || typeof obj !== 'object') return;
      for (var k in obj) {
        if (Object.prototype.hasOwnProperty.call(obj, k) && isRef(k) && typeof obj[k] === 'string') {
          cells[k] = obj[k];
        }
      }
    } catch (e) {
      /* data rusak diabaikan */
    }
  }

  /* ---------- event ---------- */

  table.addEventListener('input', function (e) {
    var inp = e.target;
    if (!inp || inp.tagName !== 'INPUT') return;
    var id = inp.dataset.id;
    if (inp.value === '') delete cells[id];
    else cells[id] = inp.value;
    save();
    refresh(id);
  });

  table.addEventListener('focusin', function (e) {
    var inp = e.target;
    if (!inp || inp.tagName !== 'INPUT') return;
    var id = inp.dataset.id;
    inp.value = (cells[id] === undefined) ? '' : cells[id];
  });

  table.addEventListener('focusout', function (e) {
    var inp = e.target;
    if (!inp || inp.tagName !== 'INPUT') return;
    refresh(null);
  });

  table.addEventListener('keydown', function (e) {
    var inp = e.target;
    if (!inp || inp.tagName !== 'INPUT') return;
    if (e.key === 'Enter' || e.key === 'Escape') {
      e.preventDefault();
      inp.blur();
    }
  });

  document.getElementById('clear').addEventListener('click', function () {
    for (var k in cells) delete cells[k];
    save();
    refresh(null);
  });

  /* ---------- mulai ---------- */

  buildTable();
  load();
  refresh(null);
})();
</script>
</body>
</html>
```

**Cek kebutuhan**
- **Grid kolom A–J dan baris 1–20 yang bisa diedit** → `buildTable()` membuat thead A–J dan 20 baris dengan `<input>` di setiap sel; kolom/baris kunci memakai `COLS` dan `ROWS`.
- **Sel bisa berisi angka, teks, atau rumus berawalan `=`** → `evalCell()` membedakan string kosong, awalan `=`, dan teks/angka biasa.
- **Rumus `=A1+B2*2`, `=SUM(A1:A5)`, `=AVERAGE(...)`, `=MIN(...)`, `=MAX(...)`** → `tokenize()`, `makeParser()` (parseExpr/parseTerm/parseFactor/parsePrimary), `parseArg()` untuk rentang, dan `callFunc()` untuk SUM/AVERAGE/MIN/MAX.
- **Hasil rumus diperbarui saat sel rujukan berubah** → handler `input` memanggil `refresh(id)` yang mengosongkan `cache` dan menghitung ulang semua sel.
- **Rujukan melingkar menampilkan `#SIKLUS`** → penanda `onStack`/`stack`, `cycleSet`, dan lemparan `CYCLE` di `evalCell()` serta `refValue()`/`rangeValues()`.
- **Rumus salah menampilkan `#ERROR`** → objek `ERR` dilempar dari tokenizer/parser/evaluator dan ditangkap di blok `catch` `evalCell()`.
- **Tanpa `eval` atau `Function`** → seluruh perhitungan memakai tokenizer dan parser rekursif turun buatan sendiri.
- **Isi sel tersimpan di localStorage** → `save()`/`load()` dengan kunci `mini-spreadsheet-v1`, dipanggil pada setiap perubahan dan saat mulai.
- **Akses localStorage dibungkus try/catch** → seluruh isi `save()` dan `load()` berada di dalam `try { ... } catch (e) { ... }`.
- **Responsif sampai 375 px** → `.gridwrap{overflow:auto}`, `table{min-width:552px}`, header/kolom kunci `position:sticky`, serta media query `@media (max-width:420px)`.
- **Tanpa error JavaScript** → kode dibungkus IIFE `'use strict'`, tidak ada dependensi luar, dan semua akses DOM dilakukan setelah elemen ada.

**Catatan teknis**
Angka desimal dibulatkan ke 10 digit di belakang koma saat ditampilkan; pembagian dengan nol, fungsi tak dikenal, dan hasil tak hingga diperlakukan sebagai `#ERROR`. Sel kosong dianggap `0` dalam operasi aritmetika, sedangkan sel berisi teks non-angka yang dirujuk dalam rumus menghasilkan `#ERROR`. Rentang pada fungsi mengabaikan sel kosong dan sel teks. Saat sebuah sel sedang diedit, sel itu menampilkan teks mentah, sementara sel lain langsung menampilkan hasil terbaru.
