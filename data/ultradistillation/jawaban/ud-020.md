Asumsi yang dipakai: siswa maksimal 3 buku dengan masa pinjam 7 hari; guru maksimal 5 buku dengan masa pinjam 14 hari; denda Rp1.000 per hari; satu eksemplar hanya boleh memiliki satu peminjaman aktif. Batas jumlah pinjam dan larangan peminjaman ganda tidak bisa dijaga hanya dengan `CHECK`, jadi dipakai trigger.

## DDL PostgreSQL

```sql
CREATE TYPE copy_status AS ENUM ('tersedia', 'dipinjam', 'rusak', 'hilang');

CREATE TABLE member_types (
    type_code        text PRIMARY KEY,
    label            text NOT NULL,
    max_active_loans integer NOT NULL CHECK (max_active_loans > 0),
    loan_period_days integer NOT NULL CHECK (loan_period_days > 0),
    daily_fine       numeric(12,2) NOT NULL DEFAULT 1000.00
                     CHECK (daily_fine = 1000.00),
    CONSTRAINT member_types_type_code_check
        CHECK (type_code IN ('siswa', 'guru'))
);

INSERT INTO member_types (type_code, label, max_active_loans, loan_period_days)
VALUES
    ('siswa', 'Siswa', 3, 7),
    ('guru',  'Guru',  5, 14);

CREATE TABLE members (
    member_id    bigserial PRIMARY KEY,
    member_type  text NOT NULL
                 REFERENCES member_types(type_code)
                 ON UPDATE CASCADE ON DELETE RESTRICT,
    nomor_induk  text NOT NULL UNIQUE,
    full_name    text NOT NULL,
    email        text,
    is_active    boolean NOT NULL DEFAULT true,
    created_at   timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT members_nomor_induk_not_blank
        CHECK (btrim(nomor_induk) <> ''),
    CONSTRAINT members_full_name_not_blank
        CHECK (btrim(full_name) <> '')
);

CREATE TABLE books (
    book_id          bigserial PRIMARY KEY,
    isbn             text UNIQUE,
    title            text NOT NULL,
    author           text,
    publisher        text,
    publication_year integer CHECK (publication_year IS NULL OR publication_year > 0),
    created_at       timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT books_title_not_blank
        CHECK (btrim(title) <> '')
);

CREATE TABLE book_copies (
    copy_id    bigserial PRIMARY KEY,
    book_id    bigint NOT NULL
               REFERENCES books(book_id)
               ON DELETE RESTRICT,
    copy_code  text NOT NULL UNIQUE,
    status     copy_status NOT NULL DEFAULT 'tersedia',
    note       text,
    CONSTRAINT book_copies_copy_code_not_blank
        CHECK (btrim(copy_code) <> '')
);

CREATE TABLE loans (
    loan_id    bigserial PRIMARY KEY,
    member_id  bigint NOT NULL
               REFERENCES members(member_id)
               ON DELETE RESTRICT,
    copy_id    bigint NOT NULL
               REFERENCES book_copies(copy_id)
               ON DELETE RESTRICT,
    loan_date  date NOT NULL DEFAULT CURRENT_DATE,
    due_date   date NOT NULL,
    created_at timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT loans_due_after_loan
        CHECK (due_date >= loan_date)
);

CREATE TABLE loan_returns (
    return_id   bigserial PRIMARY KEY,
    loan_id     bigint NOT NULL UNIQUE
                REFERENCES loans(loan_id)
                ON DELETE RESTRICT,
    returned_at date NOT NULL DEFAULT CURRENT_DATE,
    received_by text,
    note        text,
    created_at  timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX idx_loans_member   ON loans(member_id);
CREATE INDEX idx_loans_copy     ON loans(copy_id);
CREATE INDEX idx_loans_due_date ON loans(due_date);
```

Trigger untuk batas pinjam, ketersediaan eksemplar, dan status eksemplar:

```sql
CREATE OR REPLACE FUNCTION fn_loan_before_insert()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
    v_max     integer;
    v_period  integer;
    v_active  bigint;
    v_status  copy_status;
BEGIN
    SELECT mt.max_active_loans, mt.loan_period_days
      INTO v_max, v_period
      FROM members m
      JOIN member_types mt ON mt.type_code = m.member_type
     WHERE m.member_id = NEW.member_id
       AND m.is_active = true
       FOR UPDATE OF m;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Anggota % tidak ditemukan atau tidak aktif', NEW.member_id;
    END IF;

    SELECT count(*)
      INTO v_active
      FROM loans l
      LEFT JOIN loan_returns r ON r.loan_id = l.loan_id
     WHERE l.member_id = NEW.member_id
       AND r.loan_id IS NULL;

    IF v_active >= v_max THEN
        RAISE EXCEPTION 'Batas pinjam anggota % adalah % buku', NEW.member_id, v_max;
    END IF;

    SELECT status
      INTO v_status
      FROM book_copies
     WHERE copy_id = NEW.copy_id
       FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Eksemplar % tidak ditemukan', NEW.copy_id;
    END IF;

    IF v_status <> 'tersedia' THEN
        RAISE EXCEPTION 'Eksemplar % tidak tersedia, status: %', NEW.copy_id, v_status;
    END IF;

    IF EXISTS (
        SELECT 1
          FROM loans l
          LEFT JOIN loan_returns r ON r.loan_id = l.loan_id
         WHERE l.copy_id = NEW.copy_id
           AND r.loan_id IS NULL
    ) THEN
        RAISE EXCEPTION 'Eksemplar % sedang dipinjam', NEW.copy_id;
    END IF;

    NEW.loan_date := COALESCE(NEW.loan_date, CURRENT_DATE);
    NEW.due_date  := NEW.loan_date + v_period;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_loan_before_insert
BEFORE INSERT ON loans
FOR EACH ROW
EXECUTE FUNCTION fn_loan_before_insert();

CREATE OR REPLACE FUNCTION fn_loan_after_insert()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
    UPDATE book_copies
       SET status = 'dipinjam'
     WHERE copy_id = NEW.copy_id;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_loan_after_insert
AFTER INSERT ON loans
FOR EACH ROW
EXECUTE FUNCTION fn_loan_after_insert();
```

Trigger pengembalian:

```sql
CREATE OR REPLACE FUNCTION fn_return_before_insert()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
    v_loan loans%ROWTYPE;
BEGIN
    SELECT *
      INTO v_loan
      FROM loans
     WHERE loan_id = NEW.loan_id
       FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Peminjaman % tidak ditemukan', NEW.loan_id;
    END IF;

    IF EXISTS (SELECT 1 FROM loan_returns WHERE loan_id = NEW.loan_id) THEN
        RAISE EXCEPTION 'Peminjaman % sudah dikembalikan', NEW.loan_id;
    END IF;

    NEW.returned_at := COALESCE(NEW.returned_at, CURRENT_DATE);

    IF NEW.returned_at < v_loan.loan_date THEN
        RAISE EXCEPTION 'Tanggal kembali tidak boleh sebelum tanggal pinjam';
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_return_before_insert
BEFORE INSERT ON loan_returns
FOR EACH ROW
EXECUTE FUNCTION fn_return_before_insert();

CREATE OR REPLACE FUNCTION fn_return_after_insert()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
    UPDATE book_copies c
       SET status = 'tersedia'
      FROM loans l
     WHERE l.loan_id = NEW.loan_id
       AND c.copy_id = l.copy_id;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_return_after_insert
AFTER INSERT ON loan_returns
FOR EACH ROW
EXECUTE FUNCTION fn_return_after_insert();
```

Trigger di atas hanya berjalan saat `INSERT`. Supaya aturan itu tidak bisa dilewati dengan `UPDATE` atau `DELETE` (misalnya memindahkan pinjaman ke eksemplar yang sedang dipinjam, atau menghapus data pengembalian sehingga pinjaman aktif lagi), kolom kunci riwayat dikunci:

```sql
CREATE OR REPLACE FUNCTION fn_tolak_ubah_riwayat()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
    RAISE EXCEPTION 'Data % tidak boleh diubah atau dihapus; catat koreksi sebagai transaksi baru', TG_TABLE_NAME;
END;
$$;

CREATE TRIGGER trg_loans_tolak_ubah
BEFORE UPDATE OF member_id, copy_id, loan_date, due_date OR DELETE ON loans
FOR EACH ROW
EXECUTE FUNCTION fn_tolak_ubah_riwayat();

CREATE TRIGGER trg_returns_tolak_ubah
BEFORE UPDATE OF loan_id OR DELETE ON loan_returns
FOR EACH ROW
EXECUTE FUNCTION fn_tolak_ubah_riwayat();
```

## Query daftar peminjaman terlambat per hari ini

```sql
SELECT
    l.loan_id,
    m.nomor_induk,
    m.full_name AS nama_anggota,
    mt.label AS jenis_anggota,
    b.title AS judul_buku,
    c.copy_code AS kode_eksemplar,
    l.loan_date,
    l.due_date,
    (CURRENT_DATE - l.due_date) AS hari_terlambat,
    (CURRENT_DATE - l.due_date)::numeric * mt.daily_fine AS denda
FROM loans l
JOIN members m      ON m.member_id = l.member_id
JOIN member_types mt ON mt.type_code = m.member_type
JOIN book_copies c  ON c.copy_id = l.copy_id
JOIN books b        ON b.book_id = c.book_id
LEFT JOIN loan_returns r ON r.loan_id = l.loan_id
WHERE r.loan_id IS NULL
  AND l.due_date < CURRENT_DATE
ORDER BY hari_terlambat DESC, m.full_name;
```

Query di atas hanya menampilkan pinjaman yang **belum dikembalikan** dan sudah melewati `due_date`. `CURRENT_DATE - l.due_date` menghasilkan jumlah hari keterlambatan. Karena syaratnya `due_date < CURRENT_DATE`, nilainya selalu positif.

Jika ingin menghitung denda final untuk pinjaman yang **sudah dikembalikan terlambat**, ganti `CURRENT_DATE` dengan `r.returned_at`:

```sql
SELECT
    l.loan_id,
    m.full_name AS nama_anggota,
    b.title AS judul_buku,
    l.due_date,
    r.returned_at,
    (r.returned_at - l.due_date) AS hari_terlambat,
    (r.returned_at - l.due_date)::numeric * mt.daily_fine AS denda
FROM loans l
JOIN loan_returns r  ON r.loan_id = l.loan_id
JOIN members m       ON m.member_id = l.member_id
JOIN member_types mt ON mt.type_code = m.member_type
JOIN book_copies c   ON c.copy_id = l.copy_id
JOIN books b         ON b.book_id = c.book_id
WHERE r.returned_at > l.due_date
ORDER BY hari_terlambat DESC;
```

## Contoh perhitungan

Misalkan hari ini `2026-10-21`, seorang siswa meminjam buku pada `2026-10-01`. Masa pinjam siswa adalah 7 hari, maka:

1. `due_date = loan_date + 7 hari = 2026-10-01 + 7 = 2026-10-08`.
2. Buku belum dikembalikan.
3. Hari terlambat: `CURRENT_DATE - due_date = 2026-10-21 - 2026-10-08 = 13 hari`.
4. Denda: `13 hari × Rp1.000 = Rp13.000`.

Jadi baris query akan menampilkan `hari_terlambat = 13` dan `denda = 13000.00`.
