```html
  /* ---------- utilitas ---------- */

  function pad2(n) { return n < 10 ? '0' + n : '' + n; }

  function formatTime(d) {
    return pad2(d.getHours()) + ':' + pad2(d.getMinutes()) + ':' + pad2(d.getSeconds());
  }

  function nextVal(cur, base, amp) {
    var drift = (base - cur) * 0.06;
    var noise = (Math.random() - 0.5) * amp;
    return cur + drift + noise;
  }

  /* ---------- penyimpanan batas aman ---------- */

  function loadSettings() {
    try {
      var raw = window.localStorage.getItem(STORAGE_KEY);
      if (!raw) return;
      var obj = JSON.parse(raw);
      if (!obj || typeof obj !== 'object') return;
      var keys = ['tempMin', 'tempMax', 'humMin', 'humMax'];
      for (var i = 0; i < keys.length; i++) {
        var v = Number(obj[keys[i]]);
        if (isFinite(v)) state.limits[keys[i]] = v;
      }
      if (state.limits.tempMin >= state.limits.tempMax ||
          state.limits.humMin >= state.limits.humMax) {
        state.limits = { tempMin: 18, tempMax: 28, humMin: 40, humMax: 70 };
      }
    } catch (e) {
      /* localStorage tidak tersedia — pakai nilai bawaan */
    }
  }

  function saveSettings() {
    try {
      window.localStorage.setItem(STORAGE_KEY, JSON.stringify(state.limits));
    } catch (e) {
      /* diabaikan */
    }
  }

  function syncInputs() {
    inputEls.tempMin.value = state.limits.tempMin;
    inputEls.tempMax.value = state.limits.tempMax;
    inputEls.humMin.value = state.limits.humMin;
    inputEls.humMax.value = state.limits.humMax;
  }

  /* ---------- evaluasi status ---------- */

  function evalStatus(metric, value) {
    var L = state.limits;
    if (typeof value !== 'number' || !isFinite(value)) return 'ok';
    if (metric === 'temp') {
      if (value > L.tempMax) return 'high';
      if (value < L.tempMin) return 'low';
    } else {
      if (value > L.humMax) return 'high';
      if (value < L.humMin) return 'low';
    }
    return 'ok';
  }

  function updateStatus(room, metric, value) {
    var prev = state.status[room.id][metric];
    var next = evalStatus(metric, value);
    if (next === prev) return;
    state.status[room.id][metric] = next;

    var label = metric === 'temp' ? 'Suhu' : 'Kelembapan';
    var unit = metric === 'temp' ? ' °C' : ' %';
    var lim = metric === 'temp'
      ? (next === 'high' ? state.limits.tempMax : state.limits.tempMin)
      : (next === 'high' ? state.limits.humMax : state.limits.humMin);

    var text;
    if (next === 'ok') {
      text = room.name + ' · ' + label + ' kembali normal (' + value.toFixed(1) + unit + ')';
    } else {
      text = room.name + ' · ' + label + ' ' + value.toFixed(1) + unit +
             (next === 'high' ? ' melebihi batas atas ' : ' di bawah batas bawah ') +
             lim + unit;
    }
    addAlert(next === 'ok' ? 'info' : 'alarm', text);
  }

  function checkAll() {
    ROOMS.forEach(function (r) {
      var d = state.series[r.id];
      if (!d.temp.length || !d.hum.length) return;
      updateStatus(r, 'temp', d.temp[d.temp.length - 1]);
      updateStatus(r, 'hum', d.hum[d.hum.length - 1]);
    });
  }

  /* ---------- log peringatan ---------- */

  function addAlert(level, text) {
    state.alerts.unshift({ level: level, text: text, time: new Date() });
    if (state.alerts.length > 60) state.alerts.length = 60;
    renderLog();
  }

  function renderLog() {
    logEl.innerHTML = '';
    if (!state.alerts.length) {
      var empty = document.createElement('li');
      empty.className = 'empty';
      empty.textContent = 'Belum ada peringatan. Semua sensor dalam batas aman.';
      logEl.appendChild(empty);
      return;
    }
    var shown = state.alerts.slice(0, 20);
    var frag = document.createDocumentFragment();
    shown.forEach(function (a) {
      var li = document.createElement('li');
      li.className = 'log-item' + (a.level === 'alarm' ? ' alarm' : '');
      var t = document.createElement('span');
      t.className = 'log-time';
      t.textContent = formatTime(a.time);
      var m = document.createElement('span');
      m.className = 'log-msg';
      m.textContent = a.text;
      li.appendChild(t);
      li.appendChild(m);
      frag.appendChild(li);
    });
    logEl.appendChild(frag);
  }

  /* ---------- kartu ruangan ---------- */

  function makeRow(metric, label, unit) {
    var row = document.createElement('div');
    row.className = 'row';
    row.setAttribute('data-metric', metric);

    var l = document.createElement('span');
    l.className = 'row-label';
    l.textContent = label;

    var v = document.createElement('span');
    v.className = 'row-value';
    var b = document.createElement('b');
    b.setAttribute('data-role', metric);
    b.textContent = '--';
    var i = document.createElement('i');
    i.textContent = unit;
    v.appendChild(b);
    v.appendChild(i);

    row.appendChild(l);
    row.appendChild(v);
    return row;
  }

  function buildCards() {
    var frag = document.createDocumentFragment();
    ROOMS.forEach(function (r) {
      var card = document.createElement('article');
      card.className = 'card';
      card.id = 'card-' + r.id;

      var top = document.createElement('div');
      top.className = 'card-top';
      var h3 = document.createElement('h3');
      h3.textContent = r.name;
      var pill = document.createElement('span');
      pill.className = 'pill';
      pill.setAttribute('data-role', 'pill');
      pill.textContent = 'Aman';
      top.appendChild(h3);
      top.appendChild(pill);

      var rows = document.createElement('div');
      rows.appendChild(makeRow('temp', 'Suhu', '°C'));
      rows.appendChild(makeRow('hum', 'Kelembapan', '%'));

      card.appendChild(top);
      card.appendChild(rows);
      frag.appendChild(card);
      cards[r.id] = card;
    });
    cardsEl.appendChild(frag);
  }

  function buildLegend() {
    ROOMS.forEach(function (r) {
      var s = document.createElement('span');
      var d = document.createElement('span');
      d.className = 'dot';
      d.style.background = COLORS[r.id];
      s.appendChild(d);
      s.appendChild(document.createTextNode(r.name));
      legendEl.appendChild(s);
    });
  }

  function render() {
    ROOMS.forEach(function (r) {
      var card = cards[r.id];
      if (!card) return;
      var d = state.series[r.id];
      var st = state.status[r.id];

      var tEl = card.querySelector('[data-role="temp"]');
      var hEl = card.querySelector('[data-role="hum"]');
      tEl.textContent = d.temp.length ? d.temp[d.temp.length - 1].toFixed(1) : '--';
      hEl.textContent = d.hum.length ? d.hum[d.hum.length - 1].toFixed(1) : '--';

      var tRow = card.querySelector('[data-metric="temp"]');
      var hRow = card.querySelector('[data-metric="hum"]');
      tRow.classList.toggle('bad', st.temp !== 'ok');
      hRow.classList.toggle('bad', st.hum !== 'ok');

      var count = 0;
      if (st.temp !== 'ok') count++;
      if (st.hum !== 'ok') count++;

      card.classList.toggle('alarm', count > 0);
      var pill = card.querySelector('[data-role="pill"]');
      pill.textContent = count === 0 ? 'Aman' : (count === 1 ? 'Waspada' : 'Bahaya');
      pill.className = 'pill' + (count === 0 ? '' : (count === 1 ? ' warn' : ' danger'));
    });
  }

  /* ---------- grafik canvas ---------- */

  function getRange(metric) {
    var lo = Infinity, hi = -Infinity;
    ROOMS.forEach(function (r) {
      var arr = state.series[r.id][metric];
      for (var i = 0; i < arr.length; i++) {
        if (arr[i] < lo) lo = arr[i];
        if (arr[i] > hi) hi = arr[i];
      }
    });
    if (!isFinite(lo) || !isFinite(hi)) { lo = 0; hi = 1; }

    if (metric === 'temp') {
      lo = Math.min(lo, state.limits.tempMin);
      hi = Math.max(hi, state.limits.tempMax);
    } else {
      lo = Math.min(lo, state.limits.humMin);
      hi = Math.max(hi, state.limits.humMax);
    }
    if (hi - lo < 4) {
      var mid = (hi + lo) / 2;
      lo = mid - 2;
      hi = mid + 2;
    }
    var pad = (hi - lo) * 0.12;
    return [lo - pad, hi + pad];
  }

  function fmtTick(v, span) {
    return span < 8 ? v.toFixed(1) : String(Math.round(v));
  }

  function drawChart() {
    var rect = canvas.getBoundingClientRect();
    var w = Math.max(1, Math.round(rect.width));
    var h = Math.max(1, Math.round(rect.height));
    var dpr = Math.min(window.devicePixelRatio || 1, 2);
    var pw = Math.round(w * dpr);
    var ph = Math.round(h * dpr);

    if (canvas.width !== pw || canvas.height !== ph) {
      canvas.width = pw;
      canvas.height = ph;
    }

    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    ctx.clearRect(0, 0, w, h);

    var padL = 44, padR = 12, padT = 14, padB = 22;
    var plotW = Math.max(10, w - padL - padR);
    var plotH = Math.max(10, h - padT - padB);

    var range = getRange(state.metric);
    var lo = range[0];
    var hi = range[1];
    var span = (hi - lo) || 1;

    ctx.font = '10px system-ui, -apple-system, "Segoe UI", Roboto, Arial, sans-serif';
    ctx.textBaseline = 'middle';
    ctx.textAlign = 'right';

    var steps = 4;
    for (var i = 0; i <= steps; i++) {
      var val = lo + span * i / steps;
      var y = padT + plotH * (1 - i / steps);
      ctx.strokeStyle = 'rgba(120,160,220,0.14)';
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(padL, Math.round(y) + 0.5);
      ctx.lineTo(padL + plotW, Math.round(y) + 0.5);
      ctx.stroke();

      ctx.fillStyle = '#7f93b3';
      ctx.fillText(fmtTick(val, span), padL - 8, y);
    }

    /* garis batas aman */
    var limA, limB;
    if (state.metric === 'temp') {
      limA = state.limits.tempMax;
      limB = state.limits.tempMin;
    } else {
      limA = state.limits.humMax;
      limB = state.limits.humMin;
    }
    ctx.setLineDash([4, 4]);
    ctx.lineWidth = 1;
    ctx.strokeStyle = 'rgba(248,113,113,0.55)';
    [limA, limB].forEach(function (v) {
      if (v < lo || v > hi) return;
      var y = padT + plotH * (1 - (v - lo) / span);
      ctx.beginPath();
      ctx.moveTo(padL, Math.round(y) + 0.5);
      ctx.lineTo(padL + plotW, Math.round(y) + 0.5);
      ctx.stroke();
    });
    ctx.setLineDash([]);

    /* deret data */
    ctx.save();
    ctx.beginPath();
    ctx.rect(padL, padT - 8, plotW, plotH + 16);
    ctx.clip();

    ROOMS.forEach(function (r) {
      var arr = state.series[r.id][state.metric];
      var n = arr.length;
      if (!n) return;
      var off = MAX_POINTS - n;

      ctx.beginPath();
      for (var k = 0; k < n; k++) {
        var x = padL + plotW * ((off + k) / (MAX_POINTS - 1));
        var yv = padT + plotH * (1 - (arr[k] - lo) / span);
        if (k === 0) ctx.moveTo(x, yv);
        else ctx.lineTo(x, yv);
      }
      ctx.strokeStyle = COLORS[r.id];
      ctx.lineWidth = 2;
      ctx.lineJoin = 'round';
      ctx.lineCap = 'round';
      ctx.stroke();

      var lx = padL + plotW * ((off + n - 1) / (MAX_POINTS - 1));
      var ly = padT + plotH * (1 - (arr[n - 1] - lo) / span);
      ctx.beginPath();
      ctx.arc(lx, ly, 3, 0, Math.PI * 2);
      ctx.fillStyle = COLORS[r.id];
      ctx.fill();
    });
    ctx.restore();

    /* label sumbu waktu */
    ctx.textAlign = 'center';
    ctx.textBaseline = 'top';
    ctx.fillStyle = '#7f93b3';
    ctx.fillText('−60 s', padL, padT + plotH + 6);
    ctx.fillText('−30 s', padL + plotW * 0.5, padT + plotH + 6);
    ctx.fillText('kini', padL + plotW, padT + plotH + 6);
  }

  /* ---------- simulasi data ---------- */

  function prefill() {
    for (var i = 0; i < 24; i++) {
      ROOMS.forEach(function (r) {
        var d = state.series[r.id];
        var lt = d.temp.length ? d.temp[d.temp.length - 1] : r.baseTemp + (Math.random() - 0.5) * 2;
        var lh = d.hum.length ? d.hum[d.hum.length - 1] : r.baseHum + (Math.random() - 0.5) * 4;
        d.temp.push(nextVal(lt, r.baseTemp, 0.7));
        d.hum.push(nextVal(lh, r.baseHum, 2.4));
      });
    }
  }

  function tick() {
    if (state.paused) return;

    ROOMS.forEach(function (r) {
      var d = state.series[r.id];
      var lt = d.temp.length ? d.temp[d.temp.length - 1] : r.baseTemp;
      var lh = d.hum.length ? d.hum[d.hum.length - 1] : r.baseHum;

      d.temp.push(nextVal(lt, r.baseTemp, 0.7));
      d.hum.push(nextVal(lh, r.baseHum, 2.4));

      if (d.temp.length > MAX_POINTS) d.temp.shift();
      if (d.hum.length > MAX_POINTS) d.hum.shift();
    });

    checkAll();
    render();
    drawChart();
  }

  /* ---------- batas aman: input ---------- */

  function applyLimits() {
    var vals = {
      tempMin: parseFloat(inputEls.tempMin.value),
      tempMax: parseFloat(inputEls.tempMax.value),
      humMin: parseFloat(inputEls.humMin.value),
      humMax: parseFloat(inputEls.humMax.value)
    };

    var valid = true;
    for (var k in vals) {
      if (!isFinite(vals[k])) valid = false;
    }
    if (valid && vals.tempMin >= vals.tempMax) valid = false;
    if (valid && vals.humMin >= vals.humMax) valid = false;

    Object.keys(inputEls).forEach(function (key) {
      inputEls[key].classList.toggle('invalid', !valid);
    });

    if (!valid) {
      limitHint.textContent = 'Nilai tidak valid: isi angka dan pastikan batas minimum lebih kecil dari maksimum.';
      return;
    }

    limitHint.textContent = '';
    state.limits = vals;
    saveSettings();
    checkAll();
    render();
    drawChart();
  }

  Object.keys(inputEls).forEach(function (key) {
    inputEls[key].addEventListener('change', applyLimits);
    inputEls[key].addEventListener('blur', applyLimits);
  });

  /* ---------- interaksi lain ---------- */

  toggleBtn.addEventListener('click', function () {
    state.paused = !state.paused;
    toggleBtn.textContent = state.paused ? 'Lanjutkan' : 'Jeda';
    statusBadge.textContent = state.paused ? 'Dijeda' : 'Berjalan';
    statusBadge.classList.toggle('paused', state.paused);
  });

  segmented.addEventListener('click', function (ev) {
    var btn = ev.target.closest ? ev.target.closest('button[data-metric]') : null;
    if (!btn) return;
    var m = btn.getAttribute('data-metric');
    if (m !== 'temp' && m !== 'hum') return;
    state.metric = m;
    var all = segmented.querySelectorAll('button');
    for (var i = 0; i < all.length; i++) {
      all[i].classList.toggle('active', all[i] === btn);
    }
    drawChart();
  });

  window.addEventListener('resize', drawChart);
  window.addEventListener('orientationchange', drawChart);

  if (typeof window.ResizeObserver === 'function') {
    try {
      var ro = new window.ResizeObserver(function () { drawChart(); });
      ro.observe(canvasWrap);
    } catch (e) {
      /* diabaikan */
    }
  }

  /* ---------- inisialisasi ---------- */

  loadSettings();
  syncInputs();
  buildCards();
  buildLegend();
  prefill();
  renderLog();
  checkAll();
  render();
  drawChart();
  window.setInterval(tick, 1000);
})();
</script>
</body>
</html>
```

**Cek kebutuhan**
- **Dashboard pemantauan sensor gudang dalam satu file HTML** → seluruh dokumen `<!DOCTYPE html>` dengan CSS di `<style>` dan JavaScript di `<script>` dalam satu file.
- **Data simulasi suhu dan kelembapan dari 3 ruangan** → array `ROOMS` (Ruang A/B/C) dengan `baseTemp` dan `baseHum`, serta fungsi `nextVal()` dan `prefill()`.
- **Diperbarui setiap detik** → `window.setInterval(tick, 1000)` pada bagian inisialisasi.
- **Grafik garis real-time 60 detik terakhir dengan canvas** → elemen `<canvas id="chart">` dan fungsi `drawChart()` dengan `MAX_POINTS = 60`.
- **Batas aman yang bisa diatur** → panel "Batas Aman" (`#tempMin`, `#tempMax`, `#humMin`, `#humMax`) dan fungsi `applyLimits()`.
- **Kartu berubah merah jika melewati batas** → kelas `.card.alarm` dan `.row.bad` yang diatur di `render()` lewat `evalStatus()`.
- **Log peringatan** → `addAlert()`, `renderLog()`, dan daftar `<ul class="log" id="log">`.
- **Tombol jeda/lanjut** → `#toggleBtn` dengan penanganan `state.paused` pada `tick()`.
- **Responsif sampai lebar 375 px** → `grid-template-columns: repeat(auto-fit, minmax(215px, 1fr))`, `@media (max-width: 480px)`, dan `canvas` berukuran persentase.
- **Tanpa library, font, gambar, atau file dari luar** → tidak ada `<link>`, `<script src>`, `@import`, atau `url()` eksternal; hanya `font-family` sistem.
- **Tanpa error JavaScript** → pembungkus IIFE dengan `'use strict'`, pengecekan `isFinite`/`null`, dan penjagaan elemen sebelum dipakai.
- **Akses `localStorage` dibungkus `try/catch`** → fungsi `loadSettings()` dan `saveSettings()`.

**Catatan teknis**
- Batas aman disimpan di `localStorage` dengan kunci `gudang.sensor.limits.v1`; bila data rusak atau tidak tersedia, nilai bawaan (18–28 °C, 40–70 %) dipakai otomatis.
- Nilai input yang tidak valid (kosong atau minimum ≥ maksimum) ditandai merah dan tidak diterapkan, sehingga grafik dan status tetap memakai batas terakhir yang sah.
- Grafik digambar ulang dengan penskalaan `devicePixelRatio` (dibatasi 2×) agar tajam di layar beresolusi tinggi, dan `ResizeObserver` menjaga kanvas tetap pas saat ukuran jendela berubah.
- Saat dijeda, timer tetap berjalan tetapi `tick()` langsung keluar, sehingga data, kartu, dan log membeku; penggantian metrik grafik tetap berfungsi.
