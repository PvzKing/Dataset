Token diproses dalam urutan yang meminimalkan jumlah data tak tepercaya yang disentuh sebelum keaslian terbukti. Pertama, bentuk token diperiksa secara ketat: tepat tiga segmen, alfabet base64url saja, dan encoding kanonik. Header lalu di-parse untuk memastikan `alg` persis `"HS256"`. Setelah itu tanda tangan HMAC dihitung ulang atas byte ASCII `header.payload` dan dibandingkan dengan `hmac.compare_digest`. Payload baru di-parse setelah tanda tangan sah, kemudian klaim `exp`, `nbf`, dan `aud` divalidasi. Parser JSON dibuat ketat: kunci duplikat, `NaN`/`Infinity`, dan bilangan yang meluap menjadi tak hingga ditolak. Semua masalah pada token menghasilkan `ValueError`. Argumen pemanggil yang salah tipe menghasilkan `TypeError`, karena itu kesalahan program, bukan token jahat.

```python
import base64
import binascii
import hashlib
import hmac
import json
import math
import re

_POLA_SEGMEN = re.compile(r"[A-Za-z0-9_-]+")


def _adalah_bilangan(nilai) -> bool:
    """int atau float berhingga, tetapi bukan bool."""
    if isinstance(nilai, bool):
        return False
    if isinstance(nilai, int):
        return True
    return isinstance(nilai, float) and math.isfinite(nilai)


def _dekode_segmen(segmen: str, nama: str) -> bytes:
    """Dekode base64url tanpa padding secara ketat dan kanonik (RFC 7515 §2)."""
    if not _POLA_SEGMEN.fullmatch(segmen):
        raise ValueError(f"{nama}: karakter di luar alfabet base64url atau segmen kosong")
    if len(segmen) % 4 == 1:
        raise ValueError(f"{nama}: panjang base64url tidak valid")
    try:
        data = base64.b64decode(segmen + "=" * (-len(segmen) % 4),
                                altchars=b"-_", validate=True)
    except (binascii.Error, ValueError):
        raise ValueError(f"{nama}: base64url tidak valid") from None
    # Tolak encoding non-kanonik (bit sisa tidak nol), agar satu nilai
    # hanya punya satu representasi teks.
    if base64.urlsafe_b64encode(data).rstrip(b"=").decode("ascii") != segmen:
        raise ValueError(f"{nama}: encoding base64url tidak kanonik")
    return data


def _tolak_duplikat(pasangan):
    hasil = {}
    for k, v in pasangan:
        if k in hasil:
            raise ValueError(f"kunci JSON duplikat: {k!r}")
        hasil[k] = v
    return hasil


def _tolak_konstanta(nama):
    raise ValueError(f"konstanta JSON non-standar: {nama}")


def _muat_objek_json(data: bytes, nama: str) -> dict:
    try:
        teks = data.decode("utf-8")
        obj = json.loads(teks, object_pairs_hook=_tolak_duplikat,
                         parse_constant=_tolak_konstanta)
    except (UnicodeDecodeError, ValueError, RecursionError) as e:
        raise ValueError(f"{nama}: JSON tidak valid ({e})") from None
    if not isinstance(obj, dict):
        raise ValueError(f"{nama}: harus berupa objek JSON")
    return obj


def verifikasi_jwt(token: str, kunci: bytes, sekarang: int,
                   audiens: str | None = None, toleransi: int = 0) -> dict:
    """Verifikasi JWT HS256 dan kembalikan payload-nya.

    Melempar ValueError jika token tidak boleh dipercaya.
    Melempar TypeError/ValueError jika argumen pemanggil tidak valid.
    """
    # --- Validasi argumen pemanggil ---
    if not isinstance(kunci, (bytes, bytearray)):
        raise TypeError("kunci harus bertipe bytes")
    if len(kunci) == 0:
        raise ValueError("kunci kosong tidak aman")
    if isinstance(sekarang, bool) or not isinstance(sekarang, int):
        raise TypeError("sekarang harus int (detik epoch)")
    if isinstance(toleransi, bool) or not isinstance(toleransi, int):
        raise TypeError("toleransi harus int")
    if toleransi < 0:
        raise ValueError("toleransi tidak boleh negatif")
    if audiens is not None and not isinstance(audiens, str):
        raise TypeError("audiens harus str atau None")

    # --- Struktur token ---
    if not isinstance(token, str):
        raise ValueError("token harus berupa string")
    bagian = token.split(".")
    if len(bagian) != 3:
        raise ValueError("token harus terdiri atas tepat tiga segmen")
    seg_header, seg_payload, seg_ttd = bagian

    header_bytes = _dekode_segmen(seg_header, "header")
    payload_bytes = _dekode_segmen(seg_payload, "payload")
    tanda_tangan = _dekode_segmen(seg_ttd, "tanda tangan")

    # --- Header: algoritme dikunci di sisi server ---
    header = _muat_objek_json(header_bytes, "header")
    if header.get("alg") != "HS256":
        raise ValueError("alg harus persis 'HS256'")
    if "crit" in header:
        # RFC 7515 §4.1.11: ekstensi kritis yang tidak dipahami wajib ditolak.
        raise ValueError("header 'crit' tidak didukung")

    # --- Tanda tangan (waktu-konstan) ---
    masukan = f"{seg_header}.{seg_payload}".encode("ascii")
    harapan = hmac.new(bytes(kunci), masukan, hashlib.sha256).digest()
    if not hmac.compare_digest(harapan, tanda_tangan):
        raise ValueError("tanda tangan tidak valid")

    # --- Payload (baru disentuh setelah keaslian terbukti) ---
    payload = _muat_objek_json(payload_bytes, "payload")

    if "exp" in payload:
        exp = payload["exp"]
        if not _adalah_bilangan(exp):
            raise ValueError("klaim exp harus bilangan")
        if sekarang >= exp + toleransi:
            raise ValueError("token sudah kedaluwarsa")

    if "nbf" in payload:
        nbf = payload["nbf"]
        if not _adalah_bilangan(nbf):
            raise ValueError("klaim nbf harus bilangan")
        if sekarang < nbf - toleransi:
            raise ValueError("token belum berlaku")

    if "aud" in payload:
        aud = payload["aud"]
        if isinstance(aud, str):
            daftar_aud = [aud]
        elif isinstance(aud, list) and all(isinstance(a, str) for a in aud):
            daftar_aud = aud
        else:
            raise ValueError("klaim aud harus string atau list string")
        if audiens is None:
            raise ValueError("token memiliki aud, tetapi audiens tidak ditentukan")
        if audiens not in daftar_aud:
            raise ValueError("audiens tidak cocok")
    elif audiens is not None:
        raise ValueError("token tidak memiliki klaim aud")

    return payload

```

Contoh pemakaian

```
import base64, hashlib, hmac, json

def _b64(b): return base64.urlsafe_b64encode(b).rstrip(b"=").decode()

def buat_token(payload, kunci, header=None):
    header = header or {"alg": "HS256", "typ": "JWT"}
    h = _b64(json.dumps(header, separators=(",", ":")).encode())
    p = _b64(json.dumps(payload, separators=(",", ":")).encode())
    s = _b64(hmac.new(kunci, f"{h}.{p}".encode(), hashlib.sha256).digest())
    return f"{h}.{p}.{s}"

kunci = b"k" * 32
t = buat_token({"sub": "budi", "exp": 1_800_000_100, "aud": "api"}, kunci)

verifikasi_jwt(t, kunci, sekarang=1_800_000_000, audiens="api")
# {'sub': 'budi', 'exp': 1800000100, 'aud': 'api'}

verifikasi_jwt(t, kunci, sekarang=1_800_000_000)                    # ValueError: aud tanpa audiens
verifikasi_jwt(t, kunci, sekarang=1_800_000_100, audiens="api")     # ValueError: kedaluwarsa
verifikasi_jwt(t, kunci, sekarang=1_800_000_100, audiens="api", toleransi=30)  # lolos
verifikasi_jwt(t + "=", kunci, sekarang=1_800_000_000, audiens="api")  # ValueError: alfabet

```

Aturan dan serangan yang dicegahnya
Tepat tiga segmen dan alfabet base64url yang ketat. Aturan ini menolak bentuk JWE (lima segmen) dan segmen kosong. Dekoder bawaan yang longgar diam-diam membuang karakter asing seperti `=`, `+`, `/`, spasi, dan newline. Dengan menolaknya, celah penyelundupan dan perbedaan tafsir antara komponen sistem tertutup. Pemeriksaan kanonik menutup signature malleability. Tanpa pemeriksaan ini, bit sisa pada karakter terakhir bisa diubah sehingga muncul beberapa string token berbeda yang sama-sama sah. Akibatnya, daftar cabut atau cache yang berbasis string token bisa diakali.
`alg` harus persis `"HS256"`. Algoritme dikunci di sisi server, bukan dipercaya dari header. Ini mencegah serangan `alg: none`, yaitu token tanpa tanda tangan yang diterima begitu saja. Ini juga mencegah algorithm confusion: penyerang memilih algoritme lain, misalnya `HS512` atau varian huruf kecil yang ditangani pustaka lain secara berbeda, agar diverifikasi dengan jalur kode yang tidak dimaksud. Varian paling berbahaya adalah mengganti RS256 ke HS256 dengan kunci publik sebagai rahasia HMAC. Penolakan `crit` mengikuti RFC 7515: ekstensi wajib yang tidak dipahami tidak boleh diabaikan diam-diam.
Tanda tangan atas byte ASCII `header.payload` dan `compare_digest`. Menghitung atas segmen asli, bukan hasil serialisasi ulang, memastikan byte yang ditandatangani persis byte yang diterima. Perbandingan waktu-konstan mencegah timing attack. Dengan `==` biasa, penyerang bisa menebak tanda tangan byte demi byte dengan mengukur lama respons.
Parser JSON yang ketat, hanya objek. Kunci duplikat seperti `{"sub":"budi","sub":"admin"}` ditolak. Kunci duplikat memungkinkan serangan perbedaan parser: satu komponen membaca nilai pertama, komponen lain membaca nilai terakhir. `NaN`/`Infinity`, serta bilangan seperti `1e400` yang meluap menjadi `inf`, juga ditolak. Tanpa itu, `exp: 1e400` membuat token tak pernah kedaluwarsa, dan `NaN` membuat semua perbandingan bernilai salah sehingga pemeriksaan waktu terlewati. Payload baru di-parse setelah tanda tangan sah, sehingga data tak terautentikasi tidak sempat diproses lebih jauh.
`exp`/`nbf` wajib bilangan non-bool. Di Python, `True` adalah `1`. Tanpa pengecekan ini, `"exp": true` lolos sebagai angka. Nilai string seperti `"exp": "9999999999"` bisa memicu galat tipe yang tertangkap salah atau perbandingan yang tidak dimaksud. Aturan `sekarang >= exp + toleransi` mencegah replay token lama yang bocor. Aturan `nbf` mencegah token yang diterbitkan untuk masa depan dipakai lebih awal. Toleransi dibatasi int non-negatif agar tidak bisa dipakai memperlebar jendela secara tak terduga.
Aturan `aud` dua arah. Jika `audiens` diberikan, `aud` wajib ada dan memuatnya. Ini mencegah token substitution: token sah untuk layanan A yang berbagi kunci tidak bisa dipakai di layanan B. Sebaliknya, layanan yang tidak menetapkan audiens wajib menolak token yang punya `aud` (RFC 7519 §4.1.3). Dengan begitu, token yang jelas ditujukan ke pihak lain tidak bisa "diterima di mana saja". Tipe `aud` selain string atau list string, misalnya objek atau list campuran, ditolak agar pengecekan keanggotaan tidak bisa dimanipulasi.
Kunci kosong ditolak. HMAC dengan kunci kosong dapat dipalsukan siapa pun. Fungsi ini sengaja tidak memaksakan panjang minimum. Namun, RFC 7518 §3.2 mewajibkan kunci HS256 minimal 32 byte, jadi pastikan kunci produksi memenuhinya.
Pesan galat sengaja dibuat rinci untuk log internal. Ke klien, sebaiknya cukup kembalikan respons generik seperti "token tidak valid" agar penyerang tidak mendapat petunjuk pemeriksaan mana yang gagal.
