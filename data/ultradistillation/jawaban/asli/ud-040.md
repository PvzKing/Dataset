```html
<script>
(function () {
  'use strict';

  /* ==================== Elemen ==================== */
  var stage      = document.getElementById('stage');
  var world      = document.getElementById('world');
  var edges      = document.getElementById('edges');
  var nodesLayer = document.getElementById('nodes');
  var colorsEl   = document.getElementById('colors');
  var btnAdd     = document.getElementById('btnAdd');
  var btnDel     = document.getElementById('btnDel');
  var btnZoomIn  = document.getElementById('btnZoomIn');
  var btnZoomOut = document.getElementById('btnZoomOut');
  var btnFit     = document.getElementById('btnFit');
  var btnPng     = document.getElementById('btnPng');
  var btnTxt     = document.getElementById('btnTxt');

  /* ==================== Konstanta ==================== */
  var STORAGE_KEY = 'petaPikiran.v1';
  var PALETTE = ['#6366f1', '#ec4899', '#f59e0b', '#10b981', '#06b6d4', '#ef4444'];
  var GAP_Y = 86;
  var GAP_X = 215;
  var MIN_SCALE = 0.25;
  var MAX_SCALE = 3;

  /* ==================== State ==================== */
  var nodes = [];
  var pan = { x: 0, y: 0 };
  var scale = 1;
  var uid = 1;
  var selectedId = null;
  var editingId = null;
  var saveTimer = null;

  /* ==================== Util ==================== */
  function byId(id) {
    for (var i = 0; i < nodes.length; i++) { if (nodes[i].id === id) { return nodes[i]; } }
    return null;
  }
  function rootNode() {
    for (var i = 0; i < nodes.length; i++) { if (nodes[i].parentId === null) { return nodes[i]; } }
    return null;
  }
  function childrenOf(id) {
    return nodes.filter(function (n) { return n.parentId === id; });
  }
  function clamp(v, a, b) { return v < a ? a : (v > b ? b : v); }
  function nodeEl(id) {
    return nodesLayer.querySelector('.node[data-id="' + id + '"]');
  }

  /* ==================== Penyimpanan ==================== */
  function save() {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify({
        nodes: nodes, pan: pan, scale: scale, uid: uid
      }));
    } catch (e) { /* localStorage tidak tersedia */ }
  }
  function saveSoon() {
    if (saveTimer) { clearTimeout(saveTimer); }
    saveTimer = setTimeout(save, 400);
  }
  function load() {
    try {
      var raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) { return null; }
      var data = JSON.parse(raw);
      if (!data || !Array.isArray(data.nodes)) { return null; }
      return data;
    } catch (e) { return null; }
  }

  /* ==================== Transformasi kanvas ==================== */
  function applyTransform() {
    world.style.transform = 'translate(' + pan.x + 'px,' + pan.y + 'px) scale(' + scale + ')';
    var g = 24 * scale;
    stage.style.backgroundSize = g + 'px ' + g + 'px';
    stage.style.backgroundPosition = pan.x + 'px ' + pan.y + 'px';
  }

  function zoomAt(mx, my, factor) {
    var next = clamp(scale * factor, MIN_SCALE, MAX_SCALE);
    if (next === scale) { return; }
    var k = next / scale;
    pan.x = mx - (mx - pan.x) * k;
    pan.y = my - (my - pan.y) * k;
    scale = next;
    applyTransform();
  }

  function resetView() {
    var r = stage.getBoundingClientRect();
    var root = rootNode();
    var rx = root ? root.x : 0;
    var ry = root ? root.y : 0;
    scale = 1;
    pan = { x: r.width / 2 - rx, y: r.height / 2 - ry };
    applyTransform();
    saveSoon();
  }

  /* ==================== Tata letak otomatis ==================== */
  function layoutChildren(pid) {
    var p = byId(pid);
    if (!p) { return; }
    var kids = childrenOf(pid);
    if (!kids.length) { return; }

    var root = rootNode();
    var rootX = root ? root.x : 0;
    var isRootParent = (p.parentId === null);

    var sides = kids.map(function (k, i) {
      if (isRootParent) { return (i % 2 === 0) ? 1 : -1; }
      return (p.x - rootX) >= 0 ? 1 : -1;
    });

    var totalR = 0, totalL = 0;
    for (var i = 0; i < sides.length; i++) { if (sides[i] === 1) { totalR++; } else { totalL++; } }

    var ri = 0, li = 0;
    for (var j = 0; j < kids.length; j++) {
      var dir = sides[j];
      var idx, total;
      if (dir === 1) { idx = ri++; total = totalR; } else { idx = li++; total = totalL; }
      kids[j].x = p.x + dir * GAP_X;
      kids[j].y = p.y + (idx - (total - 1) / 2) * GAP_Y;
    }
  }

  /* ==================== Render ==================== */
  function styleNode(el, n) {
    if (n.parentId === null) {
      el.style.background = n.color;
      el.style.borderColor = n.color;
      el.style.color = '#ffffff';
    } else {
      el.style.background = '#ffffff';
      el.style.borderColor = n.color;
      el.style.color = '#0f172a';
    }
  }

  function placeNode(el, n) {
    el.style.left = n.x + 'px';
    el.style.top = n.y + 'px';
  }

  function renderEdges() {
    var out = '';
    for (var i = 0; i < nodes.length; i++) {
      var n = nodes[i];
      if (n.parentId === null) { continue; }
      var p = byId(n.parentId);
      if (!p) { continue; }
      var sign = (n.x >= p.x) ? 1 : -1;
      var dx = Math.max(50, Math.abs(n.x - p.x) * 0.45) * sign;
      out += '<path d="M ' + p.x + ' ' + p.y +
             ' C ' + (p.x + dx) + ' ' + p.y + ', ' + (n.x - dx) + ' ' + n.y +
             ', ' + n.x + ' ' + n.y + '" fill="none" stroke="' + n.color +
             '" stroke-width="2.5" stroke-linecap="round" opacity="0.85"/>';
    }
    edges.innerHTML = out;
  }

  function renderAll() {
    nodesLayer.innerHTML = '';
    for (var i = 0; i < nodes.length; i++) {
      var n = nodes[i];
      var el = document.createElement('div');
      el.className = 'node' + (n.parentId === null ? ' root' : '');
      el.setAttribute('data-id', String(n.id));
      el.textContent = n.text;
      styleNode(el, n);
      placeNode(el, n);
      nodesLayer.appendChild(el);
    }
    renderEdges();
    select(selectedId);
  }

  function select(id) {
    if (id !== null && !byId(id)) { id = null; }
    selectedId = id;
    var els = nodesLayer.querySelectorAll('.node');
    for (var i = 0; i < els.length; i++) {
      var eid = Number(els[i].getAttribute('data-id'));
      els[i].classList.toggle('selected', id !== null && eid === id);
    }
    var n = id !== null ? byId(id) : null;
    btnDel.disabled = !n || n.parentId === null;
    var sw = colorsEl.querySelectorAll('.swatch');
    for (var j = 0; j < sw.length; j++) {
      sw[j].classList.toggle('active', !!n && sw[j].getAttribute('data-color') === n.color);
    }
  }

  /* ==================== Aksi simpul ==================== */
  function addChild() {
    var parent = (selectedId !== null ? byId(selectedId) : null) || rootNode();
    if (!parent) { return; }
    var isRootParent = (parent.parentId === null);
    var color = parent.color;
    if (isRootParent) {
      color = PALETTE[childrenOf(parent.id).length % PALETTE.length];
    }
    var child = {
      id: uid++,
      parentId: parent.id,
      text: 'Cabang baru',
      x: parent.x,
      y: parent.y,
      color: color
    };
    nodes.push(child);
    layoutChildren(parent.id);
    renderAll();
    select(child.id);
    save();
  }

  function deleteSelected() {
    if (selectedId === null) { return; }
    var target = byId(selectedId);
    if (!target || target.parentId === null) { return; }
    var pid = target.parentId;
    var doomed = {};
    (function collect(id) {
      doomed[id] = true;
      var kids = childrenOf(id);
      for (var i = 0; i < kids.length; i++) { collect(kids[i].id); }
    })(target.id);
    nodes = nodes.filter(function (n) { return !doomed[n.id]; });
    layoutChildren(pid);
    select(null);
    renderAll();
    save();
  }

  function applyColor(color) {
    if (selectedId === null) { return; }
    var start = byId(selectedId);
    if (!start) { return; }
    (function paint(id) {
      var n = byId(id);
      if (!n) { return; }
      n.color = color;
      var kids = childrenOf(id);
      for (var i = 0; i < kids.length; i++) { paint(kids[i].id); }
    })(start.id);
    renderAll();
    save();
  }

  /* ==================== Edit teks ==================== */
  function startEdit(id) {
    if (editingId !== null) { commitEdit(); }
    var el = nodeEl(id);
    var n = byId(id);
    if (!el || !n) { return; }
    editingId = id;
    el.setAttribute('data-editing', '1');
    el.setAttribute('contenteditable', 'true');
    el.textContent = n.text;
    el.focus();
    try {
      var range = document.createRange();
      range.selectNodeContents(el);
      var sel = window.getSelection();
      sel.removeAllRanges();
      sel.addRange(range);
    } catch (e) { /* abaikan */ }
  }

  function commitEdit() {
    if (editingId === null) { return; }
    var id = editingId;
    editingId = null;
    var el = nodeEl(id);
    var n = byId(id);
    if (el) {
      el.removeAttribute('contenteditable');
      el.removeAttribute('data-editing');
    }
    if (el && n) {
      var t = (el.textContent || '').replace(/\s+/g, ' ').trim();
      if (!t) { t = 'Tanpa judul'; }
      n.text = t;
      el.textContent = t;
    }
    save();
  }

  /* ==================== Interaksi pointer ==================== */
  function startDrag(e, el) {
    var id = Number(el.getAttribute('data-id'));
    var n = byId(id);
    if (!n) { return; }
    select(id);

    var startX = e.clientX, startY = e.clientY;
    var ox = n.x, oy = n.y;
    var moved = false;

    function move(ev) {
      var dx = (ev.clientX - startX) / scale;
      var dy = (ev.clientY - startY) / scale;
      if (Math.abs(dx) > 0.5 || Math.abs(dy) > 0.5) { moved = true; }
      n.x = ox + dx;
      n.y = oy + dy;
      placeNode(el, n);
      renderEdges();
    }
    function up() {
      try { el.releasePointerCapture(e.pointerId); } catch (err) { /* abaikan */ }
      el.removeEventListener('pointermove', move);
      el.removeEventListener('pointerup', up);
      el.removeEventListener('pointercancel', up);
      if (moved) { save(); }
    }
    try { el.setPointerCapture(e.pointerId); } catch (err) { /* abaikan */ }
    el.addEventListener('pointermove', move);
    el.addEventListener('pointerup', up);
    el.addEventListener('pointercancel', up);
  }

  function startPan(e) {
    var sx = e.clientX, sy = e.clientY;
    var ox = pan.x, oy = pan.y;
    stage.classList.add('panning');

    function move(ev) {
      pan.x = ox + (ev.clientX - sx);
      pan.y = oy + (ev.clientY - sy);
      applyTransform();
    }
    function up() {
      stage.classList.remove('panning');
      try { stage.releasePointerCapture(e.pointerId); } catch (err) { /* abaikan */ }
      stage.removeEventListener('pointermove', move);
      stage.removeEventListener('pointerup', up);
      stage.removeEventListener('pointercancel', up);
      save();
    }
    try { stage.setPointerCapture(e.pointerId); } catch (err) { /* abaikan */ }
    stage.addEventListener('pointermove', move);
    stage.addEventListener('pointerup', up);
    stage.addEventListener('pointercancel', up);
  }

  stage.addEventListener('pointerdown', function (e) {
    var el = e.target && e.target.closest ? e.target.closest('.node') : null;
    if (el) {
      if (el.getAttribute('data-editing') === '1') { return; }
      startDrag(e, el);
    } else {
      if (editingId !== null) { commitEdit(); }
      select(null);
      startPan(e);
    }
  });

  nodesLayer.addEventListener('dblclick', function (e) {
    var el = e.target && e.target.closest ? e.target.closest('.node') : null;
    if (!el) { return; }
    startEdit(Number(el.getAttribute('data-id')));
  });

  nodesLayer.addEventListener('keydown', function (e) {
    var el = e.target;
    if (!el || el.getAttribute('data-editing') !== '1') { return; }
    if (e.key === 'Enter') {
      e.preventDefault();
      commitEdit();
    } else if (e.key === 'Escape') {
      e.preventDefault();
      var n = byId(Number(el.getAttribute('data-id')));
      if (n) { el.textContent = n.text; }
      commitEdit();
    }
  });

  nodesLayer.addEventListener('focusout', function (e) {
    var el = e.target;
    if (el && el.classList && el.classList.contains('node') && el.getAttribute('data-editing') === '1') {
      commitEdit();
    }
  });

  stage.addEventListener('wheel', function (e) {
    e.preventDefault();
    var r = stage.getBoundingClientRect();
    var mx = e.clientX - r.left;
    var my = e.clientY - r.top;
    zoomAt(mx, my, e.deltaY < 0 ? 1.12 : 1 / 1.12);
    saveSoon();
  }, { passive: false });

  document.addEventListener('keydown', function (e) {
    if (editingId !== null) { return; }
    if (e.key === 'Delete' || e.key === 'Backspace') {
      if (selectedId !== null) {
        e.preventDefault();
        deleteSelected();
      }
    }
  });

  /* ==================== Tombol ==================== */
  btnAdd.addEventListener('click', addChild);
  btnDel.addEventListener('click', deleteSelected);

  btnZoomIn.addEventListener('click', function () {
    var r = stage.getBoundingClientRect();
    zoomAt(r.width / 2, r.height / 2, 1.2);
    saveSoon();
  });
  btnZoomOut.addEventListener('click', function () {
    var r = stage.getBoundingClientRect();
    zoomAt(r.width / 2, r.height / 2, 1 / 1.2);
    saveSoon();
  });
  btnFit.addEventListener('click', resetView);

  /* Palet warna */
  PALETTE.forEach(function (c) {
    var b = document.createElement('button');
    b.type = 'button';
    b.className = 'swatch';
    b.style.background = c;
    b.setAttribute('data-color', c);
    b.setAttribute('aria-label', 'Warna ' + c);
    b.title = 'Warna ' + c;
    b.addEventListener('click', function () { applyColor(c); });
    colorsEl.appendChild(b);
  });

  /* ==================== Ekspor ==================== */
  function roundRect(ctx, x, y, w, h, r) {
    ctx.beginPath();
    if (typeof ctx.roundRect === 'function') {
      ctx.roundRect(x, y, w, h, r);
      return;
    }
    var rr = Math.min(r, w / 2, h / 2);
    ctx.moveTo(x + rr, y);
    ctx.arcTo(x + w, y, x + w, y + h, rr);
    ctx.arcTo(x + w, y + h, x, y + h, rr);
    ctx.arcTo(x, y + h, x, y, rr);
    ctx.arcTo(x, y, x + w, y, rr);
    ctx.closePath();
  }

  function exportPNG() {
    commitEdit();
    if (!nodes.length) { return; }

    var items = nodes.map(function (n) {
      var el = nodeEl(n.id);
      var w = el ? el.offsetWidth : 130;
      var h = el ? el.offsetHeight : 40;
      return { n: n, w: w, h: h };
    });

    var minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
    items.forEach(function (it) {
      minX = Math.min(minX, it.n.x - it.w / 2);
      maxX = Math.max(maxX, it.n.x + it.w / 2);
      minY = Math.min(minY, it.n.y - it.h / 2);
      maxY = Math.max(maxY, it.n.y + it.h / 2);
    });

    var PAD = 56;
    var cw = maxX - minX + PAD * 2;
    var ch = maxY - minY + PAD * 2;
    var k = 2;
    if (cw * k > 4800 || ch * k > 4800) { k = 1; }

    var canvas = document.createElement('canvas');
    canvas.width = Math.max(1, Math.round(cw * k));
    canvas.height = Math.max(1, Math.round(ch * k));
    var ctx = canvas.getContext('2d');
    if (!ctx) { return; }

    ctx.fillStyle = '#f8fafc';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    ctx.setTransform(k, 0, 0, k, 0, 0);
    ctx.translate(PAD - minX, PAD - minY);

    /* Garis penghubung */
    items.forEach(function (it) {
      var n = it.n;
      if (n.parentId === null) { return; }
      var p = byId(n.parentId);
      if (!p) { return; }
      var sign = (n.x >= p.x) ? 1 : -1;
      var dx = Math.max(50, Math.abs(n.x - p.x) * 0.45) * sign;
      ctx.beginPath();
      ctx.moveTo(p.x, p.y);
      ctx.bezierCurveTo(p.x + dx, p.y, n.x - dx, n.y, n.x, n.y);
      ctx.strokeStyle = n.color;
      ctx.lineWidth = 2.5;
      ctx.lineCap = 'round';
      ctx.stroke();
    });

    /* Simpul */
    items.forEach(function (it) {
      var n = it.n;
      var isRoot = (n.parentId === null);
      var x = n.x - it.w / 2;
      var y = n.y - it.h / 2;

      roundRect(ctx, x, y, it.w, it.h, isRoot ? 16 : 12);
      ctx.fillStyle = isRoot ? n.color : '#ffffff';
      ctx.fill();
      ctx.lineWidth = 2;
      ctx.strokeStyle = n.color;
      ctx.stroke();

      ctx.fillStyle = isRoot ? '#ffffff' : '#0f172a';
      ctx.font = (isRoot ? '700 16px ' : '600 14px ') +
        'system-ui,-apple-system,"Segoe UI",Roboto,sans-serif';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText(n.text, n.x, n.y + 0.5, Math.max(20, it.w - 24));
    });

    var a = document.createElement('a');
    a.download = 'peta-pikiran.png';
    a.href = canvas.toDataURL('image/png');
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
  }

  function exportOutline() {
    commitEdit();
    var root = rootNode();
    if (!root) { return; }
    var lines = [];

    (function walk(id, depth) {
      var n = byId(id);
      if (!n) { return; }
      if (depth === 0) {
        lines.push('# ' + n.text);
      } else {
        lines.push(new Array(depth).join('  ') + '- ' + n.text);
      }
      var kids = childrenOf(id);
      for (var i = 0; i < kids.length; i++) { walk(kids[i].id, depth + 1); }
    })(root.id, 0);

    var text = lines.join('\n') + '\n';
    var blob = new Blob([text], { type: 'text/plain;charset=utf-8' });
    var url = URL.createObjectURL(blob);
    var a = document.createElement('a');
    a.download = 'kerangka-peta-pikiran.txt';
    a.href = url;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    setTimeout(function () { URL.revokeObjectURL(url); }, 2000);
  }

  btnPng.addEventListener('click', exportPNG);
  btnTxt.addEventListener('click', exportOutline);

  /* ==================== Mulai ==================== */
  function boot() {
    var saved = load();
    if (saved && saved.nodes.length) {
      nodes = saved.nodes.filter(function (n) {
        return n && typeof n.id === 'number' && typeof n.text === 'string' &&
               typeof n.x === 'number' && typeof n.y === 'number';
      });
      if (!nodes.length) { nodes = []; }
      var maxId = 0;
      nodes.forEach(function (n) { if (n.id > maxId) { maxId = n.id; } });
      uid = (typeof saved.uid === 'number' && saved.uid > maxId) ? saved.uid : maxId + 1;
      if (saved.pan && typeof saved.pan.x === 'number' && typeof saved.pan.y === 'number') {
        pan = { x: saved.pan.x, y: saved.pan.y };
      }
      if (typeof saved.scale === 'number' && saved.scale > 0) {
        scale = clamp(saved.scale, MIN_SCALE, MAX_SCALE);
      }
      nodes.forEach(function (n) {
        if (!n.color) { n.color = PALETTE[0]; }
        if (n.parentId === null) { n.x = 0; n.y = 0; }
      });
    }

    if (!nodes.length) {
      nodes = [{ id: 1, parentId: null, text: 'Ide Utama', x: 0, y: 0, color: PALETTE[0] }];
      uid = 2;
      renderAll();
      resetView();
      save();
      return;
    }

    renderAll();
    if (saved && saved.pan) {
      applyTransform();
    } else {
      resetView();
    }
  }

  boot();
})();
</script>
</body>
</html>
```

**Cek kebutuhan**

- **Satu ide utama di tengah** → `boot()` membuat simpul awal `{ id: 1, parentId: null, text: 'Ide Utama', x: 0, y: 0 }` lalu `resetView()` memusatkannya di `#stage`.
- **Menambah cabang dan anak cabang** → `addChild()` memakai `selectedId` (atau akar bila tak ada pilihan), lalu `layoutChildren()` menata posisi baru.
- **Mengubah teks dengan klik dua kali** → listener `dblclick` pada `nodesLayer` memanggil `startEdit()`, dengan `commitEdit()` pada Enter, Escape, atau `focusout`.
- **Menghapus cabang beserta anak-anaknya** → `deleteSelected()` mengumpulkan seluruh keturunan lewat fungsi rekursif `collect()` sebelum menyaring `nodes`.
- **Menggeser simpul dengan drag, garis ikut bergerak** → `startDrag()` memperbarui `n.x`/`n.y`, memanggil `placeNode()` dan `renderEdges()` pada setiap `pointermove`.
- **Memberi warna per cabang** → `applyColor()` mewarnai simpul terpilih beserta seluruh keturunannya; palet dibuat dari `PALETTE`.
- **Zoom dan geser kanvas** → `zoomAt()` (tombol +, −, dan `wheel` dengan `preventDefault`) serta `startPan()`; keduanya menulis ulang `world.style.transform` di `applyTransform()`.
- **Ekspor ke PNG** → `exportPNG()` menggambar latar, kurva Bezier, dan kotak bersudut bulat ke `<canvas>` lalu mengunduh via `toDataURL('image/png')`.
- **Ekspor ke teks kerangka bertingkat** → `exportOutline()` menelusuri pohon secara rekursif dan mengunduh `.txt` lewat `Blob` + `URL.createObjectURL`.
- **Data tersimpan di localStorage** → `save()`, `load()`, dan `saveSoon()`, semuanya dibungkus `try/catch`.
- **Responsif sampai 375 px** → `#bar` memakai `flex-wrap` dengan dua `.group`; simpul berlebar maksimum 220 px dan `#hint` berlebar maksimum 92%.
- **Tanpa library/font/gambar eksternal** → seluruh CSS, JS, dan ikon (teks/Unicode) berada di dalam file; tidak ada `src` atau `href` eksternal.

**Catatan teknis**

Koordinat simpul disimpan dalam ruang dunia; `#world` ditransformasi `translate(...) scale(...)` sehingga zoom dan geser tidak mengubah data. Garis digambar pada `<svg>` berukuran 1×1 dengan `overflow: visible`, jadi tidak perlu menghitung ulang kotak pembatas. Ekspor PNG memakai dua kali resolusi (diturunkan ke 1× bila dimensi melebihi 4800 px) agar hasil tetap tajam, dan teks dipotong dengan parameter `maxWidth` pada `fillText` supaya tidak melewati kotak simpul. Interaksi memakai Pointer Events dengan `setPointerCapture`, sehingga drag tetap berjalan di layar sentuh maupun mouse.
