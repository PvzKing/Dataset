  function rgba(hex, a) {
    var n = parseInt(hex.slice(1), 16);
    return 'rgba(' + ((n >> 16) & 255) + ',' + ((n >> 8) & 255) + ',' + (n & 255) + ',' + a + ')';
  }
  function cerah(hex, t) {
    var n = parseInt(hex.slice(1), 16);
    var r = (n >> 16) & 255, g = (n >> 8) & 255, b = n & 255;
    return 'rgb(' + Math.round(r + (255 - r) * t) + ',' +
                    Math.round(g + (255 - g) * t) + ',' +
                    Math.round(b + (255 - b) * t) + ')';

  function hitungZoomAwal() {

  function buatBintang() {
    bintang = [];
    var n = Math.round(batas((W * H) / 2600, 60, 260));
    for (var i = 0; i < n; i++) {
      bintang.push({
        x: Math.random() * W,
        y: Math.random() * H,
        r: Math.random() * 1.1 + 0.25,
        a: Math.random() * 0.55 + 0.22,
        ph: Math.random() * Math.PI * 2
      });
    }
  }

  function ukur() {
    DPR = Math.min(window.devicePixelRatio || 1, 2);
    W = canvas.clientWidth || window.innerWidth;
    H = canvas.clientHeight || window.innerHeight;
    canvas.width = Math.max(1, Math.round(W * DPR));
    canvas.height = Math.max(1, Math.round(H * DPR));
    ctx.setTransform(DPR, 0, 0, DPR, 0, 0);
    buatBintang();
    if (autoFit) {
      cam.x = 0; cam.y = 0;
      cam.zoom = hitungZoomAwal();
    }
  }

  /* ============ Posisi ============ */
  function hitungPosisi() {
    for (var i = 0; i < PLANET.length; i++) {
      var p = PLANET[i];
      var a = p.sudut0 + p.omega * waktu;
      p._x = Math.cos(a) * p.orbit;
      p._y = Math.sin(a) * p.orbit;

    var bumi = PLANET[BULAN.induk];
    var am = BULAN.sudut0 + BULAN.omega * waktu;
    BULAN._x = bumi._x + Math.cos(am) * BULAN.orbit;
    BULAN._y = bumi._y + Math.sin(am) * BULAN.orbit;
  }

      ctx.globalAlpha = s.a * tw;
      ctx.fillStyle = '#cfe0ff';
      ctx.beginPath();
      ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
      ctx.fill();

    ctx.globalAlpha = 1;
  }

    var g = ctx.createRadialGradient(0, 0, 0, 0, 0, r * 4.2);

    g.addColorStop(0.18, 'rgba(255,200,90,0.72)');
    g.addColorStop(0.45, 'rgba(255,150,40,0.18)');
    g.addColorStop(1, 'rgba(255,120,20,0)');
    ctx.fillStyle = g;
    ctx.beginPath();
    ctx.arc(0, 0, r * 4.2, 0, Math.PI * 2);
    ctx.fill();

    var g2 = ctx.createRadialGradient(-r * 0.25, -r * 0.25, r * 0.1, 0, 0, r);
    g2.addColorStop(0, '#fffdf0');
    g2.addColorStop(0.6, '#ffd45e');
    g2.addColorStop(1, '#ff9a1f');
    ctx.fillStyle = g2;
    ctx.beginPath();
    ctx.arc(0, 0, r, 0, Math.PI * 2);
    ctx.fill();
  }

  function gambarCincin(p, depan) {
    var r = p.radius;
    var a0 = depan ? 0 : Math.PI;
    var a1 = depan ? Math.PI : Math.PI * 2;
    var band = [[1.35, 1.62, 0.55], [1.70, 2.15, 0.34]];
    for (var i = 0; i < band.length; i++) {
      var i0 = band[i][0], i1 = band[i][1], al = band[i][2];
      var rr = r * (i0 + i1) / 2;
      ctx.beginPath();
      ctx.ellipse(p._x, p._y, rr, rr * 0.32, 0, a0, a1);
      ctx.lineWidth = r * (i1 - i0);
      ctx.strokeStyle = 'rgba(226,206,160,' + al + ')';
      ctx.stroke();
    }
  }

  function gambarPlanet(p) {
    var x = p._x, y = p._y, r = p.radius;

    if (p.cincin) gambarCincin(p, false);

    var g = ctx.createRadialGradient(x, y, 0, x, y, r * 2.8);
    g.addColorStop(0, rgba(p.warna, 0.30));
    g.addColorStop(1, rgba(p.warna, 0));

    ctx.beginPath();

    b.addColorStop(0, cerah(p.warna, 0.45));
    b.addColorStop(1, p.warna);
    ctx.fillStyle = b;
    ctx.beginPath();
    ctx.arc(x, y, r, 0, Math.PI * 2);

  /* ============ Kamera & interaksi ============ */
  function posLokal(e) {
    var r = canvas.getBoundingClientRect();
    return { x: e.clientX - r.left, y: e.clientY - r.top };
  }

  function zoomPada(sx, sy, faktor) {
    var nz = batas(cam.zoom * faktor, 0.12, 14);
    if (nz === cam.zoom) return;
    var k = nz / cam.zoom;
    cam.x = sx - W / 2 - (sx - W / 2 - cam.x) * k;
    cam.y = sy - H / 2 - (sy - H / 2 - cam.y) * k;
    cam.zoom = nz;
    autoFit = false;
  }

  function keLayar(o) {
    return { x: W / 2 + cam.x + o._x * cam.zoom, y: H / 2 + cam.y + o._y * cam.zoom };
  }

  function cariObjek(sx, sy) {
    var best = null, bestD = Infinity;
    for (var i = 0; i < KANDIDAT.length; i++) {
      var o = KANDIDAT[i];
      var s = keLayar(o);
      var d = Math.hypot(s.x - sx, s.y - sy);
      var tol = Math.max(o.radius * cam.zoom, 5) + 14;
      if (d <= tol && d < bestD) { bestD = d; best = o; }
    }
    return best;
  }

  function pilihDi(sx, sy) {
    var o = cariObjek(sx, sy);
    if (o) {
      terpilih = o;
      tampilkanInfo(o);
    } else {
      sembunyikanInfo();
    }
  }

  function perbaruiKursor(sx, sy) {
    canvas.style.cursor = cariObjek(sx, sy) ? 'pointer' : 'grab';
  }

  /* --- Pointer (mouse + sentuh, termasuk pinch) --- */
  var pointers = new Map();
  var drag = null;
  var pinch = null;

  canvas.addEventListener('pointerdown', function (e) {
    var p = posLokal(e);
    pointers.set(e.pointerId, p);
    try { canvas.setPointerCapture(e.pointerId); } catch (err) { /* diabaikan */ }

    if (pointers.size === 1) {
      drag = { x: p.x, y: p.y, sx: p.x, sy: p.y, gerak: false, mulai: Date.now() };
      pinch = null;
    } else if (pointers.size === 2) {
      drag = null;
      var it = pointers.values();
      var a = it.next().value, b = it.next().value;
      pinch = {
        jarak: Math.hypot(a.x - b.x, a.y - b.y),
        cx: (a.x + b.x) / 2,
        cy: (a.y + b.y) / 2
      };
    }
  });

  canvas.addEventListener('pointermove', function (e) {
    var p = posLokal(e);

    if (!pointers.has(e.pointerId)) {
      perbaruiKursor(p.x, p.y);
      return;
    }
    pointers.set(e.pointerId, p);

    if (pointers.size >= 2 && pinch) {
      var it = pointers.values();
      var a = it.next().value, b = it.next().value;
      var d = Math.hypot(a.x - b.x, a.y - b.y);
      var cx = (a.x + b.x) / 2, cy = (a.y + b.y) / 2;
      if (pinch.jarak > 0 && d > 0) zoomPada(cx, cy, d / pinch.jarak);
      cam.x += cx - pinch.cx;
      cam.y += cy - pinch.cy;
      autoFit = false;
      pinch.jarak = d; pinch.cx = cx; pinch.cy = cy;
    } else if (drag) {
      var dx = p.x - drag.x, dy = p.y - drag.y;
      cam.x += dx;
      cam.y += dy;
      drag.x = p.x;
      drag.y = p.y;
      if (Math.hypot(p.x - drag.sx, p.y - drag.sy) > 6) drag.gerak = true;
      if (drag.gerak) autoFit = false;
    }
  });

  function akhiriPointer(e) {
    var p = pointers.get(e.pointerId);
    pointers.delete(e.pointerId);
    if (canvas.hasPointerCapture && canvas.hasPointerCapture(e.pointerId)) {
      try { canvas.releasePointerCapture(e.pointerId); } catch (err) { /* diabaikan */ }
    }
    if (pointers.size < 2) pinch = null;

    var bisaKlik = !!(p && drag && !drag.gerak && pointers.size === 0 &&
                      (Date.now() - drag.mulai) < 700);
    if (bisaKlik) pilihDi(p.x, p.y);

  function loop(now) {
    var dt = last ? (now - last) / 1000 : 0;
    last = now;
    if (dt > 0.05) dt = 0.05;
    if (dt < 0) dt = 0;

    if (!jeda) waktu += dt * kecepatan;

    hitungPosisi();
    gambar(now);

**Cek kebutuhan**
- **Matahari dan 8 planet mengorbit** → array `PLANET` (8 entri) + `gambarMatahari()` dan `gambarPlanet()` di dalam `gambar()`
- **Kecepatan relatif sesuai periode orbit, skala jarak tidak nyata tetapi urutan benar** → `p.omega = (Math.PI*2) / ((p.periode/365.25) * TAHUN_DETIK)`; nilai `orbit` naik berurutan 85 → 511
- **Bulan mengitari Bumi** → objek `BULAN`, perhitungan di `hitungPosisi()`, penggambaran di `gambarBulan()`, plus garis orbit bulan
- **Cincin Saturnus** → `gambarCincin()` dipanggil dua kali (belahan belakang sebelum badan planet, belahan depan sesudahnya) dengan `p.cincin: true`
- **Slider kecepatan waktu** → `<input id="kecepatan" type="range" min="0.1" max="8" step="0.1">` + listener `slider.addEventListener('input', …)`
- **Tombol jeda** → `#btnJeda` dengan handler `btnJeda.addEventListener('click', …)` yang membalik `jeda`
- **Zoom dengan scroll atau pinch** → `canvas.addEventListener('wheel', …)` dan blok `pinch` pada `pointerdown`/`pointermove`, keduanya memakai `zoomPada()`
- **Klik planet menampilkan info diameter, jarak, periode** → `pilihDi()` → `cariObjek()` → `tampilkanInfo()` mengisi `#infoDiameter`, `#infoJarak`, `#infoPeriode`
- **Responsif sampai 375 px** → media query `@media (max-width:460px)` dan `@media (max-width:620px)`; zoom awal dihitung dari `Math.min(W, H)`
- **Tanpa error JavaScript & akses localStorage dibungkus try/catch** → `simpan()` dan `baca()` memakai `try { window.localStorage… } catch (e) { … }`; semua `releasePointerCapture` juga dibungkus
- **Satu file HTML lengkap, tanpa sumber luar** → seluruh CSS ada di `<style>` dan JS di `<script>` dalam dokumen yang sama, tanpa library, font, atau gambar eksternal

**Catatan teknis**
Skala jarak memakai pemampatan pangkat (`orbit` 85–511 unit) sehingga Neptunus tetap masuk layar pada zoom awal. Rasio kecepatan antarplanet tetap mengikuti periode orbit sebenarnya, tetapi satu tahun Bumi dipetakan ke 8 detik dan periode Bulan dipercepat menjadi ±1,6 detik agar geraknya terlihat. Kecepatan waktu terakhir disimpan di localStorage dan dipulihkan saat halaman dibuka kembali. Interaksi memakai Pointer Events sehingga mouse, sentuh, dan cubit dua jari berbagi satu jalur kode.
