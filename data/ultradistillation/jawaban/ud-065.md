Pengurai ini memakai pendekatan iteratif dengan tumpukan eksplisit, bukan rekursi. Setiap larik atau objek yang terbuka disimpan sebagai bingkai `[wadah, kunci, adalah_objek]`. Begitu satu nilai selesai diurai, nilai itu dimasukkan ke bingkai teratas. Karena itu kedalaman sarang hanya dibatasi memori, dan 50.000 tingkat tidak memicu `RecursionError`.
Untuk kecepatan, bagian yang sering muncul dicocokkan dengan ekspresi reguler yang sudah dikompilasi. Bagian itu adalah whitespace, angka sesuai tata bahasa RFC 8259, dan potongan string tanpa escape. Semua kelas karakter ditulis eksplisit dengan ASCII (`[0-9]`, `[0-9a-fA-F]`), sebab `\d` dan `int(x, 16)` di Python menerima digit Unicode, garis bawah, dan spasi. Bilangan bulat yang sangat panjang dikonversi secara bagi-dua (divide and conquer) supaya tidak terkena batas `sys.int_max_str_digits` di Python 3.11+, tanpa mengubah pengaturan global itu. Kunci kembar ditangani langsung oleh semantik `dict`: penugasan ulang mengganti nilai, sedangkan urutan kunci tetap mengikuti kemunculan pertamanya.

```python
import re
import sys

_WS = re.compile(r'[ \t\n\r]*').match
_ANGKA = re.compile(r'-?(?:0|[1-9][0-9]*)(\.[0-9]+)?([eE][+-]?[0-9]+)?').match
_STRING_CEPAT = re.compile(r'"([^"\\\x00-\x1f]*)"').match
_POTONGAN_STRING = re.compile(r'[^"\\\x00-\x1f]*').match
_HEX4 = re.compile(r'[0-9a-fA-F]{4}').fullmatch
_ESCAPE_SEDERHANA = {
    '"': '"', '\\': '\\', '/': '/',
    'b': '\b', 'f': '\f', 'n': '\n', 'r': '\r', 't': '\t',
}
_LITERAL = {'t': ('true', True), 'f': ('false', False), 'n': ('null', None)}

_batas = getattr(sys, 'get_int_max_str_digits', lambda: 0)()
_AMBANG_DIGIT = (_batas - 1) if _batas else float('inf')


def _digit_ke_int(digit):
    """Ubah deret digit ASCII menjadi int tanpa terkena batas konversi str->int."""
    if len(digit) <= _AMBANG_DIGIT:
        return int(digit)
    separuh = len(digit) // 2
    return _digit_ke_int(digit[:-separuh]) * 10 ** separuh + _digit_ke_int(digit[-separuh:])


def urai_json(teks: str) -> object:
    if not isinstance(teks, str):
        raise TypeError('urai_json membutuhkan argumen bertipe str')

    s = teks
    n = len(s)

    def galat(pesan, pos):
        return ValueError(f'JSON tidak sah: {pesan} (posisi {pos})')

    def baca_string(pos):
        # pos menunjuk tanda kutip pembuka
        m = _STRING_CEPAT(s, pos)
        if m is not None:
            return m.group(1), m.end()
        awal = pos
        pos += 1
        bagian = []
        while True:
            m = _POTONGAN_STRING(s, pos)
            if m.end() > pos:
                bagian.append(m.group())
                pos = m.end()
            if pos >= n:
                raise galat('string tidak ditutup', awal)
            c = s[pos]
            if c == '"':
                return ''.join(bagian), pos + 1
            if c != '\\':
                raise galat('karakter kontrol mentah di dalam string', pos)
            jenis = s[pos + 1:pos + 2]
            if jenis == 'u':
                h = s[pos + 2:pos + 6]
                if not _HEX4(h):
                    raise galat('escape \\u tidak sah', pos)
                kode = int(h, 16)
                pos += 6
                if 0xD800 <= kode <= 0xDBFF and s.startswith('\\u', pos):
                    h2 = s[pos + 2:pos + 6]
                    if _HEX4(h2):
                        kode2 = int(h2, 16)
                        if 0xDC00 <= kode2 <= 0xDFFF:
                            kode = 0x10000 + ((kode - 0xD800) << 10) + (kode2 - 0xDC00)
                            pos += 6
                bagian.append(chr(kode))
            elif jenis in _ESCAPE_SEDERHANA and jenis:
                bagian.append(_ESCAPE_SEDERHANA[jenis])
                pos += 2
            else:
                raise galat('escape tidak sah', pos)

    def baca_kunci(pos):
        # pos sudah melewati whitespace; hasil: (kunci, posisi awal nilai)
        if pos >= n or s[pos] != '"':
            raise galat('kunci objek harus berupa string bertanda kutip', pos)
        kunci, pos = baca_string(pos)
        pos = _WS(s, pos).end()
        if pos >= n or s[pos] != ':':
            raise galat("diharapkan ':' setelah kunci", pos)
        return kunci, _WS(s, pos + 1).end()

    tumpukan = []          # bingkai: [wadah, kunci, adalah_objek]
    pos = _WS(s, 0).end()

    while True:
        # ---- Mengurai satu nilai yang dimulai di pos ----
        if pos >= n:
            raise galat('nilai diharapkan, tetapi teks berakhir', pos)
        c = s[pos]
        if c == '{':
            pos = _WS(s, pos + 1).end()
            if pos < n and s[pos] == '}':
                nilai = {}
                pos += 1
            else:
                kunci, pos = baca_kunci(pos)
                tumpukan.append([{}, kunci, True])
                continue
        elif c == '[':
            pos = _WS(s, pos + 1).end()
            if pos < n and s[pos] == ']':
                nilai = []
                pos += 1
            else:
                tumpukan.append([[], None, False])
                continue
        elif c == '"':
            nilai, pos = baca_string(pos)
        elif c == '-' or '0' <= c <= '9':
            m = _ANGKA(s, pos)
            if m is None:
                raise galat('angka tidak sah', pos)
            lambang = m.group()
            if m.group(1) is not None or m.group(2) is not None:
                nilai = float(lambang)
            elif lambang[0] == '-':
                nilai = -_digit_ke_int(lambang[1:])
            else:
                nilai = _digit_ke_int(lambang)
            pos = m.end()
        elif c in _LITERAL and s.startswith(_LITERAL[c][0], pos):
            kata, nilai = _LITERAL[c]
            pos += len(kata)
        else:
            raise galat(f'karakter tak terduga {c!r}', pos)

        # ---- Menyisipkan nilai yang sudah lengkap ke wadah induknya ----
        while True:
            if not tumpukan:
                pos = _WS(s, pos).end()
                if pos != n:
                    raise galat('ada sisa teks setelah nilai JSON', pos)
                return nilai
            bingkai = tumpukan[-1]
            wadah, kunci, adalah_objek = bingkai
            if adalah_objek:
                wadah[kunci] = nilai
            else:
                wadah.append(nilai)
            pos = _WS(s, pos).end()
            c = s[pos] if pos < n else ''
            if c == ',':
                pos = _WS(s, pos + 1).end()
                if adalah_objek:
                    bingkai[1], pos = baca_kunci(pos)
                break  # lanjut mengurai elemen berikutnya
            penutup = '}' if adalah_objek else ']'
            if c == penutup:
                pos += 1
                tumpukan.pop()
                nilai = wadah
                continue
            raise galat(f"diharapkan ',' atau '{penutup}'", pos)

```

Contoh pemakaian:

```
>>> urai_json('{"a": [1, 2.5, -3e2, true, null], "b": "x\\u00e9\\ud83d\\ude00"}')
{'a': [1, 2.5, -300.0, True, None], 'b': 'xé😀'}
>>> urai_json('{"k": 1, "z": 0, "k": 2}')
{'k': 2, 'z': 0}
>>> urai_json('"\\ud800"') == '\ud800'          # surrogate tunggal dibiarkan
True
>>> type(urai_json('1' * 10000))                 # int besar tanpa batas digit
<class 'int'>
>>> d = urai_json('[' * 50000 + ']' * 50000)     # sarang dalam, tanpa RecursionError
>>> urai_json('[1,]')
ValueError: JSON tidak sah: karakter tak terduga ']' (posisi 3)

```

Kasus khusus yang ditangani:

* Angka: nol di depan (`01`), `+1`, `.5`, `1.`, `1e`, dan `-` saja ditolak. `NaN` dan `Infinity` ditolak sebagai karakter tak terduga.
* Koma di akhir: baik pada larik (`[1,]`) maupun objek (`{"a":1,}`) ditolak.
* Kunci: kunci tanpa tanda kutip, atau kunci yang bukan string, ditolak.
* Isi string: karakter kontrol mentah U+0000–U+001F ditolak. Escape yang tidak dikenal (`\x`, `\'`) dan `\u` dengan kurang dari empat digit heksadesimal ASCII juga ditolak.
* Surrogate: pasangan surrogate tinggi-rendah digabung menjadi satu karakter, sedangkan surrogate tunggal atau yang urutannya terbalik dibiarkan apa adanya.
* Sisa teks: teks setelah nilai puncak ditolak, misalnya `1 2`, `truex`, atau `{}}`. Dokumen kosong atau yang hanya berisi whitespace juga ditolak.
* Whitespace: hanya spasi, tab, LF, dan CR yang diterima. BOM, NBSP, form feed, dan vertical tab dianggap karakter tak terduga.
* Float raksasa: `1e400` menjadi `inf`, sama dengan perilaku modul `json`.
* Bilangan bulat panjang: angka dengan ratusan ribu digit tetap menjadi `int` utuh, tanpa mengubah pengaturan global interpreter.
