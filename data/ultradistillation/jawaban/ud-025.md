Pendekatannya: hitung dulu cabang anak yang berhak—anak hidup atau anak meninggal yang masih punya keturunan berhak. Bagian pasangan disamakan dengan satu cabang anak; sisanya dibagi rata per cabang. Untuk cabang anak yang meninggal, bagian cabangnya dibagi lagi secara rekursif kepada keturunannya, dengan keturunan yang meninggal tanpa ahli waris berhak diabaikan.

```python
from fractions import Fraction

def bagian_waris(pasangan_hidup: bool, anak: list) -> dict[str, Fraction]:
    def punya_keturunan_berhak(daftar: list) -> bool:
        for x in daftar:
            if isinstance(x, str):
                return True
            if isinstance(x, dict):
                if punya_keturunan_berhak(x.get("keturunan", [])):
                    return True
        return False

    def bagi_keturunan(jumlah: Fraction, daftar: list) -> dict[str, Fraction]:
        cabang = []
        for x in daftar:
            if isinstance(x, str):
                cabang.append(x)
            elif isinstance(x, dict):
                if punya_keturunan_berhak(x.get("keturunan", [])):
                    cabang.append(x)

        if not cabang:
            return {}

        bagian = jumlah / len(cabang)
        hasil = {}

        for x in cabang:
            if isinstance(x, str):
                hasil[x] = hasil.get(x, Fraction(0)) + bagian
            else:
                sub = bagi_keturunan(bagian, x.get("keturunan", []))
                for nama, nilai in sub.items():
                    hasil[nama] = hasil.get(nama, Fraction(0)) + nilai

        return hasil

    cabang_anak = []
    for x in anak:
        if isinstance(x, str):
            cabang_anak.append(x)
        elif isinstance(x, dict):
            if punya_keturunan_berhak(x.get("keturunan", [])):
                cabang_anak.append(x)

    jumlah_cabang = len(cabang_anak)

    if jumlah_cabang == 0:
        if pasangan_hidup:
            return {"pasangan": Fraction(1, 1)}
        return {}

    hasil = {}

    if pasangan_hidup:
        bagian_satu_cabang = Fraction(1, jumlah_cabang + 1)
        hasil["pasangan"] = bagian_satu_cabang
    else:
        bagian_satu_cabang = Fraction(1, jumlah_cabang)

    for x in cabang_anak:
        if isinstance(x, str):
            hasil[x] = hasil.get(x, Fraction(0)) + bagian_satu_cabang
        else:
            sub = bagi_keturunan(bagian_satu_cabang, x.get("keturunan", []))
            for nama, nilai in sub.items():
                hasil[nama] = hasil.get(nama, Fraction(0)) + nilai

    return hasil
```

Contoh pemakaian:

```python
bagian_waris(True, ["A", {"nama": "B", "keturunan": ["C", "D"]}])
# {"pasangan": Fraction(1, 3), "A": Fraction(1, 3), "C": Fraction(1, 6), "D": Fraction(1, 6)}

bagian_waris(False, [{"nama": "A", "keturunan": [{"nama": "B", "keturunan": ["C"]}, "D"]}])
# {"C": Fraction(1, 2), "D": Fraction(1, 2)}
```

Kasus khusus yang ditangani: pasangan hidup tanpa anak mendapat seluruh warisan; anak meninggal tanpa keturunan berhak tidak dihitung; penggantian tempat bertingkat sampai cicit ditangani secara rekursif; dan jumlah semua bagian selalu tepat `1`.
