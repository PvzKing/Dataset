```html
  function isAttacked(b, r, c, by) {
    var pd = by === "w" ? 1 : -1;
    var pawn = by === "w" ? "P" : "p";
    var i, rr, cc;

    for (i = -1; i <= 1; i += 2) {
      rr = r + pd; cc = c + i;
      if (inB(rr, cc) && b[idx(rr, cc)] === pawn) return true;
    }

    var knight = by === "w" ? "N" : "n";
    for (i = 0; i < KNIGHT_D.length; i++) {
      rr = r + KNIGHT_D[i][0]; cc = c + KNIGHT_D[i][1];
      if (inB(rr, cc) && b[idx(rr, cc)] === knight) return true;
    }

    var king = by === "w" ? "K" : "k";
    for (i = 0; i < KING_D.length; i++) {
      rr = r + KING_D[i][0]; cc = c + KING_D[i][1];
      if (inB(rr, cc) && b[idx(rr, cc)] === king) return true;
    }

    var rook = by === "w" ? "R" : "r";
    var bishop = by === "w" ? "B" : "b";
    var queen = by === "w" ? "Q" : "q";

    for (i = 0; i < ROOK_D.length; i++) {
      rr = r + ROOK_D[i][0]; cc = c + ROOK_D[i][1];
      while (inB(rr, cc)) {
        var p1 = b[idx(rr, cc)];
        if (p1) {
          if (p1 === rook || p1 === queen) return true;
          break;
        }
        rr += ROOK_D[i][0]; cc += ROOK_D[i][1];
      }
    }

    for (i = 0; i < BISHOP_D.length; i++) {
      rr = r + BISHOP_D[i][0]; cc = c + BISHOP_D[i][1];

          if (p2 === bishop || p2 === queen) return true;
          break;
        }
        rr += BISHOP_D[i][0]; cc += BISHOP_D[i][1];
      }
    }

    return false;
  }

  function findKing(b, color) {
    var k = color === "w" ? "K" : "k";

  function inCheck(b, color) {
    var ki = findKing(b, color);
    if (ki < 0) return false;

  function generatePseudo(st, color) {
    var b = st.board;
    var out = [];
    var i, r, c, t, dr, dc, rr, cc, ti, tp, k;

    for (i = 0; i < 64; i++) {
      var p = b[i];
      if (!p || colorOf(p) !== color) continue;
      r = (i / 8) | 0; c = i % 8; t = p.toUpperCase();

      if (t === "P") {
        var dir = color === "w" ? -1 : 1;
        var startRow = color === "w" ? 6 : 1;
        var promoRow = color === "w" ? 0 : 7;
        var r1 = r + dir;

        if (inB(r1, c) && !b[idx(r1, c)]) {
          if (r1 === promoRow) {
            for (k = 0; k < 4; k++) {
              out.push({ from: i, to: idx(r1, c), capture: false, promotion: ["Q","R","B","N"][k] });
            }
          } else {
            out.push({ from: i, to: idx(r1, c), capture: false });
            var r2 = r + 2 * dir;
            if (r === startRow && !b[idx(r2, c)]) {
              out.push({ from: i, to: idx(r2, c), capture: false });
            }
          }
        }

        for (k = -1; k <= 1; k += 2) {
          cc = c + k;
          if (!inB(r1, cc)) continue;
          ti = idx(r1, cc);
          tp = b[ti];
          if (tp && colorOf(tp) !== color) {
            if (r1 === promoRow) {
              for (var q = 0; q < 4; q++) {
                out.push({ from: i, to: ti, capture: true, promotion: ["Q","R","B","N"][q] });
              }
            } else {
              out.push({ from: i, to: ti, capture: true });
            }
          } else if (!tp && st.ep === ti) {
            out.push({ from: i, to: ti, capture: true, ep: true });
          }
        }
        continue;
      }

      if (t === "N" || t === "K") {
        var dirs = t === "N" ? KNIGHT_D : KING_D;
        for (k = 0; k < dirs.length; k++) {
          rr = r + dirs[k][0]; cc = c + dirs[k][1];
          if (!inB(rr, cc)) continue;
          ti = idx(rr, cc);
          tp = b[ti];
          if (tp && colorOf(tp) === color) continue;
          out.push({ from: i, to: ti, capture: !!tp });
        }
        continue;
      }

      var slide = t === "B" ? BISHOP_D : (t === "R" ? ROOK_D : ALL_D);
      for (k = 0; k < slide.length; k++) {
        rr = r + slide[k][0]; cc = c + slide[k][1];
        while (inB(rr, cc)) {
          ti = idx(rr, cc);
          tp = b[ti];
          if (tp && colorOf(tp) === color) break;
          out.push({ from: i, to: ti, capture: !!tp });
          if (tp) break;
          rr += slide[k][0]; cc += slide[k][1];

  function applyMove(st, m) {
    var s = cloneState(st);
    var b = s.board;
    var piece = b[m.from];
    var color = colorOf(piece);

    b[m.from] = null;

    if (m.ep) {
      var capRow = ((m.to / 8) | 0) + (color === "w" ? 1 : -1);
      b[idx(capRow, m.to % 8)] = null;
    }

    if (m.promotion) {
      b[m.to] = color === "w" ? m.promotion : m.promotion.toLowerCase();
    } else {
      b[m.to] = piece;
    }

    if (m.castle) {
      var row = (m.from / 8) | 0;
      if (m.castle === "K") {
        b[idx(row, 5)] = b[idx(row, 7)];
        b[idx(row, 7)] = null;
      } else {
        b[idx(row, 3)] = b[idx(row, 0)];
        b[idx(row, 0)] = null;
      }
    }

    if (b[60] !== "K") { s.castling.K = false; s.castling.Q = false; }
    if (b[4] !== "k")  { s.castling.k = false; s.castling.q = false; }
    if (b[63] !== "R") s.castling.K = false;
    if (b[56] !== "R") s.castling.Q = false;
    if (b[7] !== "r")  s.castling.k = false;
    if (b[0] !== "r")  s.castling.q = false;

    s.ep = null;
    if (piece.toUpperCase() === "P" && Math.abs(m.to - m.from) === 16) {
      s.ep = (m.from + m.to) / 2;
    }

    s.halfmove = (piece.toUpperCase() === "P" || m.capture) ? 0 : st.halfmove + 1;
    s.fullmove = st.fullmove + (color === "b" ? 1 : 0);
    s.turn = opponent(color);

    return s;
  }

      if (!inCheck(next.board, color)) res.push(pseudo[i]);

  function moveToSAN(st, m, legal) {
    var san;
    var piece = st.board[m.from];
    var type = piece.toUpperCase();

    if (m.castle) {
      san = m.castle === "K" ? "O-O" : "O-O-O";
    } else if (type === "P") {
      san = m.capture ? FILES[m.from % 8] + "x" + sqName(m.to) : sqName(m.to);
      if (m.promotion) san += "=" + m.promotion;
    } else {
      san = type;
      var rivals = legal.filter(function (x) {
        return x.from !== m.from && x.to === m.to && st.board[x.from] === piece;
      });
      if (rivals.length) {
        var sameFile = rivals.some(function (x) { return (x.from % 8) === (m.from % 8); });
        var sameRank = rivals.some(function (x) { return ((x.from / 8) | 0) === ((m.from / 8) | 0); });
        if (!sameFile) san += FILES[m.from % 8];
        else if (!sameRank) san += String(8 - ((m.from / 8) | 0));
        else san += sqName(m.from);
      }
      if (m.capture) san += "x";
      san += sqName(m.to);
    }

    var after = applyMove(st, m);
    if (inCheck(after.board, after.turn)) {
      san += legalMoves(after).length === 0 ? "#" : "+";
    }
    return san;
  }

      sq.type = "button";
      sq.className = "sq " + (((r + c) % 2 === 0) ? "light" : "dark");
      sq.dataset.i = String(i);
      sq.setAttribute("role", "gridcell");
      sq.setAttribute("aria-label", sqName(i));

      if (i === selected) sq.classList.add("sel");
      if (i === checkIdx) sq.classList.add("check");

      if (targets.has(i)) {
        var mv = targets.get(i);
        sq.classList.add((state.board[i] || mv.ep) ? "cap" : "move");
      }

      var p = state.board[i];
      if (p) {
        var sp = document.createElement("span");
        sp.className = "piece " + colorOf(p);
        sp.textContent = GLYPH[p.toUpperCase()];
        sq.appendChild(sp);
      }

      if (c === 0) {
        var rl = document.createElement("span");
        rl.className = "coord rank";
        rl.textContent = String(8 - r);
        sq.appendChild(rl);
      }
      if (r === 7) {
        var fl = document.createElement("span");
        fl.className = "coord file";
        fl.textContent = FILES[c];
        sq.appendChild(fl);
      }

      boardEl.appendChild(sq);
    }
  }

  function renderHistory() {
    historyEl.innerHTML = "";
    if (history.length === 0) {
      var empty = document.createElement("div");
      empty.className = "empty";
      empty.textContent = "Belum ada langkah.";
      historyEl.appendChild(empty);
      return;
    }
    for (var i = 0; i < history.length; i += 2) {
      var li = document.createElement("li");

      var num = document.createElement("span");
      num.className = "num";
      num.textContent = (i / 2 + 1) + ".";
      li.appendChild(num);

      var w = document.createElement("span");
      w.className = "san";
      w.textContent = history[i].san;
      li.appendChild(w);

      var bSpan = document.createElement("span");
      bSpan.className = "san";
      bSpan.textContent = history[i + 1] ? history[i + 1].san : "";
      li.appendChild(bSpan);

      historyEl.appendChild(li);
    }
    historyEl.scrollTop = historyEl.scrollHeight;
  }

  /* ------------------------------------------------------------------ *
   * Penyimpanan lokal (dibungkus try/catch)
   * ------------------------------------------------------------------ */
  function save() {
    try {
      localStorage.setItem(STORE_KEY, JSON.stringify({ state: state, history: history }));
    } catch (e) { /* penyimpanan tidak tersedia */ }
  }

  function load() {
    try {
      var raw = localStorage.getItem(STORE_KEY);
      if (!raw) return null;
      var obj = JSON.parse(raw);
      if (!obj || !obj.state || !Array.isArray(obj.state.board) || obj.state.board.length !== 64) return null;
      if (typeof obj.state.turn !== "string") return null;
      var hist = Array.isArray(obj.history) ? obj.history : [];
      hist = hist.filter(function (h) {
        return h && h.snap && Array.isArray(h.snap.board) && h.snap.board.length === 64 &&
               typeof h.san === "string" && h.move;
      });
      return { state: obj.state, history: hist };
    } catch (e) {
      return null;
    }
  }

  /* ------------------------------------------------------------------ *
   * Aksi permainan
   * ------------------------------------------------------------------ */
  function doMove(m) {
    var san = moveToSAN(state, m, legalCache);
    history.push({ snap: cloneState(state), san: san, move: m });
    state = applyMove(state, m);
    selected = null;
    pendingPromo = null;
    promoEl.classList.add("hidden");
    updateStatus();
    render();
    renderHistory();
    save();
  }

  function askPromotion(cands) {
    pendingPromo = cands;
    var color = state.turn;
    promoBtnsEl.innerHTML = "";
    ["Q", "R", "B", "N"].forEach(function (pr) {
      var btn = document.createElement("button");
      btn.type = "button";
      var sp = document.createElement("span");
      sp.className = "piece " + color;
      sp.textContent = GLYPH[pr];
      btn.appendChild(sp);
      btn.setAttribute("aria-label", "Promosi ke " + pr);
      btn.addEventListener("click", function () {
        var chosen = null;
        for (var i = 0; i < pendingPromo.length; i++) {
          if (pendingPromo[i].promotion === pr) { chosen = pendingPromo[i]; break; }
        }
        pendingPromo = null;
        promoEl.classList.add("hidden");
        if (chosen) doMove(chosen);
      });
      promoBtnsEl.appendChild(btn);
    });
    promoEl.classList.remove("hidden");
  }

  function handleClick(i) {
    if (gameOver || pendingPromo) return;

    if (selected !== null) {
      var cands = legalCache.filter(function (m) {
        return m.from === selected && m.to === i;
      });
      if (cands.length) {
        if (cands.length > 1 && cands[0].promotion) {
          askPromotion(cands);
        } else {
          doMove(cands[0]);
        }
        return;
      }
    }

    var p = state.board[i];
    var canSelect = !!p && colorOf(p) === state.turn &&
      legalCache.some(function (m) { return m.from === i; });

    selected = (canSelect && selected !== i) ? i : null;
    render();
  }

  function undo() {
    if (history.length === 0 || pendingPromo) return;
    var last = history.pop();
    state = cloneState(last.snap);
    selected = null;
    promoEl.classList.add("hidden");
    updateStatus();
    render();
    renderHistory();
    save();
  }

  function reset() {
    state = initialState();
    history = [];
    selected = null;
    pendingPromo = null;
    promoEl.classList.add("hidden");
    updateStatus();
    render();
    renderHistory();
    save();
  }

  /* ------------------------------------------------------------------ *
   * Inisialisasi & pengikatan peristiwa
   * ------------------------------------------------------------------ */
  boardEl.addEventListener("click", function (e) {
    var target = e.target;
    while (target && target !== boardEl && !(target.classList && target.classList.contains("sq"))) {
      target = target.parentNode;
    }
    if (!target || target === boardEl) return;
    handleClick(Number(target.dataset.i));
  });

  undoBtn.addEventListener("click", undo);
  resetBtn.addEventListener("click", reset);

  (function init() {
    var saved = load();
    if (saved) {
      state = saved.state;
      history = saved.history;
    } else {
      state = initialState();
      history = [];
    }
    selected = null;
    pendingPromo = null;
    gameOver = false;
    updateStatus();
    render();
    renderHistory();
  })();
})();
</script>
</body>
</html>
```

**Cek kebutuhan**
- **Permainan catur dua pemain di satu perangkat dalam satu file HTML** → seluruh dokumen `<!DOCTYPE html>` dengan papan, status giliran, dan panel riwayat; pergantian giliran diatur oleh `s.turn = opponent(color)` pada `applyMove`.
- **Tanpa library, font, gambar, atau file dari luar** → hanya elemen HTML, CSS internal pada `<style>`, dan JavaScript internal pada `<script>`; bidak memakai karakter Unicode (`GLYPH`), bukan berkas gambar.
- **Validasi langkah setiap bidak** → `generatePseudo()` menghasilkan langkah per jenis bidak, lalu `legalMoves()` menyaringnya dengan `applyMove()` + `inCheck()`.
- **Rokade** → blok `/* --- Rokade --- */` di `generatePseudo()` memeriksa hak rokade, petak kosong, dan petak yang dilalui tidak diserang; perpindahan benteng ditangani di `applyMove()` pada cabang `if (m.castle)`.
- **En passant** → pencatatan petak sasaran di `s.ep` (saat pion maju dua petak) dan penangkapan `st.ep === ti` di `generatePseudo()`, dengan penghapusan pion lawan pada cabang `if (m.ep)` di `applyMove()`.
- **Promosi pion** → empat langkah promosi (`Q/R/B/N`) dibuat di `generatePseudo()`, dialog pilihan lewat `askPromotion()`, dan penggantian bidak di `applyMove()`.
- **Tandai petak tujuan yang sah saat bidak dipilih** → `render()` membangun `targets` dari `legalCache` lalu menambahkan kelas `move` atau `cap`.
- **Tolak langkah yang membuat raja sendiri terkena skak** → penyaringan `if (!inCheck(next.board, color))` di dalam `legalMoves()`.
- **Deteksi skak** → `inCheck()` dan `isAttacked()`; status "skak!" ditampilkan di `updateStatus()`, petak raja diberi kelas `check` di `render()`.
- **Skakmat** → `legalCache.length === 0` disertai `inCheck` pada `updateStatus()`.
- **Remis karena pat** → `legalCache.length === 0` tanpa skak pada `updateStatus()`.
- **Riwayat langkah dalam notasi aljabar** → `moveToSAN()` (disambiguasi, `x`, `=Q`, `O-O`, sufiks `+`/`#`) dan `renderHistory()`.
- **Tombol membatalkan langkah** → `undo()` memulihkan `snap` dari `history.pop()`, terhubung ke `#undo`.
- **Halaman responsif sampai 375 px** → `max-width:460px`, `aspect-ratio:1/1` pada papan, `font-size:clamp(19px, 8.6vw, 38px)`, dan media query `@media (max-width:400px)`.
- **Tanpa error JavaScript** → kode dibungkus IIFE dengan `"use strict"`, tidak ada referensi ke elemen yang tidak ada, dan penanganan klik memakai penelusuran `parentNode` yang aman.
- **Akses localStorage dibungkus try/catch** → fungsi `save()` dan `load()`.

**Catatan teknis**
Papan disimpan sebagai larik 64 elemen (indeks 0 = a8, 63 = h1) dengan bidak putih huruf besar dan hitam huruf kecil. Pembatalan langkah memakai snapshot keadaan sebelum langkah, sehingga riwayat dapat disimpan dan dipulihkan bersama permainan melalui kunci `catur-dua-pemain-v1`. Aturan tambahan seperti remis 50 langkah, pengulangan tiga kali, dan materi tidak memadai belum diterapkan.
