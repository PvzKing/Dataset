  function fmtDate(ms) {

  }

  }

  function load() {
    try {
      var raw = localStorage.getItem(KEY);
      if (!raw) return null;
      var data = JSON.parse(raw);
      if (!Array.isArray(data)) return null;
      var list = data.map(sanitize).filter(Boolean);
      var ids = new Set(list.map(function (t) { return t.id; }));
      list.forEach(function (t) {
        t.deps = t.deps.filter(function (d) { return ids.has(d) && d !== t.id; });

      return list;
    } catch (e) {
      return null;

  /* ---------- graf & penjadwalan ---------- */
  function topoOrder(list) {
    var map = new Map();
    list.forEach(function (t) { map.set(t.id, t); });
    var state = new Map();
    var order = [];
    var ok = true;

    function visit(id) {
      var s = state.get(id) || 0;
      if (s === 2) return true;
      if (s === 1) return false;
      state.set(id, 1);
      var t = map.get(id);
      if (t) {
        for (var i = 0; i < t.deps.length; i++) {
          var d = t.deps[i];
          if (map.has(d) && !visit(d)) return false;
        }
      }
      state.set(id, 2);
      order.push(id);
      return true;
    }

    for (var i = 0; i < list.length; i++) {
      if (!visit(list[i].id)) { ok = false; break; }
    }
    if (!ok) return null;
    return order.map(function (id) { return map.get(id); });
  }

  /* Menggeser tanggal mulai agar tidak mendahului selesainya tugas prasyarat. */
  function normalize() {
    var order = topoOrder(tasks);
    if (!order) return 0;
    var endMap = new Map();
    var shifted = 0;

    order.forEach(function (t) {
      var earliest = null;
      t.deps.forEach(function (d) {
        if (!endMap.has(d)) return;
        var cand = endMap.get(d) + DAY;
        if (earliest === null || cand > earliest) earliest = cand;
      });
      var s = parseDate(t.start);
      if (isNaN(s)) s = todayUTC();
      if (earliest !== null && s < earliest) {
        t.start = fmtDate(earliest);
        shifted++;
      }
      endMap.set(t.id, parseDate(t.start) + (Math.max(1, t.duration) - 1) * DAY);
    });

    return shifted;
  }

  function buildRows() {
    var order = topoOrder(tasks) || tasks.slice();
    var rows = order.map(function (t) {
      var s = parseDate(t.start);
      if (isNaN(s)) s = todayUTC();
      var e = s + (Math.max(1, t.duration) - 1) * DAY;
      return { t: t, s: s, e: e };
    });
    var pos = new Map();
    tasks.forEach(function (t, i) { pos.set(t.id, i); });
    rows.sort(function (a, b) { return pos.get(a.t.id) - pos.get(b.t.id); });
    return rows;
  }

  function createsCycle(taskId, depIds) {
    var byId = new Map();
    tasks.forEach(function (t) { byId.set(t.id, t); });

    function depsOf(id) {
      if (id === taskId) return depIds;
      var t = byId.get(id);
      return t ? t.deps : [];
    }

    var seen = new Set();
    var stack = depIds.slice();
    while (stack.length) {
      var id = stack.pop();
      if (id === taskId) return true;
      if (seen.has(id)) continue;
      seen.add(id);
      depsOf(id).forEach(function (d) { stack.push(d); });
    }
    return false;
  }

  /* ---------- render ---------- */
  function updateSummary(rows) {
    var today = todayUTC();
    var done = 0, late = 0;
    rows.forEach(function (r) {
      if (r.t.progress >= 100) done++;
      else if (r.e < today) late++;
    });
    summaryEl.textContent = rows.length + ' tugas · ' + done + ' selesai · ' + late + ' terlambat';
  }

  function render() {
    var rows = buildRows();
    updateSummary(rows);

    if (!rows.length) {
      inner.classList.add('is-empty');
      inner.innerHTML = '<div class="empty">Belum ada tugas. Klik <strong>+ Tugas</strong> untuk menambah, atau muat data contoh.</div>';
      return;
    }
    inner.classList.remove('is-empty');

    var today = todayUTC();
    var minMs = Math.min.apply(null, rows.map(function (r) { return r.s; }));
    var maxMs = Math.max.apply(null, rows.map(function (r) { return r.e; }));

    var todayIn = (today >= minMs - 10 * DAY && today <= maxMs + 10 * DAY);
    if (todayIn) {
      minMs = Math.min(minMs, today);
      maxMs = Math.max(maxMs, today);
    }
    minMs -= 2 * DAY;
    maxMs += 2 * DAY;

    var days = Math.round((maxMs - minMs) / DAY) + 1;
    inner.style.setProperty('--days', String(days));

    var pct = function (n) { return (n / days * 100); };

    /* header bulan */
    var months = '';
    var i = 0;
    while (i < days) {
      var d0 = new Date(minMs + i * DAY);
      var j = i + 1;
      while (j < days) {
        var d1 = new Date(minMs + j * DAY);
        if (d1.getUTCMonth() !== d0.getUTCMonth() || d1.getUTCFullYear() !== d0.getUTCFullYear()) break;
        j++;
      }
      months += '<div class="month" style="width:' + pct(j - i) + '%">' +
                MONTHS_SHORT[d0.getUTCMonth()] + ' ' + d0.getUTCFullYear() + '</div>';
      i = j;
    }

    /* header tanggal + kolom latar */
    var dcells = '';
    var gcols = '';
    for (var k = 0; k < days; k++) {
      var ms = minMs + k * DAY;
      var dd = new Date(ms);
      var dow = dd.getUTCDay();
      var we = (dow === 0 || dow === 6);
      var isToday = (ms === today);
      dcells += '<div class="dcell' + (we ? ' we' : '') + (isToday ? ' today' : '') + '">' +
                dd.getUTCDate() + '</div>';
      gcols += '<div class="gcol' + (we ? ' we' : '') + '"></div>';
    }

    /* baris tugas */
    var rowsHtml = rows.map(function (r) {
      var t = r.t;
      var prog = clamp(Math.round(t.progress), 0, 100);
      var dur = Math.max(1, t.duration);
      var late = prog < 100 && r.e < today;
      var cls = prog >= 100 ? 'done' : (late ? 'late' : (r.s <= today ? 'active' : 'todo'));
      var status = cls === 'done' ? 'Selesai' : cls === 'late' ? 'Terlambat' : cls === 'active' ? 'Berjalan' : 'Belum mulai';
      var left = (Math.round((r.s - minMs) / DAY) / days) * 100;
      var width = (dur / days) * 100;
      var tip = t.name + ' • ' + fmtHuman(r.s) + ' – ' + fmtHuman(r.e) +
                ' • ' + (t.pic || 'tanpa PIC') + ' • ' + prog + '% • ' + status;

      return '<div class="row clickable" data-id="' + esc(t.id) + '">' +
               '<div class="lbl ' + cls + '">' +
                 '<div class="tname">' + esc(t.name) + '</div>' +
                 '<div class="tmeta">' + esc(t.pic || '—') + ' · ' + dur + ' hari · ' + prog + '%</div>' +
               '</div>' +
               '<div class="tl">' +
                 '<div class="bar ' + cls + '" style="left:' + left + '%;width:' + width + '%" title="' + esc(tip) + '">' +
                   '<div class="fill" style="width:' + prog + '%"></div>' +
                 '</div>' +
               '</div>' +
             '</div>';
    }).join('');

    var todayLine = '';
    if (todayIn) {
      todayLine = '<div class="today-line" style="left:' + pct(Math.round((today - minMs) / DAY) + 0.5) + '%"></div>';
    }

    inner.innerHTML =
      '<div class="grid-bg">' + gcols + '</div>' +
      '<div class="row head">' +
        '<div class="lbl">Tugas</div>' +
        '<div class="tl">' +
          '<div class="months">' + months + '</div>' +
          '<div class="days">' + dcells + '</div>' +
        '</div>' +
      '</div>' +
      rowsHtml +
      '<div class="today-layer">' + todayLine + '</div>';
  }

  /* ---------- form ---------- */
  function openForm(id) {
    editingId = id || null;
    var t = id ? tasks.filter(function (x) { return x.id === id; })[0] : null;

    $('modalTitle').textContent = t ? 'Ubah tugas' : 'Tugas baru';
    fName.value = t ? t.name : '';
    fPic.value = t ? t.pic : '';
    fDur.value = t ? t.duration : 3;
    fStart.value = t ? t.start : fmtDate(todayUTC());
    fProg.value = t ? t.progress : 0;

    var others = tasks.filter(function (x) { return x.id !== editingId; });
    if (!others.length) {
      fDeps.innerHTML = '<div class="none">Belum ada tugas lain yang bisa dijadikan prasyarat.</div>';
    } else {
      fDeps.innerHTML = others.map(function (o) {
        var checked = (t && t.deps.indexOf(o.id) !== -1) ? ' checked' : '';
        return '<label class="dep"><input type="checkbox" value="' + esc(o.id) + '"' + checked + '>' +
               '<span>' + esc(o.name) + '</span></label>';
      }).join('');
    }

    delBtn.hidden = !t;
    errEl.hidden = true;
    errEl.textContent = '';
    modal.hidden = false;
    setTimeout(function () { fName.focus(); }, 30);
  }

  function closeForm() {
    modal.hidden = true;
    editingId = null;
  }

  function showErr(msg) {
    errEl.textContent = msg;
    errEl.hidden = false;
  }

  /* ---------- aksi ---------- */
  $('addBtn').addEventListener('click', function () { openForm(null); });
  $('cancelBtn').addEventListener('click', closeForm);
  modal.addEventListener('click', function (e) { if (e.target === modal) closeForm(); });
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && !modal.hidden) closeForm();
  });

  inner.addEventListener('click', function (e) {
    var row = e.target.closest ? e.target.closest('.row[data-id]') : null;
    if (row) openForm(row.getAttribute('data-id'));
  });

  $('resetBtn').addEventListener('click', function () {
    if (!window.confirm('Muat ulang data contoh? Perubahan Anda akan hilang.')) return;
    tasks = sampleTasks();
    normalize();
    save();
    render();
    toast('Data contoh proyek renovasi kantor dimuat.');
  });

  delBtn.addEventListener('click', function () {
    if (!editingId) return;
    if (!window.confirm('Hapus tugas ini?')) return;
    var id = editingId;
    tasks = tasks.filter(function (t) { return t.id !== id; });
    tasks.forEach(function (t) {
      t.deps = t.deps.filter(function (d) { return d !== id; });
    });
    normalize();
    save();
    render();
    closeForm();
    toast('Tugas dihapus.');
  });

  form.addEventListener('submit', function (e) {
    e.preventDefault();

    var name = fName.value.trim();
    if (!name) { showErr('Nama tugas wajib diisi.'); return; }

    var dur = parseInt(fDur.value, 10);
    if (isNaN(dur)) dur = 1;
    dur = clamp(dur, 1, 365);

    var prog = parseInt(fProg.value, 10);
    if (isNaN(prog)) prog = 0;
    prog = clamp(prog, 0, 100);

    var start = fStart.value;
    if (!/^\d{4}-\d{2}-\d{2}$/.test(start) || isNaN(parseDate(start))) {
      showErr('Tanggal mulai tidak valid.');
      return;
    }

    var deps = Array.prototype.slice.call(fDeps.querySelectorAll('input[type="checkbox"]'))
      .filter(function (c) { return c.checked; })
      .map(function (c) { return c.value; });

    if (createsCycle(editingId, deps)) {
      showErr('Ketergantungan melingkar ditolak: tugas ini tidak boleh menunggu tugas yang (langsung atau tidak langsung) menunggunya.');
      return;
    }

    var pic = fPic.value.trim().slice(0, 40);

    if (editingId) {
      var t = tasks.filter(function (x) { return x.id === editingId; })[0];
      if (t) {
        t.name = name.slice(0, 80);
        t.pic = pic;
        t.start = start;
        t.duration = dur;
        t.progress = prog;
        t.deps = deps;
      }
    } else {
      tasks.push({
        id: uid(), name: name.slice(0, 80), pic: pic, start: start,
        duration: dur, progress: prog, deps: deps
      });
    }

    var diubah = editingId !== null;
    var shifted = normalize();
    save();
    render();
    closeForm();

    if (shifted) {
      toast('Tanggal mulai ' + shifted + ' tugas digeser otomatis mengikuti ketergantungan.');
    } else {
      toast(diubah ? 'Tugas diperbarui.' : 'Tugas ditambahkan.');
    }
  });

  /* ---------- init ---------- */
  var stored = load();
  if (stored === null) {
    tasks = sampleTasks();
    normalize();
    save();
  } else {
    tasks = stored;
    normalize();
  }
  render();
})();
</script>
</body>
</html>

**Cek kebutuhan**
- **Setiap tugas punya nama, penanggung jawab, tanggal mulai, durasi (hari), persentase progres, dan daftar prasyarat** → objek tugas pada `sanitize()` dan `sampleTasks()`, ditampilkan di `.lbl` dan `.bar`.
- **Batang tiap tugas pada skala hari** → `render()` menghitung `--days`, `.tl` berlebar `calc(var(--days) * var(--day-w))`, dan `.bar` diposisikan dengan `left`/`width` persentase.
- **Garis hari ini** → `.today-line` di dalam `.today-layer`, serta penanda `.dcell.today` pada header tanggal.
- **Mewarnai tugas terlambat** → `late = prog < 100 && r.e < today`, kelas `.bar.late` / `.lbl.late` (merah).
- **Menolak ketergantungan melingkar** → fungsi `createsCycle()` dipanggil pada `submit`, dengan pesan galat di `#formErr`.
- **Prasyarat harus selesai lebih dulu** → `normalize()` menggeser tanggal mulai ke hari setelah prasyarat terpanjang berakhir, lalu memberi notifikasi lewat `toast()`.
- **Data tersimpan di localStorage dengan try/catch** → `save()` dan `load()` dibungkus `try { ... } catch (e) { ... }`.
- **Data contoh renovasi kantor** → `sampleTasks()` berisi 10 tugas berantai (survei → desain → pengadaan → instalasi → finishing → serah terima).
- **Responsif sampai 375 px** → media query `max-width:600px` dan `max-width:380px`, lebar label/hari memakai variabel CSS, tabel dapat digeser horizontal dengan kolom nama yang sticky.
- **Tanpa library, font, gambar, atau berkas luar** → seluruh CSS dan JavaScript inline, hanya `system-ui` dan elemen HTML.

**Catatan teknis**
- Perhitungan tanggal memakai UTC (`Date.UTC`) agar durasi harian tidak bergeser karena zona waktu atau DST.
- `normalize()` hanya menggeser tanggal ke depan, tidak pernah memundurkannya; jika prasyarat dihapus, tanggal mulai tetap seperti hasil penyesuaian terakhir.
- Rentang grafik dibatasi: garis hari ini hanya digambar bila hari ini berada dalam rentang proyek ±10 hari, supaya kanvas tidak melebar tanpa guna.
- Semua data divalidasi ulang saat dimuat (`sanitize`), termasuk membuang ID prasyarat yang sudah tidak ada agar graf tetap konsisten.
